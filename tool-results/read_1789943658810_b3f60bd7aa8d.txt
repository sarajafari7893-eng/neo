   278→   718→function sqnOf(pnls: number[]): { sqn: number; sqnTrim: number } {
   279→   719→  const n = pnls.length;
   280→   720→  const mean = pnls.reduce((a, b) => a + b, 0) / n;
   281→   721→  const sd = Math.sqrt(pnls.reduce((a, b) => a + (b - mean) * (b - mean), 0) / n); // population (ddof 0)
   282→   722→  const sqn = sd > 0 ? Math.sqrt(n) * (mean / sd) : 0;
   283→   723→  const sorted = [...pnls].sort((a, b) => a - b);
   284→   724→  const k = Math.floor(n * 0.1);
   285→   725→  const trim = k > 0 && n - 2 * k >= 5 ? sorted.slice(k, n - k) : sorted;
   286→   726→  const tm = trim.reduce((a, b) => a + b, 0) / trim.length;
   287→   727→  const tsd = Math.sqrt(trim.reduce((a, b) => a + (b - tm) * (b - tm), 0) / trim.length);
   288→   728→  const sqnTrim = tsd > 0 ? Math.sqrt(trim.length) * (tm / tsd) : 0;
   289→   729→  return { sqn: Number(sqn.toFixed(2)), sqnTrim: Number(sqnTrim.toFixed(2)) };
   290→   730→}

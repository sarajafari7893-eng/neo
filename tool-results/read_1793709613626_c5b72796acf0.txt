   153→async function measureTakerSweep(symbol: string): Promise<TakerSweep | null> {

   181→    sellUsd: Math.round(sellUsd),

   183→    sweepUsd: Math.round(dominant),
   184→    dominance: total > 0 ? dominant / total : 0,
   185→  };
   186→  measCache.set(ck, { at: Date.now(), v: res });

   202→/** L1.5 + L2 — order-book imbalance + ±2% vacuum/wall map from one depth snapshot. */
   203→async function measureDepth(symbol: string): Promise<DepthMeasure | null> {
   204→  const ck = `depth:${symbol}`;
   205→  const hit = measCache.get(ck);
   206→  if (hit && Date.now() - hit.at < MEAS_TTL) return hit.v as DepthMeasure | null;
   207→  // v12.3 TASK-52 (v14.0): depth rides the ladder — fapi (w5) → SPOT-MIRROR book →
   208→  // SPOT-MIRROR book; OBI/vacuum math is venue-shape-agnostic (bids DESC / asks ASC).
   209→  const d = await bridgeDepth(symbol, 500);
   210→  if (!d || d.bids.length === 0 || d.asks.length === 0) {
   211→    measCache.set(ck, { at: Date.now(), v: null });
   212→    return null;
   213→  }
   214→  const bids: DepthLevel[] = d.bids.map(([p, q]) => ({ price: p, qty: q }));
   215→  const asks: DepthLevel[] = d.asks.map(([p, q]) => ({ price: p, qty: q }));
   216→  const mid = (bids[0]!.price + asks[0]!.price) / 2;
   217→
   218→  // OBI over the top ±0.5% band
   219→  let bidNot = 0;
   220→  let askNot = 0;
   221→  for (const b of bids) if (b.price >= mid * 0.995) bidNot += b.price * b.qty; else break;
   222→  for (const a of asks) if (a.price <= mid * 1.005) askNot += a.price * a.qty; else break;
   223→  const obi = askNot > 0 ? bidNot / askNot : 1;
   224→
   225→  // ±2% map: find the widest liquidity gap (vacuum) and the biggest single wall
   226→  const band = 0.02;
   227→  const buckets = 40;
   228→  const bucketUsd = new Array<number>(buckets).fill(0);
   229→  const push = (price: number, qty: number) => {
   230→    const dev = (price - mid) / mid;
   231→    if (Math.abs(dev) > band) return;
   232→    const idx = Math.min(buckets - 1, Math.max(0, Math.floor(((dev + band) / (2 * band)) * buckets)));
   233→    bucketUsd[idx]! += price * qty;
   234→  };
   235→  for (const b of bids) push(b.price, b.qty);
   236→  for (const a of asks) push(a.price, a.qty);
   237→
   238→  const nonZero = bucketUsd.filter((v) => v > 0);
   239→  const median = nonZero.length ? [...nonZero].sort((a, b) => a - b)[Math.floor(nonZero.length / 2)]! : 0;
   240→
   241→  let vacuum: { side: "ABOVE" | "BELOW"; widthPct: number; edgeDistPct: number } | null = null;
   242→  if (median > 0) {
   243→    let bestStart = -1;
   244→    let bestLen = 0;
   245→    let curStart = -1;
   246→    for (let i = 0; i < buckets; i++) {
   247→      const empty = bucketUsd[i]! < median * 0.15;
   248→      if (empty && curStart < 0) curStart = i;
   249→      if ((!empty || i === buckets - 1) && curStart >= 0) {
   250→        const len = (empty ? i + 1 : i) - curStart;
   251→        if (len > bestLen) {
   252→          bestLen = len;
   253→          bestStart = curStart;
   254→        }
   255→        curStart = -1;
   256→      }
   257→    }
   258→    if (bestLen >= 4) {
   259→      const midIdx = bestStart + bestLen / 2;
   260→      const devMid = (midIdx / buckets) * 2 * band - band;
   261→      const edgeDev = (bestStart / buckets) * 2 * band - band;
   262→      vacuum = {
   263→        side: devMid >= 0 ? "ABOVE" : "BELOW",
   264→        widthPct: Number(((bestLen / buckets) * 2 * band * 100).toFixed(2)),
   265→        edgeDistPct: Number((Math.abs(edgeDev) * 100).toFixed(2)),
   266→      };
   267→    }
   268→  }
   269→
   270→  // wall: biggest single level within ±2%
   271→  let wall: { side: "BID" | "ASK"; priceDistPct: number; usd: number } | null = null;
   272→  for (const b of bids) {
   273→    const usd = b.price * b.qty;
   274→    if (!wall || usd > wall.usd) wall = { side: "BID", priceDistPct: Number((((mid - b.price) / mid) * 100).toFixed(2)), usd: Math.round(usd) };
   275→  }
   276→  for (const a of asks) {
   277→    const usd = a.price * a.qty;
   278→    if (!wall || usd > wall.usd) wall = { side: "ASK", priceDistPct: Number((((a.price - mid) / mid) * 100).toFixed(2)), usd: Math.round(usd) };
   279→  }
   280→
   281→  const res = { mid: Number(mid.toFixed(mid > 100 ? 2 : 6)), obi: Number(obi.toFixed(3)), vacuum, wall };
   282→  measCache.set(ck, { at: Date.now(), v: res });
   283→  return res;
   284→}

   324→/** Restore counters/ring after a dev-server restart.
   325→ * TASK-40 hardening: a TRANSIENT DB error no longer latches the restore —
   326→ * failed restores retry on the next cycle and persist() stays denied until
   327→ * this succeeds. Counters merge MAX (they are monotonic), the ring merges
   328→ * deduped by at|layer|symbol: a double-restore or a racing generation can
   329→ * never move the visible numbers backwards. */
   330→let restoreOk = false;
   331→async function restore(): Promise<void> {
   332→  if (restoreOk) return;
   333→  try {
   334→    const row = await db.engineSetting.findUnique({ where: { key: PERSIST_KEY } });
   335→    if (!row) {
   336→      restoreOk = true; // confirmed fresh — first persist may create the row
   337→      return;
   338→    }
   339→    const v = JSON.parse(row.value) as { counts?: Record<TriggerLayer, number>; last?: TriggerEvent[]; lastCycleAt?: number | null; lastSpawnAt?: number };
   340→    // layer-set evolution guard + MAX-merge — a layer added after the persist
   341→    // was written would otherwise sit at undefined and NaN on the first ++
   342→    for (const k of Object.keys(S.counts) as TriggerLayer[]) {
   343→      const disk = Number(v.counts?.[k] ?? 0);
   344→      S.counts[k] = Math.max(S.counts[k] ?? 0, Number.isFinite(disk) ? disk : 0);
   345→    }
   346→    const seen = new Set(S.ring.map((e) => `${e?.at}|${e?.layer}|${e?.symbol}`));
   347→    for (const e of (v.last ?? []).slice(0, RING_CAP)) {
   348→      if (!e || typeof e.at !== "number") continue;
   349→      const k = `${e.at}|${e.layer}|${e.symbol}`;
   350→      if (seen.has(k)) continue;
   351→      seen.add(k);
   352→      S.ring.push(e);
   353→    }
   354→    S.ring.sort((a, b) => b.at - a.at);
   355→    if (S.ring.length > RING_CAP) S.ring.length = RING_CAP;
   356→    if (typeof v.lastCycleAt === "number") S.lastCycleAt = Math.max(S.lastCycleAt ?? 0, v.lastCycleAt);
   357→    S.lastSpawnAt = Math.max(S.lastSpawnAt, v.lastSpawnAt ?? 0);
   358→    restoreOk = true;
   359→  } catch {
   360→    /* transient DB hiccup — NOT latched; next cycle retries; persist stays denied */
   361→  }
   362→}

   497→const CYCLE_TIMEOUT_MS = 60_000;
   498→const SYMBOL_TIMEOUT_MS = 15_000;

   500→function withTimeout<T>(p: Promise<T>, ms: number, label: string): Promise<T> {
   501→  return new Promise<T>((resolve, reject) => {
   502→    const t = setTimeout(() => reject(new Error(`${label} timeout ${ms}ms`)), ms);
   503→    p.then((v) => {
   504→      clearTimeout(t);
   505→      resolve(v);
   506→    }).catch((e) => {
   507→      clearTimeout(t);
   508→      reject(e);
   509→    });
   510→  });
   511→}

   518→async function runTriggerCycleInner(symbols: string[]): Promise<void> {
   519→  await restore();
   520→  rotationSize = Math.max(6, Math.min(64, symbols.length));
   521→  const batch = [symbols[cursor % symbols.length], symbols[(cursor + 1) % symbols.length], symbols[(cursor + 2) % symbols.length]].filter(Boolean) as string[];
   522→  cursor = (cursor + 3) % symbols.length;
   523→  for (const s of batch) {
   524→    await withTimeout(scanSymbolTriggers(s), SYMBOL_TIMEOUT_MS, `trigger scan ${s}`);
   525→  }

   527→}

   529→export async function runTriggerCycle(symbols: string[]): Promise<void> {
   530→  if (cycleInFlight || symbols.length === 0) return;
   531→  cycleInFlight = true;
   532→  cyclesStarted++;
   533→  const t0 = Date.now();
   534→  try {
   535→    await withTimeout(runTriggerCycleInner(symbols), CYCLE_TIMEOUT_MS, "trigger cycle");
   536→    cyclesCompleted++;
   537→    lastCycleMs = Date.now() - t0;
   538→    lastCycleError = null;
   539→  } catch (err) {
   540→    lastCycleError = String(err);
   541→    console.error("[trigger-bus] cycle failed:", err);
   542→  } finally {
   543→    cycleInFlight = false;
   544→  }
   545→}

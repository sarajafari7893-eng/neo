    25→function ago(ts: number): string {
    26→  const s = Math.max(0, Math.round((Date.now() - ts) / 1000));
    27→  if (s < 60) return `${s}s`;
    28→  if (s < 3600) return `${Math.round(s / 60)}m`;
    29→  if (s < 86400) return `${Math.round(s / 3600)}h`;
    30→  return `${Math.round(s / 86400)}d`;
    31→}
    32→
    33→function usd(v: number): string {

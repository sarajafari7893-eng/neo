   404→   404→async function doOpen(it: Intent): Promise<void> {
   405→   405→  const liveMode = isLiveMode();
   406→   406→  it.mode = liveMode ? "LIVE" : "PAPER";
   407→   407→  const tick = await tbTicker(it.tbSymbol);
   408→   408→  if (!tick) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_TICKER", note: it.tbSymbol }); audit("EXEC-FAIL", { serial: it.serial, why: "no toobit ticker" }); return; }
   409→   409→  const instr = instruments.get(it.tbSymbol);
   410→   410→  if (!instr) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_INSTRUMENT" }); audit("EXEC-FAIL", { serial: it.serial, why: `symbol absent on toobit: ${it.tbSymbol}` }); return; }
   411→   411→  const refPrice = it.direction === "LONG" ? tick.ask : tick.bid;
   412→   412→  // v1.1.0 FIXED-$5 MARGIN (margin never varies per signal) + v1.2.0 PER-CARD
   413→   413→  // LEVERAGE (operator mandate trace 1a15ddbdbabf84a9): notional = fixed margin
   414→   414→  // × THE CARD'S leverage — size and risk mirror the signal card exactly.
   415→   415→  const notional = S.cfg.marginUsdt * it.leverage;
   416→   416→  it.expectedNotional = Number(notional.toFixed(6));
   417→   417→  let qty = notional / refPrice;
   418→   418→
   419→   419→  // v1.3.0 ENTRY GATES (operator mandate trace 1a15f7d945d27ead):
   420→   420→  // ① ENTRY-DELAY — فاصله زمانی با تأخیر: a card whose entry window has passed
   421→   421→  //    is no longer the signal the card described; chasing it trades a stale edge.
   422→   422→  const ageAtEntry = Date.now() - (it.createdAt0 ?? Date.now());
   423→   423→  it.entryAgeMs = ageAtEntry;
   424→   424→  if (ageAtEntry > S.cfg.maxEntryAgeMs) {
   425→   425→    it.status = "SKIP";
   426→   426→    it.events.push({ at: Date.now(), ev: "ENTRY-DELAY", note: `age=${Math.round(ageAtEntry / 1000)}s > cap ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s` });
   427→   427→    countSkip("ENTRY-DELAY");
   428→   428→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-delay: card age ${Math.round(ageAtEntry / 1000)}s exceeds ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s window` });
   429→   429→    return;
   430→   430→  }
   431→   431→  // ② ENTRY-DRIFT — اختلاف قیمتی بالا: if the book has already run away from
   432→   432→  //    the card's entry, the fill is not the card's trade — skip honestly.
   433→   433→  const driftBps = Math.abs(refPrice - it.cardEntry) / it.cardEntry * 10_000;
   434→   434→  it.entryDriftBps = Number(driftBps.toFixed(2));
   435→   435→  if (it.cardEntry > 0 && driftBps > S.cfg.maxEntryDriftBps) {
   436→   436→    it.status = "SKIP";
   437→   437→    it.events.push({ at: Date.now(), ev: "ENTRY-DRIFT", note: `drift=${driftBps.toFixed(0)}bps > cap ${S.cfg.maxEntryDriftBps}bps (ref=${refPrice} vs card=${it.cardEntry})` });
   438→   438→    countSkip("ENTRY-DRIFT");
   439→   439→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-drift: ${driftBps.toFixed(0)}bps from card entry exceeds ${S.cfg.maxEntryDriftBps}bps cap` });
   440→   440→    return;
   441→   441→  }
   442→   442→  // $5-fixed mandate: a symbol whose exchange floor (minNotional / minQty×price)
   443→   443→  // cannot fit inside the mandated margin notional is SKIPPED honestly —
   444→   444→  // silently upsizing would breach the fixed-margin contract.
   445→   445→  const floorNotional = Math.max(instr.minNotional, instr.minQty * refPrice);
   446→   446→  if (floorNotional > notional) {
   447→   447→    it.status = "SKIP";
   448→   448→    it.events.push({ at: Date.now(), ev: "MIN_FLOOR_SKIP", note: `${it.tbSymbol} floor=${floorNotional.toFixed(2)} > marginNotional=${notional}` });
   449→   449→    countSkip("MIN-FLOOR");
   450→   450→    audit("EXEC-SKIP", { serial: it.serial, why: `exchange floor ${floorNotional.toFixed(2)} USDT exceeds fixed-margin notional ${notional} USDT` });
   451→   451→    return;
   452→   452→  }
   453→   453→  qty = floorStep(Math.min(qty, instr.maxQty), instr.stepSize);
   454→   454→  if (qty < instr.minQty) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "QTY_BELOW_MIN" }); audit("EXEC-FAIL", { serial: it.serial, why: `qty ${qty} < minQty ${instr.minQty}` }); return; }
   455→   455→
   456→   456→  const tSent = Date.now();
   457→   457→  let fillPrice: number | null = null;
   458→   458→  let orderId: string | null = null;
   459→   459→  if (liveMode) {
   460→   460→    await ensureLeverage(it.tbSymbol, it.leverage);
   461→   461→    const side = it.direction === "LONG" ? "BUY_OPEN" : "SELL_OPEN";
   462→   462→    const r = await placeOrder(it.tbSymbol, side, qty, { takeProfit: it.tp3, stopLoss: it.stopLoss, clientOrderId: it.clientOrderId });
   463→   463→    audit("EXEC-OPEN", { serial: it.serial, symbol: it.tbSymbol, side, qty, lev: it.leverage, tp3: it.tp3, sl: it.stopLoss, ok: r.ok, orderId: r.orderId, err: r.err });
   464→   464→    if (!r.ok) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "OPEN_REJECTED", note: (r.err ?? "").slice(0, 160) }); return; }
   465→   465→    orderId = r.orderId;
   466→   466→    fillPrice = await resolveFill(it.tbSymbol, orderId, refPrice);
   467→   467→  } else {
   468→   468→    fillPrice = refPrice; // paper fill on the real toobit book (taker side)
   469→   469→  }
   470→   470→  const tFill = Date.now();
   471→   471→  it.status = "OPEN";
   472→   472→  it.qtyOrig = qty;
   473→   473→  it.qtyRemaining = qty;
   474→   474→  it.orderId = orderId;
   475→   475→  it.fillPrice = fillPrice;
   476→   476→  it.openedAt = tFill;
   477→   477→  // v1.3.0 doctrine bookkeeping — the truth of what the book actually saw:
   478→   478→  // step-size floor rounding makes actualNotional ≤ expectedNotional by design.
   479→   479→  it.actualNotional = fillPrice ? Number((qty * fillPrice).toFixed(6)) : null;
   480→   480→  it.protected = it.mode === "LIVE" ? true : true; // LIVE: attached TP/SL in same order + recon re-check; PAPER: sim active
   481→   481→  S.doctrine.opens++;
   482→   482→  const ndelta = it.expectedNotional && it.actualNotional ? Math.abs(it.actualNotional - it.expectedNotional) / it.expectedNotional * 10_000 : 0;
   483→   483→  audit("EXEC-OPEN-DOCTRINE", { serial: it.serial, lev: it.leverage, levSrc: it.levSrc ?? "card", margin: S.cfg.marginUsdt, expectedNotional: it.expectedNotional, actualNotional: it.actualNotional, notionalDeltaBps: Number(ndelta.toFixed(1)), entryAgeMs: it.entryAgeMs, entryDriftBps: it.entryDriftBps });
   484→   484→  if (ndelta > 2000) doctrineViolation(it.serial, "NOTIONAL-DELTA", `exp=${it.expectedNotional} act=${it.actualNotional} (${ndelta.toFixed(0)}bps) — step-rounding or minQty floor`);
   485→   485→  it.lat = {
   486→   486→    detectMs: it.lat.detectMs ?? null,
   487→   487→    execMs: tFill - tSent,
   488→   488→    totalMs: it.createdAt0 !== undefined ? tFill - it.createdAt0 : null,
   489→   489→    slipBps: fillPrice && it.cardEntry > 0 ? ((it.direction === "LONG" ? fillPrice - it.cardEntry : it.cardEntry - fillPrice) / it.cardEntry) * 10_000 : null,
   490→   490→  };
   491→   491→  it.events.push({ at: tFill, ev: "OPEN", note: `${it.mode} qty=${qty} fill=${fillPrice}` });
   492→   492→  S.metrics.push({ serial: it.serial, symbol: it.symbol, direction: it.direction, mode: it.mode, detectMs: it.lat.detectMs, execMs: it.lat.execMs, totalMs: it.lat.totalMs, slipBps: it.lat.slipBps, fillPrice, cardEntry: it.cardEntry, at: tFill });
   493→   493→  if (S.metrics.length > METRICS_WINDOW) S.metrics = S.metrics.slice(-METRICS_WINDOW);
   494→   494→}

   513→   513→async function doBank(it: Intent, fracOfOrig: number, reason: string): Promise<void> {
   514→   514→  if (it.status !== "OPEN" || it.qtyRemaining <= 0) return;

   516→   516→  if (qty <= 0) return;

   518→   518→  if (it.mode === "LIVE") {

   121→  if (!xs.length) return 0;

   126→/** Stage 1 — feed the tape + screen for velocity/volume candidates. Cheap. */
   127→export function pumpScreen(assets: Array<{ symbol: string; price: number; volume24h: number }>): PumpCandidate[] {
   128→  const now = Date.now();
   129→  const out: PumpCandidate[] = [];
   130→  for (const a of assets) {
   131→    if (!a.price || a.price <= 0) continue;
   132→    let ring = tape.get(a.symbol);
   133→    if (!ring) {
   134→      ring = [];
   135→      tape.set(a.symbol, ring);
   136→    }
   137→    ring.push({ t: now, p: a.price, qv: a.volume24h });
   138→    while (ring.length && now - ring[0]!.t > PUMP_RULES.tapeWindowMs) ring.shift();
   139→
   140→    // 1m quote-volume delta samples (minute-bucketed) for the burst baseline
   141→    let vs = volSamples.get(a.symbol);
   142→    if (!vs) {
   143→      vs = [];
   144→      volSamples.set(a.symbol, vs);
   145→    }
   146→    const lastS = vs[vs.length - 1];
   147→    if (!lastS || now - lastS.t >= 55_000) {
   148→      const ref = ring.find((pt) => now - pt.t <= 65_000) ?? ring[0]!;
   149→      const dv = Math.max(0, a.volume24h - ref.qv);
   150→      vs.push({ t: now, dv });
   151→      while (vs.length > 32) vs.shift();
   152→    }
   153→
   154→    if (ring.length < PUMP_RULES.warmPoints) continue; // honest warmup after restart
   155→
   156→    const at = (ms: number): TapePoint | null => {
   157→      let best: TapePoint | null = null;
   158→      let bestDev = Number.POSITIVE_INFINITY;
   159→      for (const pt of ring!) {
   160→        const dev = Math.abs(pt.t - (now - ms));
   161→        if (dev < bestDev) {
   162→          bestDev = dev;
   163→          best = pt;
   164→        }
   165→      }
   166→      return bestDev <= 15_000 ? best : null;
   167→    };
   168→    const p1 = at(60_000);
   169→    const p5 = at(300_000);
   170→    if (!p1 || !p5) continue;
   171→
   172→    const vel1m = ((a.price - p1.p) / p1.p) * 100;
   173→    const vel5m = ((a.price - p5.p) / p5.p) * 100;
   174→    const dvNow = Math.max(0, a.volume24h - p1.qv);
   175→    const base = median(vs.map((s) => s.dv).slice(0, -1));
   176→    const burstX = base >= 1_000 ? dvNow / base : 0;
   177→    if (dvNow < PUMP_RULES.volFloorUsd1m) continue;
   178→
   179→    const kind: "PUMP" | "DUMP" = vel5m > 0 ? "PUMP" : "DUMP";
   180→    const av1 = Math.abs(vel1m);
   181→    const av5 = Math.abs(vel5m);
   182→    if (av5 < PUMP_RULES.vel5mMinPct || av1 < PUMP_RULES.vel1mMinPct || burstX < PUMP_RULES.volBurstMinX) continue;
   183→    if (openTracks.has(a.symbol)) continue;
   184→    if (now - (lastResolveAt.get(a.symbol) ?? 0) < PUMP_RULES.redeCooldownMs) continue;
   185→    out.push({ symbol: a.symbol, kind, vel1m, vel5m, burstX, price: a.price });
   186→  }
   187→  // strongest first (5m displacement × burst)
   188→  return out.sort((a, b) => Math.abs(b.vel5m) * b.burstX - Math.abs(a.vel5m) * a.burstX).slice(0, 3);
   189→}

   203→export async function gatherEvidence(c: PumpCandidate): Promise<{ score: number; ev: PumpEvidence }> {
   204→  const sign = c.kind === "PUMP" ? 1 : -1;
   205→  let score = 0;
   206→  // velocity core (≤45)
   207→  score += Math.min(27, (Math.abs(c.vel5m) / PUMP_RULES.vel5mMinPct) * 12);
   208→  score += Math.min(18, (Math.abs(c.vel1m) / PUMP_RULES.vel1mMinPct) * 9);
   209→  // volume burst (≤15)
   210→  score += Math.min(15, (c.burstX / PUMP_RULES.volBurstMinX) * 8);
   211→
   212→  const ev: PumpEvidence = {
   213→    oiDelta10m: null, cvdUsd60s: null, takerRatio: null, funding8h: null,
   214→    whaleUsd5m: null, whaleSideAligned: false, liqUsd10m: null, liqSideAligned: false,
   215→  };
   216→  try {
   217→    const [flow, liq] = await Promise.all([fetchFlowMetrics(c.symbol), liquidationPressure(c.symbol)]);
   218→    if (flow) {
   219→      ev.oiDelta10m = flow.oiDelta10m;
   220→      ev.cvdUsd60s = flow.cvdUsd60s;
   221→      ev.takerRatio = flow.takerRatio;
   222→      ev.funding8h = flow.funding8h;
   223→      if (flow.oiDelta10m !== null) {
   224→        // OI rising WITH the move = fresh money (strongest); falling = deleveraging squeeze (real, weaker)
   225→        if (flow.oiDelta10m * sign > 0) score += 12;
   226→        else if (flow.oiDelta10m < 0) score += 6;
   227→      }
   228→      if (flow.cvdUsd60s !== null && flow.cvdUsd60s * sign > 0) {
   229→        score += flow.cvdUsd60s * sign >= 200_000 ? 15 : 12;
   230→      }
   231→      if (flow.takerRatio !== null && flow.takerRatio * sign > 1) score += flow.takerRatio * sign >= 1.15 ? 8 : 5;
   232→      if (flow.whalePrint5m && flow.whalePrint5m.totalUsd >= 250_000) {
   233→        ev.whaleUsd5m = flow.whalePrint5m.totalUsd;
   234→        ev.whaleSideAligned = flow.whalePrint5m.buyUsd >= flow.whalePrint5m.sellUsd ? sign > 0 : sign < 0;
   235→        if (ev.whaleSideAligned) score += 8;
   236→      }
   237→      // funding tailwind: not already paying the crowd to be on our side's neck
   238→      if (flow.funding8h !== null && flow.funding8h * sign <= 0.02) score += 4;
   239→    }
   240→    if (liq) {
   241→      ev.liqUsd10m = liq.usd10m;
   242→      // forced-liquidation fuel: shorts liquidated → pump fuel; longs → dump fuel
   243→      const fuel = sign > 0 ? liq.shortsUsd10m : liq.longsUsd10m;
   244→      ev.liqSideAligned = fuel >= 100_000;
   245→      if (ev.liqSideAligned) score += 6;
   246→    }
   247→  } catch {
   248→    /* evidence unavailable — velocity/volume core still stands, honest score */
   249→  }
   250→  return { score: Math.min(100, Math.round(score)), ev };
   251→}

   310→/** Main entry — called from the engine tick with fresh asset prices. */
   311→export async function pumpTick(assets: Array<{ symbol: string; price: number; volume24h: number }>): Promise<void> {
   312→  await ensureBooted();
   313→  const now = Date.now();
   314→
   315→  // 0) reconcile memory ↔ DB (1/min)
   316→  await syncOpenTracks(now);
   317→
   318→  // 1) track maintenance first — MFE/MAE + phase machine + resolution
   319→  if (now - lastTrackSweepAt >= 5_000 && openTracks.size > 0) {
   320→    lastTrackSweepAt = now;
   321→    const priceMap = new Map(assets.map((a) => [a.symbol, a.price]));
   322→    for (const t of [...openTracks.values()]) {
   323→      const price = priceMap.get(t.symbol);
   324→      if (!price) continue;
   325→      await updateTrack(t, price, now);
   326→    }
   327→  }
   328→
   329→  // 2) stage-1 screen
   330→  if (openTracks.size >= PUMP_RULES.activeCap) return;
   331→  const cands = pumpScreen(assets);
   332→  if (!cands.length) return;
   333→
   334→  // 3) stage-2 confirm + declare
   335→  for (const c of cands) {
   336→    if (openTracks.has(c.symbol)) continue;
   337→    const lc = lastConfirmAt.get(c.symbol) ?? 0;
   338→    if (now - lc < PUMP_RULES.confirmTtlMs) continue;
   339→    lastConfirmAt.set(c.symbol, now);
   340→    const { score, ev } = await gatherEvidence(c);
   341→    if (score < PUMP_RULES.scoreDeclare) continue;
   342→    await declareTrack(c, score, ev, now);
   343→  }
   344→}

   392→async function updateTrack(t: TrackState, price: number, now: number): Promise<void> {
   393→  const sign = t.kind === "PUMP" ? 1 : -1;
   394→  const fav = ((price - t.anchor) / t.anchor) * 100 * sign;
   395→  t.mfe = Math.max(t.mfe, fav);
   396→  t.mae = Math.max(t.mae, -fav);
   397→  if (sign > 0) t.peak = Math.max(t.peak, price);
   398→  else t.peak = Math.min(t.peak, price);
   399→
   400→  // phase machine
   401→  let phase = t.phase;
   402→  if (t.phase === "ONSET" && t.mfe >= PUMP_RULES.extendPct) phase = "EXTENSION";
   403→  else if (t.phase === "EXTENSION") {
   404→    const leg = Math.max(t.mfe, 0.01);
   405→    const retrace = ((sign > 0 ? t.peak - price : price - t.peak) / (t.peak || price)) * 100;
   406→    if (retrace >= PUMP_RULES.exhaustRetraceFrac * leg) phase = "EXHAUSTION";
   407→  }
   408→
   409→  // resolution
   410→  let outcome: "SUCCESS" | "FAILURE" | null = null;
   411→  let note = "";
   412→  const ageMin = (now - t.detectedAt) / 60_000;
   413→  if (t.mfe >= PUMP_RULES.successMovePct) {
   414→    outcome = "SUCCESS";
   415→    note = `MFE +${t.mfe.toFixed(2)}% ≥ +${PUMP_RULES.successMovePct}% in ${ageMin.toFixed(0)}min — continuation validated`;
   416→  } else if (t.mae >= PUMP_RULES.collapseMaePct) {
   417→    outcome = "FAILURE";
   418→    note = `collapsed — MAE ${t.mae.toFixed(2)}% against the thesis (MFE was +${t.mfe.toFixed(2)}%)`;
   419→  } else if (ageMin >= PUMP_RULES.horizonMin) {
   420→    outcome = "FAILURE";
   421→    note = `horizon ${PUMP_RULES.horizonMin}min elapsed — MFE +${t.mfe.toFixed(2)}% never reached +${PUMP_RULES.successMovePct}%`;
   422→  }
   423→
   424→  const dirty = phase !== t.phase || now - t.lastWriteAt >= PUMP_RULES.trackUpdateMs || outcome;
   425→  if (!dirty) return;
   426→  t.phase = outcome ? "RESOLVED" : phase;
   427→  t.lastWriteAt = now;
   428→  try {
   429→    await db.pumpTrack.update({
   430→      where: { id: t.id },
   431→      data: {
   432→        phase: t.phase,
   433→        mfePct: Number(t.mfe.toFixed(3)),
   434→        maePct: Number(t.mae.toFixed(3)),
   435→        peakPrice: t.peak,
   436→        ...(outcome
   437→          ? { outcome, resolvedAt: new Date(), successNote: note }
   438→          : {}),
   439→      },
   440→    });
   441→  } catch (e) {
   442→    console.error("[pump-radar] track update failed:", e);
   443→    return;
   444→  }
   445→  if (outcome) {
   446→    openTracks.delete(t.symbol);
   447→    lastResolveAt.set(t.symbol, now);
   448→    const engine = await import("./engine");
   449→    await engine.fireEvent(
   450→      null,
   451→      t.symbol,
   452→      outcome === "SUCCESS" ? "PUMP_RADAR" : "ENGINE",
   453→      outcome === "SUCCESS" ? "success" : "info",
   454→      `${t.kind} TRACK ${outcome} — ${t.symbol} ${outcome === "SUCCESS" ? "✓" : "✕"} · ${note} (MAE ${t.mae.toFixed(2)}%) · gate ${await gateLine()}`,
   455→    );
   456→  }
   457→}

   462→export async function pumpGateAllows(): Promise<{ unlocked: boolean; successful: number; failed: number; total: number; required: number; reason: string }> {
   463→  if (!gateMemo || Date.now() - gateMemo.at >= 30_000) {
   464→    const groups = await db.pumpTrack.groupBy({ by: ["outcome"], _count: { _all: true } });
   465→    const get = (k: string) => groups.find((g) => g.outcome === k)?._count._all ?? 0;
   466→    gateMemo = { at: Date.now(), successful: get("SUCCESS"), failed: get("FAILURE"), total: get("SUCCESS") + get("FAILURE") };
   467→    if (gateMemo.successful >= PUMP_RULES.requiredSuccesses) {
   468→      const cur = await db.engineSetting.findUnique({ where: { key: "pump.gate.unlockedAt" } }).catch(() => null);
   469→      if (!cur) {
   470→        await db.engineSetting.upsert({
   471→          where: { key: "pump.gate.unlockedAt" },
   472→          create: { key: "pump.gate.unlockedAt", value: new Date().toISOString() },
   473→          update: { value: new Date().toISOString() },
   474→        }).catch(() => undefined);
   475→        const engine = await import("./engine");
   476→        await engine
   477→          .fireEvent(null, null, "PUMP_RADAR", "success", `PUMP/DUMP RADAR UNLOCKED — ${gateMemo.successful} successful tracks confirmed the detector; dedicated PUMP/DUMP signal cards now ARMED (all normal engine gates still apply to every deploy)`)
   478→          .catch(() => undefined);
   479→      }
   480→    }
   481→  }
   482→  const unlocked = gateMemo.successful >= PUMP_RULES.requiredSuccesses;
   483→  return {
   484→    unlocked,
   485→    successful: gateMemo.successful,
   486→    failed: gateMemo.failed,
   487→    total: gateMemo.total,
   488→    required: PUMP_RULES.requiredSuccesses,
   489→    reason: unlocked
   490→      ? `radar confirmed — ${gateMemo.successful} successful tracks (≥${PUMP_RULES.requiredSuccesses})`
   491→      : `signals forbidden until ${PUMP_RULES.requiredSuccesses} successful tracks — ${gateMemo.successful} so far (${gateMemo.total} resolved)`,
   492→  };
   493→}

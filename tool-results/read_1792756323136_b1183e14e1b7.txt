   247→function findApp(): { pid: number; parents: number[] } | null {
   248→  const out = sh(`ss -tlnp 2>/dev/null | grep ':3000 ' | grep -o 'pid=[0-9]*' | head -1 | cut -d= -f2`, 10);
   249→  const pid = parseInt(out, 10);
   250→  if (!pid || !pidAlive(pid)) return null;
   251→  const parents: number[] = [];
   252→  let cur = pid;
   253→  for (let i = 0; i < 4; i++) {
   254→    try {
   255→      const stat = readFileSync(`/proc/${cur}/stat`, "utf8").split(" ");
   256→      const ppid = parseInt(stat[3], 10);
   257→      if (!ppid || ppid <= 1) break;
   258→      parents.push(ppid); cur = ppid;
   259→    } catch { break; }
   260→  }
   261→  return { pid, parents };
   262→}

   425→    await releaseWatchdog();
   426→  } finally {
   427→    restarting = false;
   428→    persistState();

   488→/** v2.2 — the shared remote-recovery choreography (used by BOTH the date-based
   489→ *  staleness path and the new engine-version sentinel + DB-stale auto-heal). */
   490→async function recoverFromRemote(ledgerHead: string, why: string): Promise<"recovered" | "fetch-failed" | "reset-failed" | "remote-agrees"> {
   491→  const h0 = git(["rev-parse", "HEAD"]);
   492→  const sha = h0.ok ? h0.out.split("\n")[0].trim() : "";
   493→  const remote = git(["ls-remote", REPO, "main"], 45_000);
   494→  const remoteSha = remote.ok ? remote.out.split("\t")[0].trim() : "";
   495→  if (!remoteSha || remoteSha === sha) return "remote-agrees";
   496→  const fetched = git(["fetch", "--depth=6", "origin", remoteSha], 240_000);
   497→  if (!fetched.ok) { await notify(`⚠️ ROLLBACK-SENTINEL (${why}): stale تشخیص داده شد ولی fetch remote ناموفق بود — دستی بررسی شود.`, "crit"); return "fetch-failed"; }
   498→  const reset = git(["reset", "--hard", remoteSha], 60_000);
   499→  if (!reset.ok) { await notify(`❌ ROLLBACK-SENTINEL (${why}): reset شکست خورد: ${reset.out.slice(0, 100)}`, "crit"); return "reset-failed"; }
   500→  const fresh = readFresh();
   501→  if (fresh) {
   502→    fresh.recoveries.push({ at: new Date().toISOString(), from: sha.slice(0, 8), to: remoteSha.slice(0, 8) });
   503→    if (fresh.recoveries.length > 20) fresh.recoveries = fresh.recoveries.slice(-20);
   504→    writeFresh(fresh);
   505→  }
   506→  const h1 = git(["log", "-1", "--format=%cI|%s", remoteSha]);
   507→  const headDate = h1.ok ? (h1.out.split("|")[0] ?? "") : "";
   508→  await notify(`🧬 ROLLBACK-SENTINEL (${why}): درخت به عقب برگشته بود (${why === "engine-version" ? `موتور دیسک ${diskEngineVer() || "?"} ≠ لجر ${fresh?.lastKnownEngineVer ?? "?"}` : `HEAD ${sha.slice(0, 8)} قدیمی‌تر از لجر ${ledgerHead.slice(0, 8)}`}) — به remote ${remoteSha.slice(0, 8)} (${headDate.slice(0, 16)}) بازنشانی شد. regen کلاینت + ری‌استارت اپ در جریان.`, "warn");
   509→  // tree reset can change prisma/schema.prisma → regenerate the client BEFORE
   510→  // the app keeps running on it (Task 37 incident class), then restart the app
   511→  // to load both the new code and the new client. The running app always holds
   512→  // the OLD code in memory after a rollback → restart is unconditional.
   513→  await schemaClientGate("post-rollback-recovery");
   514→  if (app.pid > 0) await memRestart("rollback-sentinel: tree recovered from remote");
   515→  else pendingSchemaRestart = true;
   516→  log("info", "rollback-recovered", { to: remoteSha.slice(0, 8), why });
   517→  return "recovered";
   518→}

   520→async function freshnessGate(): Promise<void> {
   521→  lastFreshCheckAt = Date.now();
   522→  const h0 = git(["rev-parse", "HEAD"]);
   523→  if (!h0.ok) { log("warn", "fresh-head-unreadable", {}); return; }
   524→  let sha = h0.out.split("\n")[0].trim();
   525→  const info = git(["log", "-1", "--format=%cI|%s", sha]);
   526→  let headDate = info.out.split("|")[0] ?? "";
   527→  let headSubject = info.out.split("|")[1] ?? "";
   528→  const fresh = readFresh();
   529→  const diskVer = diskEngineVer();
   530→  const dbStat = dbSignalStat();
   531→  let action = "none";
   532→
   533→  // ── v2.2 SENTINEL ①: content truth — engine version divergence = rollback,
   534→  // even when armor's fresh-timestamped commits make HEAD look current.
   535→  if (fresh && fresh.lastKnownEngineVer && diskVer && fresh.lastKnownEngineVer !== diskVer) {
   536→    log("crit", "engine-version-rollback-detected", { disk: diskVer, ledger: fresh.lastKnownEngineVer });
   537→    const r = await recoverFromRemote(fresh.lastKnownHead, "engine-version");
   538→    if (r === "recovered") { action = "recovered-from-remote(engine-version)"; }
   539→    else if (r === "remote-agrees") { await notify("⚠️ نسخهٔ موتور دیسک با لجر فرق دارد ولی remote هم همین نسخه را دارد — لجر به‌روزرسانی می‌شود (احتمالاً deploy عمدی).", "warn"); }
   540→    // re-read post-recovery state
   541→    const h1 = git(["rev-parse", "HEAD"]);
   542→    if (h1.ok) sha = h1.out.split("\n")[0].trim();
   543→    const i1 = git(["log", "-1", "--format=%cI|%s", sha]);
   544→    headDate = i1.ok ? (i1.out.split("|")[0] ?? headDate) : headDate;
   545→    headSubject = i1.ok ? (i1.out.split("|")[1] ?? headSubject) : headSubject;
   546→  } else if (fresh && fresh.lastKnownHead && fresh.lastKnownHead !== sha) {
   547→    // ── classic date-based staleness (kept from v2.1) ──
   548→    const ledgerMs = Date.parse(fresh.lastKnownHeadAt);
   549→    const headMs = Date.parse(headDate);
   550→    if (!isNaN(ledgerMs) && !isNaN(headMs) && headMs < ledgerMs - 60_000) {
   551→      log("warn", "stale-filesystem-detected", { head: sha.slice(0, 8), headDate, ledgerHead: fresh.lastKnownHead.slice(0, 8), ledgerDate: fresh.lastKnownHeadAt });
   552→      const r = await recoverFromRemote(fresh.lastKnownHead, "head-date");
   553→      if (r === "recovered") action = "recovered-from-remote";
   554→      else if (r === "remote-agrees") action = "remote-agrees-stale";
   555→      const h2 = git(["rev-parse", "HEAD"]);
   556→      if (h2.ok) sha = h2.out.split("\n")[0].trim();
   557→    }
   558→  }
   559→
   560→  // ── v2.2 SENTINEL ②: DB-stale AUTO-HEAL (was alert-only in v2.1 — operator
   561→  // mandate 2026-10-16). Signal rows are never pruned by design, so a >20%
   562→  // drop is always a rollback artifact, never housekeeping. The same tree
   563→  // reset restores db/custom.db (+wal) from the remote; SQLite discards a
   564→  // mismatched stale WAL on next open, so the restored pair is coherent.
   565→  const prevCount = fresh?.lastDbSignalCount ?? -1;
   566→  if (prevCount > 0 && dbStat.count >= 0 && dbStat.count < prevCount * 0.8 && action === "none") {
   567→    log("crit", "db-rollback-detected", { live: dbStat.count, ledger: prevCount });
   568→    const r = await recoverFromRemote(fresh!.lastKnownHead, "db-count");
   569→    action = r === "recovered" ? "recovered-from-remote(db-count)" : `db-stale-${r}`;
   570→    const after = dbSignalStat();
   571→    if (r !== "recovered") {
   572→      await notify(`🗄 DB-STALE: سیگنال‌های زنده ${dbStat.count} < ۸۰٪ لجر (${prevCount}) و بازیابی خودکار نشد (${r}) — آخرین zip کامل: /tmp/my-project/backups/ — دستی بررسی شود.`, "crit");
   573→    } else {
   574→      await notify(`🗄 بازیابی DB کامل شد: ${dbStat.count} → ${after.count} ردیف.`, "warn");
   575→    }
   576→  }
   577→
   578→  const finalFresh: Freshness = {
   579→    lastKnownHead: sha,
   580→    lastKnownHeadAt: isNaN(Date.parse(headDate)) ? new Date().toISOString() : new Date(Date.parse(headDate)).toISOString(),
   581→    lastKnownHeadSubject: headSubject,
   582→    lastKnownEngineVer: diskVer || fresh?.lastKnownEngineVer || "",
   583→    lastDbSignalCount: Math.max(dbStat.count, 0),
   584→    lastDbMaxCreatedAt: dbStat.maxCreated,
   585→    lastSeenAt: new Date().toISOString(),
   586→    recoveries: fresh?.recoveries ?? [],
   587→  };
   588→  writeFresh(finalFresh);
   589→  log("info", "freshness-ok", { head: sha.slice(0, 8), headDate, engineVer: diskVer, dbSignals: dbStat.count, action });
   590→}

   592→// ── LAYER 3: schema-client gate (v2.1 — stale-Prisma-client incident class) ──
   593→function signalModelFields(p: string): Set<string> {
   594→  try {
   595→    const src = readFileSync(p, "utf8");
   596→    const m = src.match(/model Signal \{([\s\S]*?)\n\}/);
   597→    if (!m) return new Set();
   598→    const out = new Set<string>();
   599→    for (const line of m[1].split("\n")) {
   600→      const t = line.trim();
   601→      if (!t || t.startsWith("//") || t.startsWith("@@") || t.startsWith("@")) continue;
   602→      const name = t.split(/\s+/)[0];
   603→      if (name && /^[a-zA-Z_]\w*$/.test(name)) out.add(name);
   604→    }
   605→    return out;
   606→  } catch { return new Set(); }
   607→}

   609→/** Diff prisma/schema.prisma vs the generated client's own schema copy.
   610→ *  Returns true when a heal happened (client was stale, regenerate succeeded)
   611→ *  and the running app must be restarted to load the fresh client. */
   612→async function schemaClientGate(reason: string): Promise<boolean> {
   613→  schema.lastCheckAt = Date.now();
   614→  const src = signalModelFields(`${ROOT}/prisma/schema.prisma`);
   615→  const cli = signalModelFields(`${ROOT}/node_modules/.prisma/client/schema.prisma`);
   616→  if (src.size === 0) { schema.lastResult = "source-unreadable"; log("warn", "schema-source-unreadable", {}); return false; }
   617→  if (cli.size === 0) { schema.lastResult = "client-unreadable"; }
   618→  const missing = [...src].filter(f => !cli.has(f));
   619→  const extra = [...cli].filter(f => !src.has(f));
   620→  if (missing.length === 0 && extra.length === 0) {
   621→    schema.lastResult = "match"; schema.missing = [];
   622→    return false;
   623→  }
   624→  schema.missing = missing;
   625→  schema.lastResult = `mismatch (missing-in-client: ${missing.slice(0, 6).join(",") || "none"} | stale-extra: ${extra.slice(0, 6).join(",") || "none"})`;
   626→  log("crit", "schema-client-mismatch", { reason, missing: missing.slice(0, 10), extra: extra.slice(0, 10), note: "app update() with these fields throws PrismaClientValidationError → lifecycle freeze + per-tick event spam (2026-10-15 incident)" });
   627→  if (Date.now() - schema.lastHealAt < 30 * 60_000) {
   628→    log("warn", "schema-heal-throttled", { lastHealMinAgo: Math.round((Date.now() - schema.lastHealAt) / 60_000) });
   629→    return false;
   630→  }
   631→  schema.lastHealAt = Date.now();
   632→  sh(`cd ${ROOT} && bunx prisma generate >/dev/null 2>&1`, 240);
   633→  const cli2 = signalModelFields(`${ROOT}/node_modules/.prisma/client/schema.prisma`);
   634→  const stillMissing = [...src].filter(f => !cli2.has(f));
   635→  if (stillMissing.length === 0 && cli2.size > 0) {
   636→    schema.lastResult = "healed"; schema.missing = [];
   637→    log("info", "schema-client-healed", { reason, regenerated: missing.length });
   638→    if (app.pid === 0) pendingSchemaRestart = true; // boot probe: app not up yet → restart on first tick sighting
   639→    await notify(`🧬 SCHEMA-CLIENT GUARD: کلاینت Prisma stale بود (ستون‌های ${missing.slice(0, 3).join(",")}… نبود) — بازتولید شد (${reason}). اپ برای بارگذاری کلاینت جدید ری‌استارت می‌شود.`, "warn");
   640→    return true;
   641→  }
   642→  schema.lastResult = `heal-failed (${stillMissing.slice(0, 4).join(",")})`;
   643→  await notify(`❌ SCHEMA-CLIENT: بازتولید کلاینت مشکل را حل نکرد (${stillMissing.slice(0, 3).join(",")}…) — بررسی دستی لازم است.`, "crit");
   644→  return false;
   645→}

   690→// ── main tick (self-reaping chain — one failed tick can never kill the loop) ─
   691→async function tick(): Promise<void> {
   692→  ticks++;
   693→  mem = { availableMb: readMemInfo(), level: "ok", at: Date.now() };
   694→  mem.level = mem.availableMb < 350 ? "crit" : mem.availableMb < 600 ? "warn" : "ok";
   695→  await maybeReleaseQuarantine(); // v2.3 — auto-release when the preemptive window ages out
   696→
   697→  const found = findApp();
   698→  if (found) {
   699→    if (found.pid !== app.pid) {
   700→      app = { pid: found.pid, rssMb: readRssMb(found.pid), firstSeenAt: Date.now(), parentPids: found.parents };
   701→      highRssTicks = 0; lowAvailTicks = 0;
   702→      log("info", "app-pid-seen", { pid: found.pid, parents: found.parents.slice(0, 2) });
   703→    } else {
   704→      app.rssMb = readRssMb(found.pid);
   705→      app.parentPids = found.parents;
   706→    }
   707→    samples.push({ t: Date.now(), rss: app.rssMb, avail: mem.availableMb });
   708→    if (samples.length > SAMPLE_CAP) samples = samples.slice(-SAMPLE_CAP);
   709→
   710→    const bootGrace = Date.now() - app.firstSeenAt < BOOT_GRACE_MS;

   747→

   754→  }

   756→  logHygiene();

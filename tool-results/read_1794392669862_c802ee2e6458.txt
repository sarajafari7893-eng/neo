   423→// ── weight table (Binance USDT-M futures documented weights) ────────────────
   424→// klines ~2 (limit≤100) · depth 2-5-10 by limit · aggTrades 20 · trades 20 ·
   425→// ticker/24hr 1 per-symbol / 40 full-universe · premiumIndex 1 per-symbol /
   426→// 10 all-markers · futures/data/* 1 (openInterestHist, L/S ratios, basis…) ·
   427→// exchangeInfo 1 · ping/time 1 · unknown default 5 (conservative).
   428→function resolveWeight(path: string): number {
   429→  const qIdx = path.indexOf("?");
   430→  const ep = (qIdx === -1 ? path : path.slice(0, qIdx)).replace(/^\//, "").replace(/\/+$/, "");
   431→  const query = qIdx === -1 ? "" : path.slice(qIdx + 1);
   432→  const limit = (): number | null => {
   433→    const m = /[?&]limit=(\d+)/.exec(query);
   434→    return m ? Number(m[1]) : null;
   435→  };
   436→  switch (ep) {
   437→    case "fapi/v1/ping":
   438→    case "fapi/v1/time":
   439→    case "fapi/v1/exchangeInfo":
   440→      return 1;
   441→    case "fapi/v1/klines": {
   442→      const l = limit() ?? 500;
   443→      return l <= 100 ? 2 : l <= 499 ? 5 : 10;
   444→    }
   445→    case "fapi/v1/depth": {
   446→      const l = limit() ?? 100;
   447→      return l <= 100 ? 2 : l <= 500 ? 5 : 10;
   448→    }
   449→    case "fapi/v1/aggTrades":
   450→    case "fapi/v1/trades":
   451→      return 20;
   452→    case "fapi/v1/ticker/24hr":
   453→      return /[?&]symbol=/.test(query) ? 1 : 40;
   454→    case "fapi/v1/premiumIndex":
   455→      return /[?&]symbol=/.test(query) ? 1 : 10;
   456→    default:
   457→      if (ep.startsWith("futures/data/")) return 1; // openInterestHist / L-S ratios / basis family
   458→      return 5; // unknown → conservative default (covers fapi/v1/fundingRate etc.)
   459→  }
   460→}

   462→function pruneWindow(now: number): void {
   463→  const cutoff = now - WINDOW_MS;
   464→  while (state.spend.length > 0 && state.spend[0]!.at <= cutoff) {
   465→    const e = state.spend.shift()!;
   466→    state.windowSum -= e.w;
   467→  }
   468→  if (state.spend.length > SPEND_CAP) {
   469→    // defensive: drop the oldest half rather than grow unbounded
   470→    const keep = state.spend.slice(Math.floor(state.spend.length / 2));
   471→    let sum = 0;
   472→    for (const e of keep) sum += e.w;
   473→    state.spend = keep;
   474→    state.windowSum = sum;
   475→  }
   476→}

   478→/** Reserve `weight` from the rolling window. Returns the spend entry (for refunds) or null. */
   479→// ── v14.2 BAN-AWARE RECOVERY PACER ────────────────────────────────────────
   480→// The ONLY budget the rolling window spends against. A flat budget after a
   481→// ban ended = instant re-saturation stampede (the 10-19 spiral); the pacer
   482→// gives the pool a 10min ramp and caps telemetry while the pool shows live
   483→// distress. Global-window minutes are NOT handled here (gateFetch drops
   484→// everything while the window is live) — the pacer shapes the BEFORE
   485→// (distress) and AFTER (recovery) of each ban episode.
   486→function namedLiveBanCount(now: number): number {
   487→  let n = 0;
   488→  for (const [, until] of state.ipBans) if (until > now) n++;
   489→  return n;
   490→}

   492→// ── v14.3 ROLLING-CYCLE AIMD PACER (chapter-7 item ② «backoff واقعی», operator
   493→// mandate trace 1a1a8ea4186c6571) — the missing third leg of the v14.2 pacer.
   494→// Measured failure mode (engineering report NX3247+LIVE): the 3-IP egress pool
   495→// rides a ROLLING ~20min ban cycle — bans expire, the full budget floods back
   496→// instantly (named-IP bans never armed the global window ⇒ no ramp), the pool
   497→// re-saturates in the same minute and re-bans. Fix: an AIMD shed factor that
   498→// DECREASES multiplicatively on every named-IP ban event (×0.7, floor 0.25) and
   499→// recovers ADDITIVELY during clean minutes (+0.05/min, cap 1.0). Sustained ban
   500→// churn ⇒ progressively quieter pool without any operator tuning; a genuinely
   501→// healthy pool heals back to full budget in ~15 clean minutes.
   502→const AIMD_DROP = 0.7;
   503→const AIMD_FLOOR = 0.25;
   504→const AIMD_RECOVER_PER_MIN = 0.05;
   505→
   506→function recoverAimd(now: number): void {
   507→  if (state.aimdFactor >= 1) { state.lastAimdRecoverAt = now; return; }
   508→  if (state.lastBanEventAt === 0) return;
   509→  const anchor = Math.max(state.lastAimdRecoverAt, state.lastBanEventAt);
   510→  const cleanMin = Math.floor((now - anchor) / 60_000);
   511→  if (cleanMin <= 0) return;
   512→  state.aimdFactor = Math.min(1, state.aimdFactor + cleanMin * AIMD_RECOVER_PER_MIN);
   513→  state.lastAimdRecoverAt = now;
   514→}
   515→
   516→function noteAimdBanEvent(now: number): void {
   517→  recoverAimd(now);
   518→  state.aimdFactor = Math.max(AIMD_FLOOR, state.aimdFactor * AIMD_DROP);
   519→  state.lastBanEventAt = now;
   520→  state.lastAimdRecoverAt = now;
   521→}

   523→export function effectiveBudgetPer60s(): number {
   524→  const now = Date.now();
   525→  let factor = 1;
   526→  recoverAimd(now); // v14.3 — additive clean-minute recovery before clamping
   527→  factor = Math.min(factor, state.aimdFactor);
   528→  // POOL DISTRESS — ≥2 named egress IPs inside live ban windows: cap telemetry.
   529→  if (namedLiveBanCount(now) >= 2) factor = Math.min(factor, PACER_DISTRESS_CAP);
   530→  // RECOVERY RAMP — 10min linear 25%→100% after the last global window ends.
   531→  if (state.banUntil > 0) {
   532→    const sinceBan = now - state.banUntil;
   533→    if (sinceBan >= 0 && sinceBan < PACER_RECOVERY_MS) {
   534→      factor = Math.min(factor, PACER_RAMP_FLOOR + (1 - PACER_RAMP_FLOOR) * (sinceBan / PACER_RECOVERY_MS));
   535→    }
   536→  }
   537→  return Math.max(PACER_MIN_BUDGET, Math.round(state.budgetPer60s * factor));
   538→}

   540→function takeWeight(weight: number): SpendEntry | null {
   541→  const now = Date.now();
   542→  pruneWindow(now);
   543→  if (state.windowSum + weight > effectiveBudgetPer60s()) return null;
   544→  const entry: SpendEntry = { at: now, w: weight };
   545→  state.spend.push(entry);
   546→  state.windowSum += weight;
   547→  return entry;
   548→}

   550→/** 418/429 responses carry no weight cost — give the reservation back. */
   551→function refundWeight(entry: SpendEntry | null): void {
   552→  if (!entry) return;
   553→  const idx = state.spend.indexOf(entry);
   554→  if (idx === -1) return; // already pruned out of the window (its cost is gone too)
   555→  state.spend.splice(idx, 1);
   556→  state.windowSum -= entry.w;
   557→}

   566→/** Record a 418/429 body. Returns { ip, until } when the body NAMES the banned egress IP. */
   567→function noteBanBody(status: number, body: string): { ip: string; until: number } | null {
   568→  const now = Date.now();
   569→  const named = /IP\(([\d.]+)\)\s*banned until (\d+)/i.exec(body);
   570→  if (named) {
   571→    const ip = named[1]!;
   572→    // v12.1 TASK-49: the LEDGER carries Binance's REAL end (sanity-capped 24h) —
   573→    // the cosmetic +35min cap only applies to the global silence window below.
   574→    // Capping the ledger made the early probe fire at a fake expiry and the
   575→    // resulting 418 contact re-escalated the real ban (11:58→12:34→13:09 spiral).
   576→    const realUntil = Math.min(Number(named[2]), now + 24 * 60_000);
   577→    const until = Math.min(realUntil, now + BAN_CAP_MS);
   578→    state.ipBans.set(ip, realUntil);
   579→    state.lastBannedIp = ip;
   580→    persistBans(); // v12.0 — named-IP ban ledger survives restarts (30s debounce)
   581→    if (state.ipBans.size > IP_BAN_MAP_CAP) {
   582→      // evict the soonest-expiring record
   583→      let oldestKey: string | null = null;
   584→      let oldestVal = Infinity;
   585→      for (const [k, v] of state.ipBans) if (v < oldestVal) { oldestVal = v; oldestKey = k; }
   586→      if (oldestKey) state.ipBans.delete(oldestKey);
   587→    }
   588→    state.recentBanEvents.push({ at: now, ip });
   589→    if (state.recentBanEvents.length > 64) state.recentBanEvents.splice(0, state.recentBanEvents.length - 64);
   590→    noteAimdBanEvent(now); // v14.3 — every named-IP ban deepens the rolling-cycle shed
   591→    return { ip, until };
   592→  }
   593→  const untilMatch = /banned until (\d+)/i.exec(body);
   594→  if (untilMatch) {
   595→    armGlobalBan(Math.min(Number(untilMatch[1]), now + BAN_CAP_MS)); // unknown scope — assume it's us
   596→    return null;
   597→  }
   598→  // unparseable body — conservative defaults (taxonomy precedent for 418)
   599→  armGlobalBan(now + (status === 429 ? BAN_429_DEFAULT_MS : BAN_418_DEFAULT_MS));
   600→  return null;
   601→}

   627→/**
   628→ * THE gate body. Resolves parsed JSON, or null on: live IP ban, weight-budget
   629→ * shed, HTTP failure, timeout, or parse error. Never throws, never queues.
   630→ */
   631→async function executeGated<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   632→  const lane: FapiGatePriority = opts?.priority ?? "telemetry";
   633→  const weight = Math.max(1, Math.round(opts?.weight ?? resolveWeight(path)));
   634→  const timeoutMs = opts?.timeoutMs ?? DEFAULT_TIMEOUT_MS;
   635→
   636→  let entry = takeWeight(weight);
   637→  if (!entry) {
   638→    if (lane !== "critical") {
   639→      state.stats.droppedBudget++;
   640→      return null; // telemetry: drop instead of queueing a storm
   641→    }
   642→    await delay(CRITICAL_RETRY_MS); // one small negative retry
   643→    entry = takeWeight(weight);
   644→    if (!entry) {
   645→      state.stats.droppedBudget++;
   646→      return null;
   647→    }
   648→  }
   649→
   650→  const host = opts?.host ?? FAPI_HOST;
   651→  let attempt = 0;
   652→
   653→  while (true) {
   654→    const ctrl = new AbortController();
   655→    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   656→    let res: Response;
   657→    try {
   658→      // v12.6.1 CONNECTION-RECYCLE (measured 2026-10-20 19:41Z, bun runtime):
   659→      // a keep-alive fetch POOLS one TCP flow ⇒ ONE egress IP for its whole
   660→      // life — a pinned-banned connection turned 231 gate calls into 0 allowed
   661→      // while 6/10 fresh connections succeeded. `Connection: close` = a fresh
   662→      // NAT lottery draw per request; the TLS cost (~60ms) is trivial next to
   663→      // the starvation it prevents.
   664→      res = await fetch(`${host}${path}`, { signal: ctrl.signal, cache: "no-store", headers: { Connection: "close" } });
   665→    } catch {
   666→      return null; // timeout/abort/network — weight may have been consumed, keep the reservation
   667→    } finally {
   668→      clearTimeout(timer);
   669→    }
   670→
   671→    if (res.ok) {
   672→      const text = await res.text().catch(() => "");
   673→      if (!text) return null; // empty 200 — nothing to invent
   674→      state.stats.allowed++;
   675→      state.allBanStreak = 0; // a clean 2xx proves the pool still serves us
   676→      try {
   677→        return JSON.parse(text) as T;
   678→      } catch {
   679→        return null;
   680→      }
   681→    }
   682→
   683→    if (res.status === 418 || res.status === 429) {
   684→      const body = await res.text().catch(() => "");
   685→      refundWeight(entry); // banned responses carry no weight cost
   686→      entry = null;
   687→      const named = noteBanBody(res.status, body);
   688→      if (opts?.singleProbe) return null; // v12.0 early probe — single attempt, window stays armed
   689→      if (!named) return null; // unknown-scope ban — global zero-network window already armed
   690→      if (attempt >= BAN_RETRIES) {
   691→        escalatePoolBlock(named.until);
   692→        return null;
   693→      }
   694→      attempt++;
   695→      state.stats.retriesAfterBan++;
   696→      await delay(BAN_RETRY_DELAYS_MS[Math.min(attempt - 1, BAN_RETRY_DELAYS_MS.length - 1)]!); // span the NAT rotation window
   697→      entry = takeWeight(weight);
   698→      if (!entry) {
   699→        state.stats.droppedBudget++;
   700→        return null;
   701→      }
   702→      continue;
   703→    }
   704→
   705→    return null; // other non-ok status — honest null, no retry
   706→  }
   707→}

   752→export async function gateFetch<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   753→  state.stats.attempted++;
   754→  const now = Date.now();
   755→
   756→  // v12.1 TASK-49 — LAZY LEDGER GUARD: the instant our global window expires,
   757→  // re-check the shared ledger (throttled to one disk read / 15s) BEFORE any
   758→  // network call — closes the poke-gap where the window died but the refresh
   759→  // interval hadn't fired yet. While the pool is ledger-banned, zero pokes.
   760→  if (now >= state.banUntil && now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   761→  if (now < state.banUntil) state.stats.droppedBan++; // ledger re-armed the window — shed instantly
   762→
   763→  // GLOBAL BAN AWARENESS — while a global ban window is live, zero network calls
   764→  // …except the v12.0 ROTATION-AWARE EARLY PROBE: one critical-lane attempt per
   765→  // PROBE_COOLDOWN_MS when the sampled egress IP is clean — the pool rotates,
   766→  // so the window can be LIFTED the moment a clean draw appears instead of
   767→  // waiting out the full 35min cap. A failed probe keeps the window armed.
   768→  if (now < state.banUntil) {
   769→    state.stats.droppedBan++;
   770→    const canProbe = (opts?.priority ?? "telemetry") === "critical" && now - state.lastProbeAt >= PROBE_COOLDOWN_MS;
   771→    if (!canProbe) return null;
   772→    // v12.1 TASK-49 — LEDGER FRESHNESS BEFORE PROBE: state may hold a cosmetic
   773→    // cap while the disk carries a longer REAL end another process discovered.
   774→    // A probe fired on stale state pokes a still-banned pool and re-escalates.
   775→    // Re-read the ledger (15s throttle) and re-arm BEFORE the probe decision.
   776→    if (now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   777→    if (now < state.banUntil) return null; // ledger extended the silence — no probe
   778→    state.lastProbeAt = now;
   779→    const ip = await sampleEgressIp();
   780→    // v12.1 TASK-49: expiry-aware check — a recorded ban whose real end passed
   781→    // no longer blocks the probe (the old .has() kept ghost entries armed).
   782→    if (!ip || isIpBanned(ip, now)) return null; // sampled draw is banned/unknown — stay silent
   783→    state.stats.probesIssued++;
   784→    const probeResult = await executeGated<T>(path, { ...opts, singleProbe: true });
   785→    if (probeResult !== null) {
   786→      state.banUntil = 0; // clean draw proved the pool rotated — lift instantly
   787→      state.allBanStreak = 0;
   788→      state.stats.probeLifts++;
   789→      console.log(`[fapi-gate] v12.0 ban window LIFTED early — egress rotated to clean IP ${ip}`);
   790→    }
   791→    return probeResult;
   792→  }
   793→
   794→  // SINGLE-FLIGHT per (host,path): share the request while it is in flight and
   795→  // for a 2s result window — concurrent module askers pay the weight once.
   796→  const key = `${opts?.host ?? FAPI_HOST}${path}`;
   797→  const existing = state.inflight.get(key);
   798→  if (existing) {
   799→    const age = now - existing.startedAt;
   800→    if (!existing.settled || age < SF_RESULT_MS) {
   801→      state.stats.singleFlightShared++;
   802→      return existing.promise as Promise<T | null>;
   803→    }
   804→    state.inflight.delete(key);
   805→  }
   806→  if (state.inflight.size >= SF_MAP_CAP) pruneInflight(now);
   807→
   808→  const promise = executeGated<T>(path, opts);
   809→  const entry = { startedAt: now, settled: false, promise };
   810→  state.inflight.set(key, entry);
   811→  void promise
   812→    .catch(() => null) // executeGated never rejects — belt & suspenders
   813→    .finally(() => {
   814→      entry.settled = true;
   815→    });
   816→  return promise;
   817→}
   818→

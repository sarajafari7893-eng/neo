    99→async function cachedConsensusVal<T>(key: string, ttlMs: number, fn: () => Promise<T>): Promise<T | null> {
   100→  const hit = consensusCache.get(key);
   101→  if (hit && Date.now() - hit.at < ttlMs) return (hit.val as T | null) ?? null;
   102→  const inflight = consensusInFlight.get(key);
   103→  if (inflight) return (await inflight as T | null) ?? null;
   104→  const p = (async () => {
   105→    let val: T | null = null;
   106→    try {
   107→      val = await fn();
   108→    } catch {
   109→      val = null;
   110→    }
   111→    consensusCache.set(key, { at: Date.now(), val });
   112→    if (consensusCache.size > 800) {
   113→      const oldest = consensusCache.keys().next().value;
   114→      if (oldest) consensusCache.delete(oldest);
   115→    }
   116→    return val;
   117→  })().finally(() => consensusInFlight.delete(key));
   118→  consensusInFlight.set(key, p);
   119→  return (await p as T | null) ?? null;
   120→}

   155→export async function fetchFlowMetrics(symbol: string): Promise<FlowMetrics | null> {
   156→  const cached = flowCache.get(symbol);
   157→  const cachedTtl = cached?.blind ? BLIND_CACHE_TTL : CACHE_TTL;
   158→  if (cached && Date.now() - cached.at < cachedTtl) return cached.metrics;
   159→  if (cached && Date.now() - cached.at < STALE_GRACE && flowInFlight.has(symbol)) return cached.metrics;
   160→
   161→  const existing = flowInFlight.get(symbol);
   162→  if (existing) return existing;
   163→
   164→  const p = (async (): Promise<FlowMetrics | null> => {
   165→    try {
   166→      const base = symbol.replace("USDT", "");
   167→      const [oiHist, takerLS, crowd, premium, aggTrades, topPos] = await Promise.allSettled([
   168→        fapi<Array<{ symbol: string; sumOpenInterest: string; sumOpenInterestValue: string; timestamp: number }>>(`/futures/data/openInterestHist?symbol=${symbol}&period=5m&limit=13`),
   169→        fapi<Array<{ symbol: string; buySellRatio: string; timestamp: number }>>(`/futures/data/takerlongshortRatio?symbol=${symbol}&period=5m&limit=13`),
   170→        fapi<Array<{ symbol: string; longShortRatio: string; longAccount: string; shortAccount: string; timestamp: number }>>(`/futures/data/globalLongShortAccountRatio?symbol=${symbol}&period=5m&limit=2`),
   171→        fapi<{ lastFundingRate: string; markPrice: string }>(`/fapi/v1/premiumIndex?symbol=${symbol}`),
   172→        fapi<AggTrade[]>(`/fapi/v1/aggTrades?symbol=${symbol}&limit=1000`),
   173→        fapi<Array<{ symbol: string; longShortRatio: string; timestamp: number }>>(`/futures/data/topLongShortPositionRatio?symbol=${symbol}&period=5m&limit=2`),
   174→      ]);
   175→
   176→      // v12.2 TASK-52 — FLOW BRIDGE: when a fapi lane starves (ban/pool silence),
   177→      // recover the recoverable lanes over the bridge instead of sliding into
   178→      // blindness. Honest sourcing: the bundle records the venue that fed it.
   179→      let flowSrc: FlowSource | "MIXED" = "BINANCE";
   180→      const srcNote = (s: FlowSource): void => {
   181→        if (flowSrc === "BINANCE") flowSrc = s;
   182→        else if (flowSrc !== s) flowSrc = "MIXED";
   183→      };
   184→
   185→      let oi = oiHist.status === "fulfilled" ? oiHist.value : [];
   186→      if (oi.length === 0) {
   187→        const b = await bridgeOiHist(symbol, "5m", 13).catch(() => null);

   195→            const lastOi = Number(oi[oi.length - 1]!.sumOpenInterest);
   196→            const notional = t?.openInterestValue ?? (t && lastOi > 0 ? t.lastPrice * lastOi : 0);
   197→            if (notional > 0) oi[oi.length - 1]!.sumOpenInterestValue = String(Math.round(notional));
   198→          }
   199→        }
   200→      }
   201→
   202→      const taker = takerLS.status === "fulfilled" ? takerLS.value : [];
   203→      const crowdArr = crowd.status === "fulfilled" ? crowd.value : [];
   204→      // crowd/top-trader L/S ratios have NO public equivalent on any other
   205→      // venue — during a fapi starvation they stay null (honest absence).
   206→
   207→      let prem = premium.status === "fulfilled" ? premium.value : null;
   208→      if (!prem) {
   209→        const t = await bybitTickerRow(symbol).catch(() => null);
   210→        if (t && t.fundingRate !== null) {
   211→          prem = { lastFundingRate: String(t.fundingRate), markPrice: String(t.markPrice ?? t.lastPrice) };
   212→          srcNote("BYBIT");

   217→      let oiDelta1h: number | null = null;
   218→      let oiValueUsd: number | null = null;

   244→      // (trades ≤1000 ≈ last minutes on liquid symbols; windows clipped by T).

   252→          srcNote(b.source);

   254→      }
   255→      const t60 = Date.now() - 60_000;
   256→      const t300 = Date.now() - 300_000;
   257→      let buy60 = 0;
   258→      let sell60 = 0;
   259→      const wp = { count: 0, totalUsd: 0, maxUsd: 0, buyUsd: 0, sellUsd: 0 };
   260→      for (const t of agg) {
   261→        const usd = t.p * t.q;
   262→        if (t.T >= t60) {
   263→          if (t.m) sell60 += usd;
   264→          else buy60 += usd;
   265→        }
   266→        if (t.T >= t300 && usd >= WHALE_PRINT_MIN_USD) {
   267→          wp.count++;
   268→          wp.totalUsd += usd;
   269→          if (usd > wp.maxUsd) wp.maxUsd = usd;
   270→          if (t.m) wp.sellUsd += usd;
   271→          else wp.buyUsd += usd;
   272→        }
   273→      }
   274→      const cvdUsd60s = agg.length > 0 ? Math.round(buy60 - sell60) : null;
   275→      const whalePrint5m: WhalePrint5m | null = wp.count > 0 ? { count: wp.count, totalUsd: Math.round(wp.totalUsd), maxUsd: Math.round(wp.maxUsd), buyUsd: Math.round(wp.buyUsd), sellUsd: Math.round(wp.sellUsd) } : null;
   276→
   277→      // v12.2 TASK-52: when the fapi taker-ratio series is absent but a REAL
   278→      // tape exists (bridge), the net-buy % of volume is measured from that
   279→      // same tape — window-capped at 10min (a 1000-trade slice spanning hours
   280→      // on an illiquid symbol is a stale read, not a flow read).
   281→      const winStart = agg.length > 0 ? agg[0]!.T : 0;
   282→      let buyWin = 0;
   283→      let sellWin = 0;
   284→      for (const t of agg) {
   285→        const usd = t.p * t.q;
   286→        if (t.m) sellWin += usd;
   287→        else buyWin += usd;
   288→      }
   289→      const cvdFromTape = agg.length > 0 && buyWin + sellWin > 0 && winStart > 0 && Date.now() - winStart <= 600_000 ? ((buyWin - sellWin) / (buyWin + sellWin)) * 100 : null;
   290→      const cvdNetPctOfVol = takerMean !== null ? (takerMean - 1) * 100 : cvdFromTape; // % of taker volume net buys
   291→
   292→      const metrics: FlowMetrics = {
   293→        oiDelta10m: oiDelta10m === null ? null : Number(oiDelta10m.toFixed(3)),
   294→        oiDelta1h: oiDelta1h === null ? null : Number(oiDelta1h.toFixed(3)),
   295→        oiValueUsd,
   296→        cvdNetBase: null,
   297→        cvdNetPctOfVol: cvdNetPctOfVol === null ? null : Number(cvdNetPctOfVol.toFixed(3)),
   298→        takerRatio,
   299→        takerMean,
   300→        crowdLongShort,
   301→        crowdLongPct,
   302→        topLongShort: topLongShort === null ? null : Number(topLongShort.toFixed(3)),
   303→        funding8h: funding8h === null ? null : Number(funding8h.toFixed(4)),
   304→        cvdUsd60s,
   305→        whalePrint5m,
   306→      };
   307→      const blind = flowCoverageOf(metrics) === 0;
   308→      flowCache.set(symbol, { at: Date.now(), metrics, blind });
   309→      flowSrcMap.set(symbol, flowSrc);
   310→
   311→      // history ring (fresh samples only)
   312→      if (metrics.oiDelta1h !== null) {
   313→        const ring = flowHistory.get(symbol) ?? [];
   314→        ring.push({ at: Date.now(), oiDelta1h: metrics.oiDelta1h });
   315→        if (ring.length > HISTORY_CAP) ring.shift();
   316→        flowHistory.set(symbol, ring);
   317→      }
   318→      return metrics;
   319→    } catch {
   320→      return null;
   321→    } finally {
   322→      flowInFlight.delete(symbol);
   323→    }
   324→  })();
   325→
   326→  flowInFlight.set(symbol, p);
   327→  return p;
   328→}

   330→export async function ensureFlow(symbols: string[], concurrency = 3): Promise<void> {
   331→  const queue = [...symbols];
   332→  const workers = Array.from({ length: Math.min(concurrency, queue.length) }, async () => {
   333→    while (queue.length > 0) {
   334→      const s = queue.shift();
   335→      if (s) await fetchFlowMetrics(s);
   336→    }
   337→  });
   338→  await Promise.all(workers);
   339→}

   351→/**
   352→ * v6.7 SMC-lite liquidity-sweep detector (operator-approved package item ⑥):
   353→ * the last CLOSED bar trades through the prior 24-bar swing high/low but closes
   354→ * back inside with a ≥50% rejection wick — the classic liquidity-grab print.
   355→ * `supportive` marks whether the sweep aligns with the candidate direction.
   356→ */
   357→export function detectLiquiditySweep(
   358→  candles: Array<{ high: number; low: number; close: number }> | undefined | null,
   359→  direction: "LONG" | "SHORT"
   360→): { swept: boolean; supportive: boolean; detail: string } {
   361→  if (!candles || candles.length < 26) return { swept: false, supportive: false, detail: "" };
   362→  const c = candles[candles.length - 1]!;
   363→  const prior = candles.slice(candles.length - 25, candles.length - 1);
   364→  const swingHigh = Math.max(...prior.map((k) => k.high));
   365→  const swingLow = Math.min(...prior.map((k) => k.low));
   366→  const range = c.high - c.low;
   367→  if (range <= 0) return { swept: false, supportive: false, detail: "" };
   368→  if (c.high > swingHigh && c.close < swingHigh && (c.high - c.close) / range >= 0.5) {
   369→    const detail = `SMC sweep: bar traded ${(((c.high - swingHigh) / swingHigh) * 100).toFixed(2)}% above the 24-bar swing high then closed back inside (${Math.round(((c.high - c.close) / range) * 100)}% rejection wick) — buy-side liquidity grabbed`;
   370→    return { swept: true, supportive: direction === "SHORT", detail: supportive_(detail, direction === "SHORT") };
   371→  }
   372→  if (c.low < swingLow && c.close > swingLow && (c.close - c.low) / range >= 0.5) {
   373→    const detail = `SMC sweep: bar pierced ${(((swingLow - c.low) / swingLow) * 100).toFixed(2)}% below the 24-bar swing low then closed back inside (${Math.round(((c.close - c.low) / range) * 100)}% rejection wick) — sell-side liquidity grabbed`;
   374→    return { swept: true, supportive: direction === "LONG", detail: supportive_(detail, direction === "LONG") };
   375→  }
   376→  return { swept: false, supportive: false, detail: "" };
   377→}

   379→function supportive_(detail: string, ok: boolean): string {
   380→  return ok ? `${detail}, supports the candidate direction` : detail;
   381→}

   383→/** Score flow along the candidate direction → ±25 (v4.5 OI-notional weighted). */
   384→export async function scoreFlow(symbol: string, direction: "LONG" | "SHORT", price1hAgoPct: number | null): Promise<FlowAssessment> {
   385→  const m = await fetchFlowMetrics(symbol);
   386→  if (!m) {
   387→    return { verdict: "UNAVAILABLE", score: 0, reasons: ["flow metrics unavailable — no numbers invented"], metrics: { oiDelta10m: null, oiDelta1h: null, oiValueUsd: null, cvdNetBase: null, cvdNetPctOfVol: null, takerRatio: null, takerMean: null, crowdLongShort: null, crowdLongPct: null, topLongShort: null, funding8h: null, cvdUsd60s: null, whalePrint5m: null }, oiWeight: 1, calibrated: false, coverage: 0, blind: true };
   388→  }
   389→  const coverage = flowCoverageOf(m);
   390→  const dir = direction === "LONG" ? 1 : -1;
   391→  const reasons: string[] = [];
   392→  // v12.1 Task-48 ② — HONEST NO-DATA: a fetch that completed with every
   393→  // decision metric null (fapi ban window / budget shed) is BLINDNESS, not a
   394→  // neutral book. The 48h census showed 70% of deploys rode this state while
   395→  // the card read "FLOW-NEUTRAL". Blind flow never votes (score 0) but now
   396→  // carries the NO-DATA verdict so cards, journals and gates can tell the
   397→  // difference. (Gate on blindness = EngineSetting "flow-gate", default
   398→  // tag-only: 10d ledger blind +$204.20 WR 57.1% vs sighted +$7.11 WR 52.8% —
   399→  // blocking the blind cohort would have CUT +$204 of net.)
   400→  if (coverage === 0) {
   401→    return { verdict: "NO-DATA", score: 0, reasons: ["flow-blind — every decision metric null (fapi-gate starvation/ban window); no numbers invented, no neutral read claimed"], metrics: m, oiWeight: 1, calibrated: true, coverage: 0, blind: true };
   402→  }
   403→  // v12.2 TASK-52 — honest sourcing: when the bundle rode the bridge, say so
   404→  // on the card/journal instead of silently mixing venues.
   405→  const bridgeSrc = getFlowSource(symbol);
   406→  if (bridgeSrc && bridgeSrc !== "BINANCE") {
   407→    reasons.push(`flow source: ${bridgeSrc} bridge (fapi starved — flow numbers measured on the bridge venue, not Binance)`);
   408→  }
   409→  let raw = 0;
   410→
   411→  // 1. CVD imbalance ±9
   412→  if (m.cvdNetPctOfVol !== null) {
   413→    const aligned = m.cvdNetPctOfVol * dir;

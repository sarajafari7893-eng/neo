   136→function note429(): void {
   137→  // v8.4.1 N5 — escalation hygiene: a 429 observed INSIDE an open cooldown
   138→  // window must not extend or escalate it. Operator-tier attempts (and the N3
   139→  // probe) legitimately pass through the window; without this guard each such
   140→  // attempt inflated the streak and pushed recovery further away — the exact
   141→  // "429 COOLDOWN 37s → 240s → 8m" treadmill the operator reported.
   142→  if (Date.now() < llm429Until) {
   143→    console.warn(`[ai] 429 inside open cooldown window — window keeps its course (streak ${llm429Streak}, no escalation)`);
   144→    return;
   145→  }
   146→  llm429Streak += 1;
   147→  const cd = Math.min(30_000 * Math.pow(2, llm429Streak - 1), LLM_COOLDOWN_CAP_MS); // 30s→60s→2m→4m→8m→10m cap
   148→  llm429Until = Date.now() + cd;
   149→  console.warn(`[ai] 429 budget shield: cooldown ${Math.round(cd / 1000)}s (streak ${llm429Streak}) — background tier parked, operator tier protected`);
   150→}

   158→/** Live LLM health for the supervisor/UI — is the neural link cooling down? */
   159→export function llmHealth(): { cooling: boolean; cooldownRemainsMs: number; consecutive429: number; lastOkAt: number | null; bgUsed: number; bgBudget: number } {
   160→  const now = Date.now();
   161→  const cooling = now < llm429Until;
   162→  const bucket = Math.floor(now / 3_600_000);
   163→  const bgUsed = bucket === bgHourBucket ? bgCallsThisHour : 0;
   164→  return { cooling, cooldownRemainsMs: cooling ? llm429Until - now : 0, consecutive429: llm429Streak, lastOkAt: lastLlmOkAt || null, bgUsed, bgBudget: BG_BUDGET_PER_HOUR };
   165→}

   167→// v8.4.1 N3 — ACTIVE RECOVERY PROBE: the shield is passive (it only learns
   168→// the link is back when real traffic happens to fire). The probe makes
   169→// recovery ACTIVE: one minimal "pong" call punches through the cooldown
   170→// (operator-tier semantics — the probe exists to test THROUGH the window);
   171→// success resets the shield instantly so MODEL verdicts resume without
   172→// waiting for the next natural call. Auto-probe rides the engine tick every
   173→// 5min while the link is cooling or no call has succeeded in 15min.
   174→let probeInFlight = false;
   175→let lastProbeAt = 0;
   176→let lastAutoProbeAt = 0;
   177→const PROBE_MIN_SPACING_MS = 120_000; // manual/API probes: ≥1 per 2min
   178→const AUTO_PROBE_SPACING_MS = 300_000; // background probes: ≥1 per 5min
   179→const AUTO_PROBE_IDLE_MS = 900_000; // auto-probe when no OK for 15min (or cooling)
   180→
   181→export async function probeNeuralLink(): Promise<{ ok: boolean; detail: string; health: ReturnType<typeof llmHealth> }> {
   182→  const now = Date.now();
   183→  if (probeInFlight) return { ok: false, detail: "probe already in flight", health: llmHealth() };
   184→  if (now - lastProbeAt < PROBE_MIN_SPACING_MS)
   185→    return { ok: false, detail: `probe spacing — retry in ${Math.ceil((PROBE_MIN_SPACING_MS - (now - lastProbeAt)) / 1000)}s`, health: llmHealth() };
   186→  probeInFlight = true;
   187→  lastProbeAt = now;
   188→  try {
   189→    const zai = await getZAI();
   190→    await Promise.race([
   191→      zai.chat.completions.create({ messages: [{ role: "user", content: "Reply with exactly: pong" }] }),
   192→      new Promise<never>((_, reject) => {
   193→        const t = setTimeout(() => reject(new Error(`probe timeout after ${AI_CALL_TIMEOUT_MS}ms`)), AI_CALL_TIMEOUT_MS);
   194→        t.unref?.();
   195→      }),
   196→    ]);
   197→    noteLlmOk();
   198→    console.log("[ai] neural probe OK — link restored, 429 shield reset");
   199→    return { ok: true, detail: "pong — neural link OK", health: llmHealth() };
   200→  } catch (err) {
   201→    if (is429(err)) note429(); // N5 decides whether this escalates (in-window → no)
   202→    const detail = String(err).slice(0, 140);
   203→    console.warn(`[ai] neural probe failed: ${detail}`);
   204→    return { ok: false, detail, health: llmHealth() };
   205→  } finally {
   206→    probeInFlight = false;
   207→  }
   208→}

   212→export async function maybeNeuralAutoProbe(): Promise<{ ok: boolean } | null> {
   213→  try {
   214→    const now = Date.now();

   222→  } catch {
   223→    return null;
   224→  }

   239→async function withZai<T>(fn: (zai: Awaited<ReturnType<typeof ZAI.create>>) => Promise<T>, opts?: { tier?: "operator" | "background" }): Promise<T> {
   240→  if (opts?.tier === "background" && Date.now() < llm429Until) {
   241→    throw new Error(`LLM 429-cooldown ${Math.ceil((llm429Until - Date.now()) / 1000)}s — background call skipped, heuristic tier`);
   242→  }
   243→  // v10.16 audit (watchdog memory-action channel): under OOM-proximity pressure
   244→  // both watchdog layers write .data/ai/background-suspend (10min lease) — the
   245→  // background tier fails fast so the box keeps RAM for the serving paths.
   246→  if (opts?.tier === "background" && memSuspendActive()) {
   247→    throw new Error("memory-pressure suspend active (.data/ai/background-suspend) — background LLM call skipped, heuristic tier");
   248→  }
   249→  if (opts?.tier === "background") {
   250→    const bucket = Math.floor(Date.now() / 3_600_000);
   251→    if (bucket !== bgHourBucket) {
   252→      bgHourBucket = bucket;
   253→      bgCallsThisHour = 0;
   254→    }
   255→    if (bgCallsThisHour >= BG_BUDGET_PER_HOUR) {
   256→      throw new Error(`LLM background budget exhausted (${bgCallsThisHour}/${BG_BUDGET_PER_HOUR} this hour) — call skipped, heuristic tier`);
   257→    }
   258→    bgCallsThisHour += 1;
   259→  }
   260→  const attempt = async (): Promise<T> => {
   261→    const zai = await getZAI();
   262→    return Promise.race([
   263→      fn(zai),
   264→      new Promise<never>((_, reject) => {
   265→        const t = setTimeout(() => reject(new Error(`AI call timeout after ${AI_CALL_TIMEOUT_MS}ms`)), AI_CALL_TIMEOUT_MS);
   266→        t.unref?.();
   267→      }),
   268→    ]) as Promise<T>;
   269→  };
   270→  let result: T;
   271→  try {
   272→    result = await attempt();
   273→  } catch (err) {
   274→    if (is429(err)) note429();
   275→    if (retriableAuth(err)) {
   276→      console.warn("[ai] auth failure — reloading rotated .z-ai-config and retrying once");
   277→      try {
   278→        result = await attempt();
   279→      } catch (retryErr) {
   280→        if (is429(retryErr)) note429();
   281→        // escalate to the AI SUPERVISOR (dynamic import — avoids ai↔supervisor cycle)
   282→        void import("./supervisor")
   283→          .then((m) => m.noteLlmAuthFailure())
   284→          .catch(() => undefined);
   285→        throw retryErr;
   286→      }
   287→    } else {
   288→      throw err;
   289→    }
   290→  }
   291→  noteLlmOk();
   292→  return result;
   293→}

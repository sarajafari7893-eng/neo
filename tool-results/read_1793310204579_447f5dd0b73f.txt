   645→  // silently upsizing would breach the fixed-margin contract.
   646→  // v1.4.3 — floors are REAL notional (multiplier-aware).
   647→  const floorNotional = Math.max(instr.minNotional, instr.minQty * refPrice * mult, S.floorLearn?.[it.tbSymbol] ?? 0);
   648→  if (floorNotional > notional) {
   649→    it.status = "SKIP";
   650→    it.events.push({ at: Date.now(), ev: "MIN_FLOOR_SKIP", note: `${it.tbSymbol} floor=${floorNotional.toFixed(2)} > marginNotional=${notional}${S.floorLearn?.[it.tbSymbol] ? " (learned)" : ""} ` });
   651→    countSkip("MIN-FLOOR");
   652→    audit("EXEC-SKIP", { serial: it.serial, why: `exchange floor ${floorNotional.toFixed(2)} USDT exceeds fixed-margin notional ${notional} USDT` });
   653→    return;
   654→  }
   655→  qty = floorStep(Math.min(qty, instr.maxQty), instr.stepSize);
   656→  if (qty < instr.minQty) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "QTY_BELOW_MIN" }); audit("EXEC-FAIL", { serial: it.serial, why: `qty ${qty} < minQty ${instr.minQty}` }); return; }
   657→
   658→  const tSent = Date.now();
   659→  let fillPrice: number | null = null;
   660→  let orderId: string | null = null;
   661→  if (liveMode) {
   662→    await ensureLeverage(it.tbSymbol, it.leverage);
   663→    const side = it.direction === "LONG" ? "BUY_OPEN" : "SELL_OPEN";
   664→    const r = await placeOrder(it.tbSymbol, side, qty, { takeProfit: it.tp3, stopLoss: it.stopLoss, clientOrderId: it.clientOrderId });
   665→    audit("EXEC-OPEN", { serial: it.serial, symbol: it.tbSymbol, side, qty, lev: it.leverage, tp3: it.tp3, sl: it.stopLoss, ok: r.ok, orderId: r.orderId, err: r.err });
   666→    if (!r.ok) {
   667→      // v1.4.2 — server-side real floor: exchangeInfo hides it (minNotional=0)
   668→      // but the book rejects with -1199/-1202 "quantity too small". Teach the
   669→      // floor, reclassify as an honest MIN-FLOOR SKIP, keep the ledger truthful.
   670→      if (/quantity too small/i.test(r.err ?? "")) {
   671→        const prevFloor = S.floorLearn?.[it.tbSymbol] ?? 0;
   672→        S.floorLearn[it.tbSymbol] = Math.max(prevFloor, notional);
   673→        it.status = "SKIP";
   674→        it.events.push({ at: Date.now(), ev: "MIN_FLOOR_LEARNED", note: `${it.tbSymbol} real floor ≥ ${notional} USDT (server -1199/-1202; exchangeInfo said ${floorNotional.toFixed(2)})` });
   675→        countSkip("MIN-FLOOR");
   676→        audit("EXEC-SKIP", { serial: it.serial, why: `server floor learned: ${it.tbSymbol} ≥ ${notional} USDT notional (rejected quantity-too-small; exchangeInfo floor ${floorNotional.toFixed(2)})` });
   677→        persistState();
   678→        return;
   679→      }
   680→      it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "OPEN_REJECTED", note: (r.err ?? "").slice(0, 160) }); return;
   681→    }
   682→    orderId = r.orderId;
   683→    fillPrice = await resolveFill(it.tbSymbol, orderId, refPrice);
   684→  } else {
   685→    fillPrice = refPrice; // paper fill on the real toobit book (taker side)
   686→  }
   687→  const tFill = Date.now();
   688→  it.status = "OPEN";
   689→  it.qtyOrig = qty;
   690→  it.qtyRemaining = qty;
   691→  it.orderId = orderId;
   692→  it.fillPrice = fillPrice;
   693→  it.openedAt = tFill;
   694→  // v1.3.0 doctrine bookkeeping — the truth of what the book actually saw:
   695→  // step-size floor rounding makes actualNotional ≤ expectedNotional by design.
   696→  // v1.4.3 — actual notional is REAL: qty(contracts) × fill × contractMultiplier.
   697→  it.actualNotional = fillPrice ? Number((qty * fillPrice * mult).toFixed(6)) : null;
   698→  it.protected = it.mode === "LIVE" ? true : true; // LIVE: attached TP/SL in same order + recon re-check; PAPER: sim active
   699→  S.doctrine.opens++;
   700→  const ndelta = it.expectedNotional && it.actualNotional ? Math.abs(it.actualNotional - it.expectedNotional) / it.expectedNotional * 10_000 : 0;
   701→  audit("EXEC-OPEN-DOCTRINE", { serial: it.serial, lev: it.leverage, levSrc: it.levSrc ?? "card", margin: Number(effMargin.toFixed(4)), riskCapped: cap.capped, contractMult: mult, expectedNotional: it.expectedNotional, actualNotional: it.actualNotional, notionalDeltaBps: Number(ndelta.toFixed(1)), entryAgeMs: it.entryAgeMs, entryDriftBps: it.entryDriftBps });
   702→  if (ndelta > 2000) doctrineViolation(it.serial, "NOTIONAL-DELTA", `exp=${it.expectedNotional} act=${it.actualNotional} (${ndelta.toFixed(0)}bps) — step-rounding or minQty floor`);
   703→  it.lat = {
   704→    detectMs: it.lat.detectMs ?? null,
   705→    execMs: tFill - tSent,
   706→    totalMs: it.createdAt0 !== undefined ? tFill - it.createdAt0 : null,
   707→    slipBps: fillPrice && it.cardEntry > 0 ? ((it.direction === "LONG" ? fillPrice - it.cardEntry : it.cardEntry - fillPrice) / it.cardEntry) * 10_000 : null,
   708→  };
   709→  it.events.push({ at: tFill, ev: "OPEN", note: `${it.mode} qty=${qty} mult=×${mult} fill=${fillPrice}` });
   710→  S.metrics.push({ serial: it.serial, symbol: it.symbol, direction: it.direction, mode: it.mode, detectMs: it.lat.detectMs, execMs: it.lat.execMs, totalMs: it.lat.totalMs, slipBps: it.lat.slipBps, fillPrice, cardEntry: it.cardEntry, at: tFill });
   711→  if (S.metrics.length > METRICS_WINDOW) S.metrics = S.metrics.slice(-METRICS_WINDOW);
   712→}

   714→// resolve the real average fill price of a market order (best effort)
   715→async function resolveFill(tbSymbol: string, orderId: string | null, fallback: number): Promise<number> {
   716→  if (orderId) {
   717→    for (let i = 0; i < 3; i++) {
   718→      await new Promise((r) => setTimeout(r, 350));
   719→      const r2 = await signed<{ status?: string; orderStatus?: string; avgPrice?: string; price?: string; executedQty?: string; cumQty?: string }>("GET", "/api/v1/futures/order", { orderId });
   720→      const d = r2.data as Record<string, unknown> | null;
   721→      const st = String(d?.status ?? d?.orderStatus ?? "").toUpperCase();
   722→      const avg = Number(d?.avgPrice ?? d?.price ?? 0);
   723→      if (avg > 0) return avg;
   724→      if (/CANCELED|REJECTED|EXPIRED/.test(st)) break;
   725→      if (st === "FILLED" || i === 2) break;
   726→    }
   727→  }
   728→  return fallback;
   729→}

   731→// v1.4.3 LIVE PnL TELEMETRY — exchange-verified realized PnL for an intent's
   732→// lifetime (fills since openedAt; realizedPnL arrives on closing fills).
   733→// This is the money-truth behind «چرا هیچ سیگنالی در سود بسته نشده».
   734→async function fetchRealizedPnl(it: Intent): Promise<number | null> {
   735→  try {
   736→    // v1.4.5 — a NEVER-EXECUTED intent (no openedAt) owns ZERO exchange fills;
   737→    // with since=0 the symbol's whole fill history was once re-summed into it
   738→    // (phantom PnL: ENA -0.7445 = morning -0.6893-0.0552, BTW -3.2605, PEPE -0.1066).
   739→    if (!it.openedAt) return null;
   740→    const r = await signed<Array<{ time?: string; realizedPnl?: string; commission?: string; makerRebate?: string }>>("GET", "/api/v1/futures/userTrades", { symbol: it.tbSymbol, limit: 100 });
   741→    const rows = Array.isArray(r.data) ? r.data : [];
   742→    const since = it.openedAt;
   743→    let pnl = 0, seen = false;
   744→    for (const f of rows) {
   745→      if (Number(f.time ?? 0) < since) continue;
   746→      seen = true;
   747→      pnl += Number(f.realizedPnl ?? 0) - Number(f.commission ?? 0) + Number(f.makerRebate ?? 0);
   748→    }
   749→    if (!seen) return null;
   750→    return Number(pnl.toFixed(4));
   751→  } catch { return null; }
   752→}
   753→// revert the card-status transition so the next poll re-fires the failed
   754→// action (v1.4.3 — transitions are no longer one-shot on transient rejections;
   755→// capped so a permanently-rejected action cannot loop forever).
   756→function revertTransition(it: Intent, what: string, err: string): void {
   757→  const attempts = (it.actionAttempts ?? 0) + 1;
   758→  it.actionAttempts = attempts;
   759→  if (attempts <= 20 && it.prevStatusSeen) {
   760→    it.lastStatusSeen = it.prevStatusSeen;
   761→    it.events.push({ at: Date.now(), ev: `${what}_RETRY-ARMED`, note: err.slice(0, 140) });
   762→    audit("EXEC-RETRY-ARMED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   763→  } else {
   764→    it.events.push({ at: Date.now(), ev: `${what}_ABANDONED`, note: `attempts=${attempts} — will not retry: ${err.slice(0, 100)}` });
   765→    audit("EXEC-RETRY-ABANDONED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   766→  }
   767→  persistState();
   768→}

   770→async function doBank(it: Intent, fracOfOrig: number, reason: string): Promise<void> {
   771→  if (it.status !== "OPEN" || it.qtyRemaining <= 0) return;
   772→  let qty = floorStep(Math.min(it.qtyOrig * fracOfOrig, it.qtyRemaining), instruments.get(it.tbSymbol)?.stepSize ?? 0);
   773→  if (qty <= 0) return;
   774→  if (qty >= it.qtyRemaining) return doClose(it, reason);
   775→  if (it.mode === "LIVE") {
   776→    const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   777→    // v1.4.3 — clientOrderId is mandatory on Toobit (pre-1.4.3 every bank was
   778→    // rejected -1004 → no signal ever closed in profit).
   779→    const r = await placeOrder(it.tbSymbol, side, qty, { clientOrderId: `NX${it.serial}${reason}${Date.now() % 100000}` });
   780→    audit("EXEC-BANK", { serial: it.serial, reason, qty, ok: r.ok, err: r.err });
   781→    if (!r.ok) { it.events.push({ at: Date.now(), ev: "BANK_REJECTED", note: (r.err ?? "").slice(0, 160) }); revertTransition(it, "BANK", r.err ?? "bank order rejected"); return; }
   782→  } else {
   783→    // v1.4.0 — PAPER سیو سود was invisible in audit.jsonl (only in state.json);
   784→    // the paper fill rides the REAL toobit book so it deserves the same trail.
   785→    audit("EXEC-BANK-PAPER", { serial: it.serial, reason, qty, note: "paper fill on real book" });
   786→  }
   787→  it.qtyRemaining = Number((it.qtyRemaining - qty).toFixed(10));
   788→  if (reason === "TP1") it.qtyBankedTp1 = qty;
   789→  if (reason === "TP2") it.qtyBankedTp2 = qty;
   790→  it.actionAttempts = 0;
   791→  it.events.push({ at: Date.now(), ev: reason, note: `banked ${qty} (سیو سود)` });
   792→}

   794→async function doClose(it: Intent, reason: string): Promise<void> {
   795→  if (it.status === "CLOSED" || it.status === "CLOSING") return;
   796→  it.status = "CLOSING";
   797→  if (it.mode === "LIVE" && it.qtyRemaining > 0) {
   798→    const flash = await flashClose(it.tbSymbol, it.direction);
   799→    audit("EXEC-CLOSE", { serial: it.serial, reason, flash: flash.ok, err: flash.err });
   800→    if (!flash.ok) {
   801→      // fallback: explicit MARKET close of the remaining qty
   802→      const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   803→      // v1.4.3 — clientOrderId mandatory (pre-1.4.3 the SL-close fallback was
   804→      // rejected -1004 and the transition was lost forever).
   805→      const r = await placeOrder(it.tbSymbol, side, it.qtyRemaining, { clientOrderId: `NX${it.serial}X${Date.now() % 100000}` });
   806→      audit("EXEC-CLOSE-FALLBACK", { serial: it.serial, reason, ok: r.ok, err: r.err });
   807→      if (!r.ok) {
   808→        it.status = "OPEN";
   809→        it.events.push({ at: Date.now(), ev: "CLOSE_REJECTED", note: (r.err ?? "").slice(0, 160) });
   810→        revertTransition(it, "CLOSE", r.err ?? "close order rejected");
   811→        return;
   812→      }
   813→    }
   814→  }
   815→  it.qtyRemaining = 0;
   816→  it.status = "CLOSED";
   817→  it.closedAt = Date.now();
   818→  it.closeReason = reason;
   819→  it.actionAttempts = 0;
   820→  if (it.mode === "LIVE") {
   821→    // v1.4.5 — DETECTed-but-never-sent intents (exec gate off / gates skip) own
   822→    // no exchange fills: closing them must NOT sum the symbol's fill history.
   823→    const neverExecuted = !it.orderId && !it.openedAt;
   824→    if (neverExecuted) {
   825→      it.closeReason = `${reason} (بدون اجرا)`;
   826→      audit("EXEC-PNL-SKIP", { serial: it.serial, reason: it.closeReason, why: "intent never executed — no exchange fills belong to it" });
   827→    } else {
   828→      const pnl = await fetchRealizedPnl(it); // v1.4.3 — money truth of this close
   829→      if (pnl !== null) it.realizedPnlUsdt = pnl;
   830→      audit("EXEC-PNL", { serial: it.serial, reason, realizedPnlUsdt: pnl });
   831→    }
   832→  }
   833→  it.events.push({ at: it.closedAt, ev: "CLOSED", note: reason });
   834→  persistState();
   835→}

   837→// ── card poll — the lifecycle mirror ────────────────────────────────────────
   838→let pollBusy = false;
   839→async function pollCards(): Promise<void> {
   840→  if (pollBusy) return;
   841→  pollBusy = true;
   842→  try {
   843→    if (instrumentsAt === 0 || Date.now() - instrumentsAt > 10 * 60_000) await refreshInstruments();
   844→    const cards = fetchRecentCards(60);
   845→    if (cards.length === 0) return;
   846→    const newest = Math.max(...cards.map((c) => parseTs(c.createdAt)));
   847→    if (newest > S.lastCreatedAtMs) S.lastCreatedAtMs = newest;
   848→
   849→    // ⓪ v1.4.0 RETRY WINDOW — a freed capacity slot inside the ENTRY-DELAY
   850→    // window re-opens a MAX-CONCURRENT-skipped card (was: skipped forever).
   851→    if (retryList.length > 0 && !killFile() && S.cfg.execEnabled) {
   852→      const openCount0 = Object.values(S.intents).filter((x) => x.status === "OPEN" || x.status === "OPENING").length;
   853→      if (S.cfg.maxConcurrent > 0 && openCount0 < S.cfg.maxConcurrent) {
   854→        const nowMs = Date.now();
   855→        retryList = retryList.filter((r) => {
   856→          const it = S.intents[r.cardId];
   857→          if (!it || it.status !== "SKIP") return false; // resolved otherwise
   858→          const age = nowMs - (it.createdAt0 ?? r.createdAt0);
   859→          if (age > S.cfg.maxEntryAgeMs) { audit("RETRY-EXPIRED", { serial: it.serial, ageSec: Math.round(age / 1000) }); return false; }
   860→          it.status = "OPENING";
   861→          it.events.push({ at: nowMs, ev: "RETRY", note: "capacity freed inside entry window" });
   862→          pushAction({ kind: "open", cardId: r.cardId, reason: "RETRY-MAX-CONCURRENT" });
   863→          audit("RETRY-OPEN", { serial: it.serial, symbol: it.symbol });
   864→          return false;
   865→        });
   866→      }
   867→    }
   868→
   869→    // ① brand-new cards → intent
   870→    for (const c of cards) {
   871→      if (S.intents[c.id]) continue;
   872→      if (c.status !== "ACTIVE") continue; // only live cards are executed
   873→      const created = parseTs(c.createdAt);
   874→      const age = Date.now() - created;
   875→      const tbSymbol = toobitSymbol(c.symbol);
   876→      const known = instruments.has(tbSymbol);
   877→      const openCount = Object.values(S.intents).filter((x) => x.status === "OPEN" || x.status === "OPENING").length;
   878→      let skip: string | null = null;
   879→      if (age > STALE_SKIP_MS) skip = "STALE-CARD";
   880→      else if (!known) skip = "SYMBOL-ABSENT-ON-TOOBIT";
   881→      else if (!killFile() && S.cfg.execEnabled && S.cfg.maxConcurrent > 0 && openCount >= S.cfg.maxConcurrent) skip = "MAX-CONCURRENT";
   882→      if (skip) countSkip(skip); // v1.3.0 doctrine funnel
   883→      if (skip === "MAX-CONCURRENT") retryList.push({ cardId: c.id, createdAt0: created }); // v1.4.0 — eligible for the retry window
   884→      const dir = c.direction === "SHORT" ? "SHORT" : "LONG";
   885→      const levOk = c.leverage >= 1 && c.leverage <= 50;
   886→      const intent: Intent = {
   887→        cardId: c.id, serial: c.serial, symbol: c.symbol, tbSymbol,
   888→        direction: dir,
   889→        leverage: levOk ? c.leverage : S.cfg.leverage, // v1.2.0 per-card (fallback: cfg for legacy rows)
   890→        levSrc: levOk ? "card" : "fallback", // v1.3.0 doctrine
   891→        status: skip ? "SKIP" : "OPENING", mode: isLiveMode() ? "LIVE" : "PAPER",
   892→        qtyOrig: 0, qtyRemaining: 0, qtyBankedTp1: 0, qtyBankedTp2: 0,
   893→        cardEntry: c.entryPrice, tp1: c.tp1, tp2: c.tp2, tp3: c.tp3, stopLoss: c.stopLoss,
   894→        clientOrderId: `NX${String(c.serial).padStart(4, "0")}`,
   895→        orderId: null, fillPrice: null, openedAt: null, closedAt: null, closeReason: null,
   896→        lat: { detectMs: age, execMs: null, totalMs: null, slipBps: null },
   897→        lastStatusSeen: c.status,
   898→        events: [{ at: Date.now(), ev: skip ? `SKIP:${skip}` : "DETECTED", note: `age=${age}ms` }],
   899→      };
   900→      intent.createdAt0 = created;
   901→      S.intents[c.id] = intent;
   902→      audit(skip ? "SKIP" : "DETECT", { serial: c.serial, symbol: c.symbol, direction: dir, age, skip });
   903→      if (!skip && !killFile() && S.cfg.execEnabled) pushAction({ kind: "open", cardId: c.id, reason: "CARD-ACTIVE" });
   904→    }
   905→
   906→    // ② tracked cards → status transitions
   907→    const liveIds = Object.keys(S.intents).filter((id) => {
   908→      const st = S.intents[id]?.status;
   909→      return st === "OPENING" || st === "OPEN";
   910→    }).slice(-TRACK_LIMIT);
   911→    if (liveIds.length > 0) {
   912→      const rows = fetchCardsByIds(liveIds);
   913→      const byId = new Map(rows.map((r) => [r.id, r]));
   914→      for (const id of liveIds) {
   915→        const it = S.intents[id]!;
   916→        const c = byId.get(id);
   917→        if (!c) continue;
   918→        // v1.4.3 — mirror engine-side protective mutations (breakeven stop after
   919→        // TP1, TP edits) for open intents. Pre-1.4.3 the stop moved only in the
   920→        // ledger — NX-3139 closed LIVE at a level the ledger had long replaced.
   921→        const slNow = c.stopLoss > 0 ? c.stopLoss : null;
   922→        const tp3Now = c.tp3 > 0 ? c.tp3 : null;
   923→        const slMoved = slNow !== null && Math.abs((it.stopLoss ?? 0) - slNow) / slNow > 1e-9;
   924→        const tp3Moved = tp3Now !== null && Math.abs((it.tp3 ?? 0) - tp3Now) / tp3Now > 1e-9;
   925→        if (it.status === "OPEN" && (slMoved || tp3Moved)) {
   926→          if (slMoved && slNow) it.stopLoss = slNow;
   927→          if (tp3Moved && tp3Now) it.tp3 = tp3Now;
   928→          if (it.mode === "LIVE") {
   929→            const side = it.direction === "LONG" ? "LONG" : "SHORT";
   930→            const r = await signed("POST", "/api/v1/futures/position/trading-stop", { symbol: it.tbSymbol, side, takeProfit: String(it.tp3), stopLoss: String(it.stopLoss), tpTriggerBy: "MARK_PRICE", slTriggerBy: "MARK_PRICE" });
   931→            audit("STOP-MOVE-MIRROR", { serial: it.serial, symbol: it.tbSymbol, stopLoss: it.stopLoss, tp3: it.tp3, ok: r.ok, err: r.err });
   932→          } else {
   933→            audit("STOP-MOVE-PAPER", { serial: it.serial, symbol: it.tbSymbol, stopLoss: it.stopLoss, tp3: it.tp3 });
   934→          }
   935→          persistState();
   936→        }
   937→        const prev = it.lastStatusSeen;
   938→        if (c.status === prev) continue;
   939→        it.prevStatusSeen = prev; // v1.4.3 — revert point for rejected actions
   940→        it.lastStatusSeen = c.status;
   941→        if (it.lastAttemptedStatus !== c.status) { it.actionAttempts = 0; it.lastAttemptedStatus = c.status; } // fresh budget only for a genuinely new status
   942→        it.cardStatus = c.status; // v1.3.0 doctrine — visible open-management mirror
   943→        if (c.status === "TP1") pushAction({ kind: "bank", cardId: id, frac: S.cfg.tp1Frac, reason: "TP1" });

   946→        else if (c.status === "SL") pushAction({ kind: "close", cardId: id, reason: "SL (استاپ شده)" });
   947→        else if (c.status === "EXPIRED") pushAction({ kind: "close", cardId: id, reason: "EXPIRED (سیگنال منقضی)" });
   948→        else if (c.status === "CLOSED") pushAction({ kind: "close", cardId: id, reason: "CLOSED" });
   949→        else if (c.status === "VOID") pushAction({ kind: "close", cardId: id, reason: "VOID (qualityFlag)" });
   950→        audit("CARD-EVENT", { serial: it.serial, from: prev, to: c.status });
   951→      }
   952→    }
   953→
   954→    // ③ paper-mode protective TP/SL sim (exchange-side stand-in)
   955→    for (const it of Object.values(S.intents)) {

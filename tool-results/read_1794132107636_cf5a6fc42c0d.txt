   168→async function apiCall<T>(method: string, body: unknown, timeoutMs = 9000): Promise<{ ok: boolean; result?: T; error?: string }> {
   169→  const ctrl = new AbortController();
   170→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   171→  try {
   172→    const res = await fetch(`https://api.telegram.org/bot${TOKEN}/${method}`, {
   173→      method: "POST",
   174→      headers: { "content-type": "application/json" },
   175→      body: JSON.stringify(body),
   176→      signal: ctrl.signal,
   177→      cache: "no-store",
   178→    });
   179→    const json = (await res.json()) as { ok: boolean; result?: T; description?: string };
   180→    return json.ok ? { ok: true, result: json.result } : { ok: false, error: json.description ?? `HTTP ${res.status}` };
   181→  } catch (e: unknown) {
   182→    return { ok: false, error: e instanceof Error ? (e.name === "AbortError" ? `timeout ${timeoutMs}ms` : e.message) : String(e) };
   183→  } finally {
   184→    clearTimeout(timer);
   185→  }
   186→}

   188→/** Multipart photo upload (sendPhoto) — FormData/Blob are global in Node 18+.
   189→ * v12.0.1: 20s (was 9s — multipart uploads through the starved process/egress
   190→ * pool were aborting into the text fallback), + ONE retry on timeout/5xx. */
   191→async function apiSendPhoto(chat: string, caption: string, photo: Buffer, filename: string): Promise<{ ok: boolean; error?: string }> {
   192→  let last = { ok: false, error: "unsent" } as { ok: boolean; error?: string };
   193→  for (let attempt = 0; attempt < 2; attempt++) {
   194→    const ctrl = new AbortController();
   195→    const timer = setTimeout(() => ctrl.abort(), 20_000);
   196→    try {
   197→      const isPng = filename.toLowerCase().endsWith(".png");
   198→      const form = new FormData();
   199→      form.append("chat_id", chat);
   200→      form.append("caption", caption.slice(0, 1024));
   201→      form.append("parse_mode", "HTML");
   202→      form.append("photo", new Blob([new Uint8Array(photo)], { type: isPng ? "image/png" : "image/jpeg" }), filename);
   203→      const res = await fetch(`https://api.telegram.org/bot${TOKEN}/sendPhoto`, { method: "POST", body: form, signal: ctrl.signal, cache: "no-store" });
   204→      const json = (await res.json()) as { ok: boolean; description?: string };
   205→      last = json.ok ? { ok: true } : { ok: false, error: json.description ?? `HTTP ${res.status}` };
   206→    } catch (e: unknown) {
   207→      last = { ok: false, error: e instanceof Error ? (e.name === "AbortError" ? "timeout 20000ms" : e.message) : String(e) };
   208→    } finally {
   209→      clearTimeout(timer);
   210→    }
   211→    if (last.ok) return last;
   212→    if (!(last.error ?? "").includes("timeout") && !/HTTP 5\d\d/.test(last.error ?? "")) return last; // only transient classes retry
   213→    await new Promise((r) => setTimeout(r, 1200));
   214→  }
   215→  return last;
   216→}

   249→async function renderAnimFrames(spec: BannerV2Spec): Promise<Buffer[]> {
   250→  const frames: Buffer[] = [];
   251→  for (let i = 0; i < ANIM_FRAMES; i++) {
   252→    const phase = i / ANIM_FRAMES;
   253→    const pulse = 0.5 - 0.5 * Math.cos(2 * Math.PI * phase); // smooth 0→1→0 loop
   254→    frames.push(await renderSignalBannerV2({ ...spec, pulse, phase }));
   255→  }

   259→/** 12 PNG frames in → H.264 MP4 out (≤20s hard kill). null on any failure —
   260→ * the static photo path takes over, the alert never waits. NOTE: the MP4
   261→ * muxer cannot write to a pipe ("does not support non seekable output") so
   262→ * the encode lands in a transient temp file, is read back and unlinked. */
   263→function encodeAnimMp4(frames: Buffer[]): Promise<Buffer | null> {
   264→  return new Promise((resolve) => {
   265→    const tmpPath = path.join(tmpdir(), `nexus-anim-${process.pid}-${Date.now()}.mp4`);
   266→    let child: ReturnType<typeof spawn>;
   267→    try {
   268→      child = spawn(
   269→        "ffmpeg",
   270→        ["-hide_banner", "-loglevel", "error", "-y", "-f", "image2pipe", "-framerate", String(ANIM_FPS), "-i", "-", "-c:v", "libx264", "-preset", "veryfast", "-crf", "23", "-pix_fmt", "yuv420p", "-movflags", "+faststart", tmpPath],
   271→        { stdio: ["pipe", "ignore", "pipe"] },
   272→      );
   273→    } catch {
   274→      resolve(null);
   275→      return;
   276→    }
   277→    let errTail = "";
   278→    let done = false;
   279→    const finish = (v: Buffer | null) => {
   280→      if (!done) {
   281→        done = true;
   282→        void fs.unlink(tmpPath).catch(() => undefined);
   283→        resolve(v);
   284→      }
   285→    };
   286→    const killer = setTimeout(() => {
   287→      child.kill("SIGKILL");
   288→      finish(null);
   289→    }, 20_000);
   290→    child.stderr?.on("data", (c: Buffer) => {
   291→      errTail = (errTail + c.toString()).slice(-300);
   292→    });
   293→    child.on("error", () => {
   294→      clearTimeout(killer);
   295→      finish(null);
   296→    });
   297→    child.on("close", (code) => {
   298→      clearTimeout(killer);
   299→      if (code !== 0) {
   300→        if (errTail) console.error(`[NOTIFY] anim encode failed (code ${code}): ${errTail.trim().split("\n").pop()}`);
   301→        finish(null);
   302→        return;
   303→      }
   304→      fs.readFile(tmpPath).then((buf) => finish(buf.length > 1024 ? buf : null)).catch(() => finish(null));
   305→    });
   306→    void (async () => {
   307→      try {
   308→        for (const f of frames) {
   309→          if (done) break;
   310→          if (!child.stdin!.write(f)) await new Promise<void>((r) => child.stdin!.once("drain", () => r()));
   311→        }
   312→        child.stdin!.end();
   313→      } catch {
   314→        child.kill("SIGKILL");
   315→      }
   316→    })();
   317→  });
   318→}

   810→// ── v6.2.2 AUTO-DISCOVER (operator round) ─────────────────────────────────
   811→// The operator's credential set shipped TELEGRAM_CHAT_ID === the bot's own id
   812→// (Telegram hard-rejects bot→bot sends), leaving the relay stuck in ARMED
   813→// (chat PENDING) until someone runs discoverChat() at exactly the right
   814→// moment. Instead of babysitting, this loop polls getUpdates every 30s WHILE
   815→// the chat is unresolved; the instant the operator messages the bot once,
   816→// the chat id is learned, persisted, confirmed back, and the loop retires.
   817→let autoDiscoverTimer: ReturnType<typeof setInterval> | null = null;
   818→
   819→function chatResolvedNow(): boolean {
   820→  const chat = chatOverride ?? ENV_CHAT;
   821→  return chat.length > 0 && botId !== null && chat !== String(botId);
   822→}

   824→export function ensureAutoDiscover(): void {
   825→  if (autoDiscoverTimer || !relayArmed()) return;
   826→  // v8.3 BRIDGE COEXISTENCE: the chat id is already persisted (EngineSetting
   827→  // telegram.chatId) since v6.2.2 — discovery is DONE. The legacy 30s
   828→  // getUpdates probe now 409-terminates the v8.3 bridge long-poller, so the
   829→  // loop may only start after a DB-backed override check FAILS, and each
   830→  // tick re-checks the override BEFORE touching getUpdates.
   831→  void (async () => {
   832→    try {
   833→      await ensureBotIdentity();
   834→      await loadOverride();
   835→    } catch {
   836→      /* identity check is best-effort */
   837→    }
   838→    if (chatResolvedNow()) {
   839→      // v10.16: rate-limit this line to once per 10min — it fired on every call
   840→      // and produced ~3.5k dev.log lines/day (audit: unbounded log growth).
   841→      const now = Date.now();
   842→      if (now - lastAutoDiscoverSkipLog > 600_000) {
   843→        lastAutoDiscoverSkipLog = now;
   844→        console.log("[NOTIFY] auto-discover SKIPPED — chat already resolved (v8.3 bridge owns getUpdates)");
   845→      }
   846→      return;
   847→    }
   848→    const timer: ReturnType<typeof setInterval> = setInterval(() => {
   849→      if (chatResolvedNow()) {
   850→        clearInterval(timer);
   851→        autoDiscoverTimer = null;
   852→        return;
   853→      }
   854→      void (async () => {
   855→        try {
   856→          await ensureBotIdentity();
   857→          await loadOverride();
   858→        } catch {
   859→          /* best-effort */
   860→        }
   861→        if (chatResolvedNow()) {
   862→          clearInterval(timer);
   863→          autoDiscoverTimer = null;
   864→          return;
   865→        }
   866→        const r = await discoverChat();
   867→        if (r && r.ok) {
   868→          clearInterval(timer);
   869→          autoDiscoverTimer = null;
   870→          const proof = await sendDirect(
   871→            `🟢 ${APP_NAME} relay ARMED — operator chat discovered and persisted.\nFrom now on CREATED / TP1-3 / SL / TTL_EXPIRED / TTL_EXTENDED / AFTERLIFE events mirror here as structured cards (WIN / STOP / EXPIRED outcomes also arrive as rendered banner images).`
   872→          );
   873→          console.log(`[NOTIFY] auto-discover complete — confirmation ${proof.ok ? "delivered" : `NOT delivered (${proof.error ?? "unknown"})`}`);
   874→        }
   875→        // ok:false → "no messages waiting" (benign, keep polling) or a
   876→        // transient network error (silent retry on the next 30s pass).
   877→      })();
   878→    }, 30_000);
   879→    autoDiscoverTimer = timer;
   880→    console.log("[NOTIFY] auto-discover loop started — polling getUpdates every 30s until the operator messages the bot");
   881→  })();
   882→}

   884→/** Learn the operator chat from the bot's update stream — the operator sends
   885→ * ANY message to the bot in Telegram, we read getUpdates and persist the chat. */
   886→export async function discoverChat(): Promise<{ ok: boolean; error?: string; chatId?: string; from?: string; text?: string }> {
   887→  if (!TOKEN) return { ok: false, error: "TELEGRAM_BOT_TOKEN unset — relay dormant" };
   888→  await ensureBotIdentity();
   889→  const r = await apiCall<Array<{ message?: { chat: { id: number; first_name?: string; username?: string }; text?: string } }>>("getUpdates", { limit: 10, timeout: 0 }, 8000);
   890→  if (!r.ok || !r.result) return { ok: false, error: r.error ?? "getUpdates returned no payload" };
   891→  const withMsg = r.result.filter((u) => u.message?.chat?.id !== undefined);
   892→  if (withMsg.length === 0) return { ok: false, error: "no messages waiting — open the bot in Telegram, press START and send any message (e.g. «سلام»), then retry discover" };
   893→  const last = withMsg[withMsg.length - 1].message!;
   894→  const chatId = String(last.chat.id);
   895→  chatOverride = chatId;
   896→  overrideLoaded = true;
   897→  await db.engineSetting.upsert({ where: { key: CHAT_OVERRIDE_KEY }, create: { key: CHAT_OVERRIDE_KEY, value: chatId }, update: { value: chatId } });
   898→  console.log(`[NOTIFY] telegram chat DISCOVERED — id ${chatId} (${last.chat.first_name ?? last.chat.username ?? "unknown"}), persisted to ${CHAT_OVERRIDE_KEY}; relay fully ARMED`);
   899→  return { ok: true, chatId, from: last.chat.first_name ?? last.chat.username, text: last.text };
   900→}

   412→// ── weight table (Binance USDT-M futures documented weights) ────────────────
   413→// klines ~2 (limit≤100) · depth 2-5-10 by limit · aggTrades 20 · trades 20 ·
   414→// ticker/24hr 1 per-symbol / 40 full-universe · premiumIndex 1 per-symbol /
   415→// 10 all-markers · futures/data/* 1 (openInterestHist, L/S ratios, basis…) ·
   416→// exchangeInfo 1 · ping/time 1 · unknown default 5 (conservative).
   417→function resolveWeight(path: string): number {
   418→  const qIdx = path.indexOf("?");
   419→  const ep = (qIdx === -1 ? path : path.slice(0, qIdx)).replace(/^\//, "").replace(/\/+$/, "");
   420→  const query = qIdx === -1 ? "" : path.slice(qIdx + 1);
   421→  const limit = (): number | null => {
   422→    const m = /[?&]limit=(\d+)/.exec(query);
   423→    return m ? Number(m[1]) : null;
   424→  };
   425→  switch (ep) {
   426→    case "fapi/v1/ping":
   427→    case "fapi/v1/time":
   428→    case "fapi/v1/exchangeInfo":
   429→      return 1;
   430→    case "fapi/v1/klines": {
   431→      const l = limit() ?? 500;
   432→      return l <= 100 ? 2 : l <= 499 ? 5 : 10;
   433→    }
   434→    case "fapi/v1/depth": {
   435→      const l = limit() ?? 100;
   436→      return l <= 100 ? 2 : l <= 500 ? 5 : 10;
   437→    }
   438→    case "fapi/v1/aggTrades":
   439→    case "fapi/v1/trades":
   440→      return 20;
   441→    case "fapi/v1/ticker/24hr":
   442→      return /[?&]symbol=/.test(query) ? 1 : 40;
   443→    case "fapi/v1/premiumIndex":
   444→      return /[?&]symbol=/.test(query) ? 1 : 10;
   445→    default:
   446→      if (ep.startsWith("futures/data/")) return 1; // openInterestHist / L-S ratios / basis family
   447→      return 5; // unknown → conservative default (covers fapi/v1/fundingRate etc.)
   448→  }
   449→}

   451→function pruneWindow(now: number): void {
   452→  const cutoff = now - WINDOW_MS;
   453→  while (state.spend.length > 0 && state.spend[0]!.at <= cutoff) {
   454→    const e = state.spend.shift()!;
   455→    state.windowSum -= e.w;
   456→  }
   457→  if (state.spend.length > SPEND_CAP) {
   458→    // defensive: drop the oldest half rather than grow unbounded
   459→    const keep = state.spend.slice(Math.floor(state.spend.length / 2));
   460→    let sum = 0;
   461→    for (const e of keep) sum += e.w;
   462→    state.spend = keep;
   463→    state.windowSum = sum;
   464→  }
   465→}

   467→/** Reserve `weight` from the rolling window. Returns the spend entry (for refunds) or null. */
   468→// ── v14.2 BAN-AWARE RECOVERY PACER ────────────────────────────────────────
   469→// The ONLY budget the rolling window spends against. A flat budget after a
   470→// ban ended = instant re-saturation stampede (the 10-19 spiral); the pacer
   471→// gives the pool a 10min ramp and caps telemetry while the pool shows live
   472→// distress. Global-window minutes are NOT handled here (gateFetch drops
   473→// everything while the window is live) — the pacer shapes the BEFORE
   474→// (distress) and AFTER (recovery) of each ban episode.
   475→function namedLiveBanCount(now: number): number {
   476→  let n = 0;
   477→  for (const [, until] of state.ipBans) if (until > now) n++;
   478→  return n;
   479→}

   481→export function effectiveBudgetPer60s(): number {
   482→  const now = Date.now();
   483→  let factor = 1;
   484→  // POOL DISTRESS — ≥2 named egress IPs inside live ban windows: cap telemetry.
   485→  if (namedLiveBanCount(now) >= 2) factor = Math.min(factor, PACER_DISTRESS_CAP);
   486→  // RECOVERY RAMP — 10min linear 25%→100% after the last global window ends.
   487→  if (state.banUntil > 0) {
   488→    const sinceBan = now - state.banUntil;
   489→    if (sinceBan >= 0 && sinceBan < PACER_RECOVERY_MS) {
   490→      factor = Math.min(factor, PACER_RAMP_FLOOR + (1 - PACER_RAMP_FLOOR) * (sinceBan / PACER_RECOVERY_MS));
   491→    }
   492→  }
   493→  return Math.max(PACER_MIN_BUDGET, Math.round(state.budgetPer60s * factor));
   494→}

   496→function takeWeight(weight: number): SpendEntry | null {
   497→  const now = Date.now();
   498→  pruneWindow(now);
   499→  if (state.windowSum + weight > effectiveBudgetPer60s()) return null;
   500→  const entry: SpendEntry = { at: now, w: weight };
   501→  state.spend.push(entry);
   502→  state.windowSum += weight;
   503→  return entry;
   504→}

   506→/** 418/429 responses carry no weight cost — give the reservation back. */
   507→function refundWeight(entry: SpendEntry | null): void {
   508→  if (!entry) return;
   509→  const idx = state.spend.indexOf(entry);
   510→  if (idx === -1) return; // already pruned out of the window (its cost is gone too)
   511→  state.spend.splice(idx, 1);
   512→  state.windowSum -= entry.w;
   513→}

   515→function armGlobalBan(until: number): void {
   516→  if (until > state.banUntil) {
   517→    state.banUntil = until;
   518→    state.stats.lastBanUntil = until;
   519→  }
   520→}

   522→/** Record a 418/429 body. Returns { ip, until } when the body NAMES the banned egress IP. */
   523→function noteBanBody(status: number, body: string): { ip: string; until: number } | null {
   524→  const now = Date.now();
   525→  const named = /IP\(([\d.]+)\)\s*banned until (\d+)/i.exec(body);
   526→  if (named) {
   527→    const ip = named[1]!;
   528→    // v12.1 TASK-49: the LEDGER carries Binance's REAL end (sanity-capped 24h) —
   529→    // the cosmetic +35min cap only applies to the global silence window below.
   530→    // Capping the ledger made the early probe fire at a fake expiry and the
   531→    // resulting 418 contact re-escalated the real ban (11:58→12:34→13:09 spiral).
   532→    const realUntil = Math.min(Number(named[2]), now + 24 * 60_000);
   533→    const until = Math.min(realUntil, now + BAN_CAP_MS);
   534→    state.ipBans.set(ip, realUntil);
   535→    state.lastBannedIp = ip;
   536→    persistBans(); // v12.0 — named-IP ban ledger survives restarts (30s debounce)
   537→    if (state.ipBans.size > IP_BAN_MAP_CAP) {
   538→      // evict the soonest-expiring record
   539→      let oldestKey: string | null = null;
   540→      let oldestVal = Infinity;
   541→      for (const [k, v] of state.ipBans) if (v < oldestVal) { oldestVal = v; oldestKey = k; }
   542→      if (oldestKey) state.ipBans.delete(oldestKey);
   543→    }
   544→    state.recentBanEvents.push({ at: now, ip });
   545→    if (state.recentBanEvents.length > 64) state.recentBanEvents.splice(0, state.recentBanEvents.length - 64);
   546→    return { ip, until };
   547→  }
   548→  const untilMatch = /banned until (\d+)/i.exec(body);
   549→  if (untilMatch) {
   550→    armGlobalBan(Math.min(Number(untilMatch[1]), now + BAN_CAP_MS)); // unknown scope — assume it's us
   551→    return null;
   552→  }
   553→  // unparseable body — conservative defaults (taxonomy precedent for 418)
   554→  armGlobalBan(now + (status === 429 ? BAN_429_DEFAULT_MS : BAN_418_DEFAULT_MS));
   555→  return null;
   556→}

   558→/** Every retry landed on a (named) banned IP — escalate the global silence window. */
   559→function escalatePoolBlock(worstNamedUntil: number): void {
   560→  const now = Date.now();
   561→  state.allBanStreak++;
   562→  if (state.allBanStreak >= POOL_STREAK_FOR_LONG_BLOCK) {
   563→    // repeated independent evidence — the egress pool is saturated: full
   564→    // zero-network window, capped at 35min, then a fresh probe self-heals.
   565→    armGlobalBan(Math.min(worstNamedUntil, now + BAN_CAP_MS));
   566→    state.allBanStreak = 0;
   567→  } else {
   568→    armGlobalBan(now + POOL_BLOCK_SHORT_MS);
   569→  }
   570→}

   582→/**
   583→ * THE gate body. Resolves parsed JSON, or null on: live IP ban, weight-budget
   584→ * shed, HTTP failure, timeout, or parse error. Never throws, never queues.
   585→ */
   586→async function executeGated<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   587→  const lane: FapiGatePriority = opts?.priority ?? "telemetry";
   588→  const weight = Math.max(1, Math.round(opts?.weight ?? resolveWeight(path)));
   589→  const timeoutMs = opts?.timeoutMs ?? DEFAULT_TIMEOUT_MS;
   590→
   591→  let entry = takeWeight(weight);
   592→  if (!entry) {
   593→    if (lane !== "critical") {
   594→      state.stats.droppedBudget++;
   595→      return null; // telemetry: drop instead of queueing a storm
   596→    }
   597→    await delay(CRITICAL_RETRY_MS); // one small negative retry
   598→    entry = takeWeight(weight);
   599→    if (!entry) {
   600→      state.stats.droppedBudget++;
   601→      return null;
   602→    }
   603→  }
   604→
   605→  const host = opts?.host ?? FAPI_HOST;
   606→  let attempt = 0;
   607→
   608→  while (true) {
   609→    const ctrl = new AbortController();
   610→    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   611→    let res: Response;
   612→    try {
   613→      // v12.6.1 CONNECTION-RECYCLE (measured 2026-10-20 19:41Z, bun runtime):
   614→      // a keep-alive fetch POOLS one TCP flow ⇒ ONE egress IP for its whole
   615→      // life — a pinned-banned connection turned 231 gate calls into 0 allowed
   616→      // while 6/10 fresh connections succeeded. `Connection: close` = a fresh
   617→      // NAT lottery draw per request; the TLS cost (~60ms) is trivial next to
   618→      // the starvation it prevents.
   619→      res = await fetch(`${host}${path}`, { signal: ctrl.signal, cache: "no-store", headers: { Connection: "close" } });
   620→    } catch {
   621→      return null; // timeout/abort/network — weight may have been consumed, keep the reservation
   622→    } finally {
   623→      clearTimeout(timer);
   624→    }
   625→
   626→    if (res.ok) {
   627→      const text = await res.text().catch(() => "");
   628→      if (!text) return null; // empty 200 — nothing to invent
   629→      state.stats.allowed++;
   630→      state.allBanStreak = 0; // a clean 2xx proves the pool still serves us
   631→      try {
   632→        return JSON.parse(text) as T;
   633→      } catch {
   634→        return null;
   635→      }
   636→    }
   637→
   638→    if (res.status === 418 || res.status === 429) {
   639→      const body = await res.text().catch(() => "");
   640→      refundWeight(entry); // banned responses carry no weight cost
   641→      entry = null;
   642→      const named = noteBanBody(res.status, body);
   643→      if (opts?.singleProbe) return null; // v12.0 early probe — single attempt, window stays armed
   644→      if (!named) return null; // unknown-scope ban — global zero-network window already armed
   645→      if (attempt >= BAN_RETRIES) {
   646→        escalatePoolBlock(named.until);
   647→        return null;
   648→      }
   649→      attempt++;
   650→      state.stats.retriesAfterBan++;
   651→      await delay(BAN_RETRY_DELAYS_MS[Math.min(attempt - 1, BAN_RETRY_DELAYS_MS.length - 1)]!); // span the NAT rotation window
   652→      entry = takeWeight(weight);
   653→      if (!entry) {
   654→        state.stats.droppedBudget++;
   655→        return null;
   656→      }
   657→      continue;
   658→    }
   659→
   660→    return null; // other non-ok status — honest null, no retry
   661→  }
   662→}

   670→// v12.1 TASK-49 — SHARED LEDGER REFRESH: trade-service / price-ws write their
   671→// own 418 discoveries (with REAL ends) to the same bans.json. Merge live disk
   672→// entries into state every 30s (extend-only). v12.6 scope split: UNKNOWN-SCOPE
   673→// entries extend the global silence (pool-suspect evidence); NAMED-IP entries
   674→// merge into ipBans as intel only — a majority-clean pool must not be
   675→// blacked out because one pool member drew a ban.
   676→let lastLedgerCheckAt = 0;
   677→function refreshLedgerOnce(): void {
   678→  const nowMs = Date.now();
   679→  lastLedgerCheckAt = nowMs;
   680→  try {
   681→    const raw = JSON.parse(readFileSync(BAN_PERSIST_PATH, "utf8")) as { ipBans?: Record<string, number> } | null;
   682→    let unknownMax = 0;
   683→    for (const [key, until] of Object.entries(raw?.ipBans ?? {})) {
   684→      if (typeof until !== "number" || until <= nowMs) continue;
   685→      if (isUnknownScopeKey(key)) {
   686→        if (until > unknownMax) unknownMax = until;
   687→        continue;
   688→      }
   689→      if (until > (state.ipBans.get(key) ?? 0)) state.ipBans.set(key, until);
   690→    }
   691→    // unknown-scope knowledge extends the global silence window (extend-only)
   692→    if (unknownMax > state.banUntil) {
   693→      state.banUntil = unknownMax;
   694→      console.log(`[fapi-gate] v12.6 ledger refresh — UNKNOWN-SCOPE pool ban extends silence until ${new Date(unknownMax).toISOString()}`);
   695→    }
   696→  } catch {
   697→    /* ledger gone/unreadable — keep in-memory state */
   698→  }
   699→}
   700→try {
   701→  const refresh = setInterval(refreshLedgerOnce, 30_000);
   702→  if (typeof refresh.unref === "function") refresh.unref();
   703→} catch {
   704→  /* non-blocking env — skip refresh */
   705→}

   707→export async function gateFetch<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   708→  state.stats.attempted++;
   709→  const now = Date.now();
   710→
   711→  // v12.1 TASK-49 — LAZY LEDGER GUARD: the instant our global window expires,
   712→  // re-check the shared ledger (throttled to one disk read / 15s) BEFORE any
   713→  // network call — closes the poke-gap where the window died but the refresh
   714→  // interval hadn't fired yet. While the pool is ledger-banned, zero pokes.
   715→  if (now >= state.banUntil && now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   716→  if (now < state.banUntil) state.stats.droppedBan++; // ledger re-armed the window — shed instantly
   717→
   718→  // GLOBAL BAN AWARENESS — while a global ban window is live, zero network calls
   719→  // …except the v12.0 ROTATION-AWARE EARLY PROBE: one critical-lane attempt per
   720→  // PROBE_COOLDOWN_MS when the sampled egress IP is clean — the pool rotates,
   721→  // so the window can be LIFTED the moment a clean draw appears instead of
   722→  // waiting out the full 35min cap. A failed probe keeps the window armed.
   723→  if (now < state.banUntil) {
   724→    state.stats.droppedBan++;
   725→    const canProbe = (opts?.priority ?? "telemetry") === "critical" && now - state.lastProbeAt >= PROBE_COOLDOWN_MS;
   726→    if (!canProbe) return null;
   727→    // v12.1 TASK-49 — LEDGER FRESHNESS BEFORE PROBE: state may hold a cosmetic
   728→    // cap while the disk carries a longer REAL end another process discovered.
   729→    // A probe fired on stale state pokes a still-banned pool and re-escalates.
   730→    // Re-read the ledger (15s throttle) and re-arm BEFORE the probe decision.
   731→    if (now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   732→    if (now < state.banUntil) return null; // ledger extended the silence — no probe
   733→    state.lastProbeAt = now;
   734→    const ip = await sampleEgressIp();
   735→    // v12.1 TASK-49: expiry-aware check — a recorded ban whose real end passed
   736→    // no longer blocks the probe (the old .has() kept ghost entries armed).
   737→    if (!ip || isIpBanned(ip, now)) return null; // sampled draw is banned/unknown — stay silent
   738→    state.stats.probesIssued++;
   739→    const probeResult = await executeGated<T>(path, { ...opts, singleProbe: true });
   740→    if (probeResult !== null) {
   741→      state.banUntil = 0; // clean draw proved the pool rotated — lift instantly
   742→      state.allBanStreak = 0;
   743→      state.stats.probeLifts++;
   744→      console.log(`[fapi-gate] v12.0 ban window LIFTED early — egress rotated to clean IP ${ip}`);
   745→    }
   746→    return probeResult;
   747→  }
   748→
   749→  // SINGLE-FLIGHT per (host,path): share the request while it is in flight and
   750→  // for a 2s result window — concurrent module askers pay the weight once.
   751→  const key = `${opts?.host ?? FAPI_HOST}${path}`;
   752→  const existing = state.inflight.get(key);
   753→  if (existing) {
   754→    const age = now - existing.startedAt;
   755→    if (!existing.settled || age < SF_RESULT_MS) {
   756→      state.stats.singleFlightShared++;
   757→      return existing.promise as Promise<T | null>;
   758→    }
   759→    state.inflight.delete(key);
   760→  }
   761→  if (state.inflight.size >= SF_MAP_CAP) pruneInflight(now);
   762→
   763→  const promise = executeGated<T>(path, opts);
   764→  const entry = { startedAt: now, settled: false, promise };
   765→  state.inflight.set(key, entry);
   766→  void promise
   767→    .catch(() => null) // executeGated never rejects — belt & suspenders
   768→    .finally(() => {
   769→      entry.settled = true;
   770→    });
   771→  return promise;
   772→}

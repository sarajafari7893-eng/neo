   115→function ago(iso: string | null): string {
   116→  if (!iso) return "—";
   117→  const s = Math.max(0, Math.round((Date.now() - new Date(iso).getTime()) / 1000));
   118→  if (s < 60) return `${s}s`;
   119→  if (s < 3600) return `${Math.round(s / 60)}m`;
   120→  if (s < 86400) return `${Math.round(s / 3600)}h`;
   121→  return `${Math.round(s / 86400)}d`;
   122→}
   123→
   124→function etaClock(iso: string | null): string {
   125→  if (!iso) return "—";
   126→  try {
   127→    return new Date(iso).toLocaleTimeString("en-GB", { hour: "2-digit", minute: "2-digit" });
   128→  } catch {
   129→    return "—";
   130→  }
   131→}

   133→/** Inline SVG book-growth sparkline (settled EPOCH-2 over the journal window). */
   134→function GrowthSpark({ series }: { series: Array<{ t: string; v: number }> }) {
   135→  if (series.length < 2) {
   136→    return (
   137→      <div className="flex h-9 items-center justify-center font-terminal text-[9px] text-ink-3">
   138→        BOOK CURVE WARMING — journal fills every 10min
   139→      </div>
   140→    );
   141→  }
   142→  const W = 100;
   143→  const H = 26;
   144→  // NaN guard (v9.4.2): journal rows can carry null/gap equity values — any
   145→  // non-finite point would turn vMin/vMax into NaN and crash React children.
   146→  const vs = series.map((p) => p.v).filter((v) => Number.isFinite(v));
   147→  if (vs.length < 2) {
   148→    return (
   149→      <div className="flex h-9 items-center justify-center font-terminal text-[9px] text-ink-3">
   150→        BOOK CURVE WARMING — journal fills every 10min
   151→      </div>
   152→    );
   153→  }
   154→  const vMin = Math.min(...vs);
   155→  const vMax = Math.max(...vs);
   156→  const span = Math.max(1, vMax - vMin);
   157→  const pts = series
   158→    .map((p, i) => ({ x: (i / (series.length - 1)) * W, v: p.v }))
   159→    .filter((p): p is { x: number; v: number } => Number.isFinite(p.v))
   160→    .map((p) => {
   161→      const y = H - ((p.v - vMin) / span) * (H - 4) - 2;
   162→      return `${p.x.toFixed(1)},${y.toFixed(1)}`;
   163→    });
   164→  const area = `M0,${H} L${pts.join(" L")} L${W},${H} Z`;
   165→  return (
   166→    <div className="relative">
   167→      <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="h-9 w-full" aria-hidden>
   168→        <path d={area} fill="url(#rawcalSparkFill)" opacity={0.35} />
   169→        <polyline points={pts.join(" ")} fill="none" stroke="var(--amber-neon)" strokeWidth={1.4} vectorEffect="non-scaling-stroke" strokeLinejoin="round" />
   170→        <defs>
   171→          <linearGradient id="rawcalSparkFill" x1="0" y1="0" x2="0" y2="1">
   172→            <stop offset="0%" stopColor="var(--amber-neon)" stopOpacity={0.8} />
   173→            <stop offset="100%" stopColor="var(--amber-neon)" stopOpacity={0} />
   174→          </linearGradient>
   175→        </defs>
   176→      </svg>
   177→      <span className="absolute left-0 top-0 font-terminal text-[8px] text-ink-3">{vMin.toFixed(2)}</span>
   178→      <span className="absolute right-0 top-0 font-terminal text-[8px] text-amber-neon">{vMax.toFixed(2)}</span>
   179→    </div>
   180→  );
   181→}

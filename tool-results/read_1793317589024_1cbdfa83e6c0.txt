   647→    audit("EXEC-RISK-CAP", { serial: it.serial, baseNotional: Number(cap.base.toFixed(4)), cappedNotional: Number(notional.toFixed(4)), slDistPct: Number((slDistPct * 100).toFixed(3)), equity: Number((walletUsdt ?? balanceUsdt ?? 0).toFixed(4)), maxRiskPct: S.cfg.maxRiskPct });
   648→  }
   649→  it.expectedNotional = Number(notional.toFixed(6));
   650→  let qty = notional / (refPrice * mult);
   651→
   652→  // v1.3.0 ENTRY GATES (operator mandate trace 1a15f7d945d27ead):
   653→  // ① ENTRY-DELAY — فاصله زمانی با تأخیر: a card whose entry window has passed
   654→  //    is no longer the signal the card described; chasing it trades a stale edge.
   655→  const ageAtEntry = Date.now() - (it.createdAt0 ?? Date.now());
   656→  it.entryAgeMs = ageAtEntry;
   657→  if (ageAtEntry > S.cfg.maxEntryAgeMs) {
   658→    it.status = "SKIP";
   659→    it.events.push({ at: Date.now(), ev: "ENTRY-DELAY", note: `age=${Math.round(ageAtEntry / 1000)}s > cap ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s` });
   660→    countSkip("ENTRY-DELAY");
   661→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-delay: card age ${Math.round(ageAtEntry / 1000)}s exceeds ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s window` });
   662→    return;
   663→  }
   664→  // ② ENTRY-DRIFT — اختلاف قیمتی بالا: if the book has already run away from
   665→  //    the card's entry, the fill is not the card's trade — skip honestly.
   666→  const driftBps = Math.abs(refPrice - it.cardEntry) / it.cardEntry * 10_000;
   667→  it.entryDriftBps = Number(driftBps.toFixed(2));
   668→  if (it.cardEntry > 0 && driftBps > S.cfg.maxEntryDriftBps) {
   669→    it.status = "SKIP";
   670→    it.events.push({ at: Date.now(), ev: "ENTRY-DRIFT", note: `drift=${driftBps.toFixed(0)}bps > cap ${S.cfg.maxEntryDriftBps}bps (ref=${refPrice} vs card=${it.cardEntry})` });
   671→    countSkip("ENTRY-DRIFT");
   672→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-drift: ${driftBps.toFixed(0)}bps from card entry exceeds ${S.cfg.maxEntryDriftBps}bps cap` });
   673→    return;
   674→  }
   675→  // ③ BALANCE-CAP (v1.4.3) — the fixed margin must fit the LIVE balance; when
   676→  // the account cannot fund another isolated position, skip honestly instead of
   677→  // throwing an insufficient-margin order at the book.
   678→  if (liveMode) {
   679→    await refreshBalance();
   680→    const avail = balanceUsdt ?? 0;
   681→    if (avail < effMargin) {
   682→      it.status = "SKIP";
   683→      it.events.push({ at: Date.now(), ev: "BALANCE-CAP", note: `margin ${effMargin.toFixed(2)}$ > available ${avail.toFixed(2)}$` });
   684→      countSkip("BALANCE-CAP");
   685→      audit("EXEC-SKIP", { serial: it.serial, why: `balance-cap: margin ${effMargin.toFixed(2)} USDT exceeds available ${avail.toFixed(2)} USDT` });
   686→      return;
   687→    }
   688→    // v1.4.4 — a risk-capped margin below $1 is exchange dust; skip honestly.
   689→    if (cap.capped && effMargin < 1) {
   690→      it.status = "SKIP";
   691→      it.events.push({ at: Date.now(), ev: "RISK-CAP-DUST", note: `margin ${effMargin.toFixed(2)}$ < $1 after risk cap` });
   692→      countSkip("RISK-CAP");
   693→      audit("EXEC-SKIP", { serial: it.serial, why: `risk-cap dust: margin ${effMargin.toFixed(2)} USDT below $1 floor` });
   694→      return;
   695→    }
   696→  }
   697→  // $5-fixed mandate: a symbol whose exchange floor (minNotional / minQty×price)
   698→  // cannot fit inside the mandated margin notional is SKIPPED honestly —
   699→  // silently upsizing would breach the fixed-margin contract.
   700→  // v1.4.3 — floors are REAL notional (multiplier-aware).
   701→  const floorNotional = Math.max(instr.minNotional, instr.minQty * refPrice * mult, S.floorLearn?.[it.tbSymbol] ?? 0);
   702→  if (floorNotional > notional) {
   703→    it.status = "SKIP";
   704→    it.events.push({ at: Date.now(), ev: "MIN_FLOOR_SKIP", note: `${it.tbSymbol} floor=${floorNotional.toFixed(2)} > marginNotional=${notional}${S.floorLearn?.[it.tbSymbol] ? " (learned)" : ""} ` });
   705→    countSkip("MIN-FLOOR");
   706→    audit("EXEC-SKIP", { serial: it.serial, why: `exchange floor ${floorNotional.toFixed(2)} USDT exceeds fixed-margin notional ${notional} USDT` });
   707→    return;
   708→  }
   709→  qty = floorStep(Math.min(qty, instr.maxQty), instr.stepSize);
   710→  if (qty < instr.minQty) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "QTY_BELOW_MIN" }); audit("EXEC-FAIL", { serial: it.serial, why: `qty ${qty} < minQty ${instr.minQty}` }); return; }
   711→
   712→  const tSent = Date.now();
   713→  let fillPrice: number | null = null;
   714→  let orderId: string | null = null;
   715→  if (liveMode) {
   716→    await ensureLeverage(it.tbSymbol, it.leverage);
   717→    const side = it.direction === "LONG" ? "BUY_OPEN" : "SELL_OPEN";
   718→    const r = await placeOrder(it.tbSymbol, side, qty, { takeProfit: it.tp3, stopLoss: it.stopLoss, clientOrderId: it.clientOrderId });
   719→    audit("EXEC-OPEN", { serial: it.serial, symbol: it.tbSymbol, side, qty, lev: it.leverage, tp3: it.tp3, sl: it.stopLoss, ok: r.ok, orderId: r.orderId, err: r.err });
   720→    if (!r.ok) {
   721→      // v1.4.2 — server-side real floor: exchangeInfo hides it (minNotional=0)
   722→      // but the book rejects with -1199/-1202 "quantity too small". Teach the
   723→      // floor, reclassify as an honest MIN-FLOOR SKIP, keep the ledger truthful.
   724→      if (/quantity too small/i.test(r.err ?? "")) {
   725→        const prevFloor = S.floorLearn?.[it.tbSymbol] ?? 0;
   726→        S.floorLearn[it.tbSymbol] = Math.max(prevFloor, notional);
   727→        it.status = "SKIP";
   728→        it.events.push({ at: Date.now(), ev: "MIN_FLOOR_LEARNED", note: `${it.tbSymbol} real floor ≥ ${notional} USDT (server -1199/-1202; exchangeInfo said ${floorNotional.toFixed(2)})` });
   729→        countSkip("MIN-FLOOR");
   730→        audit("EXEC-SKIP", { serial: it.serial, why: `server floor learned: ${it.tbSymbol} ≥ ${notional} USDT notional (rejected quantity-too-small; exchangeInfo floor ${floorNotional.toFixed(2)})` });
   731→        persistState();
   732→        return;
   733→      }
   734→      it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "OPEN_REJECTED", note: (r.err ?? "").slice(0, 160) }); return;
   735→    }
   736→    orderId = r.orderId;
   737→    fillPrice = await resolveFill(it.tbSymbol, orderId, refPrice);
   738→  } else {
   739→    fillPrice = refPrice; // paper fill on the real toobit book (taker side)
   740→  }
   741→  const tFill = Date.now();
   742→  it.status = "OPEN";
   743→  it.qtyOrig = qty;
   744→  it.qtyRemaining = qty;
   745→  it.orderId = orderId;
   746→  it.fillPrice = fillPrice;
   747→  it.openedAt = tFill;
   748→  // v1.3.0 doctrine bookkeeping — the truth of what the book actually saw:
   749→  // step-size floor rounding makes actualNotional ≤ expectedNotional by design.
   750→  // v1.4.3 — actual notional is REAL: qty(contracts) × fill × contractMultiplier.
   751→  it.actualNotional = fillPrice ? Number((qty * fillPrice * mult).toFixed(6)) : null;
   752→  it.protected = it.mode === "LIVE" ? true : true; // LIVE: attached TP/SL in same order + recon re-check; PAPER: sim active
   753→  S.doctrine.opens++;
   754→  const ndelta = it.expectedNotional && it.actualNotional ? Math.abs(it.actualNotional - it.expectedNotional) / it.expectedNotional * 10_000 : 0;
   755→  audit("EXEC-OPEN-DOCTRINE", { serial: it.serial, lev: it.leverage, levSrc: it.levSrc ?? "card", margin: Number(effMargin.toFixed(4)), riskCapped: cap.capped, contractMult: mult, expectedNotional: it.expectedNotional, actualNotional: it.actualNotional, notionalDeltaBps: Number(ndelta.toFixed(1)), entryAgeMs: it.entryAgeMs, entryDriftBps: it.entryDriftBps });
   756→  if (ndelta > 2000) doctrineViolation(it.serial, "NOTIONAL-DELTA", `exp=${it.expectedNotional} act=${it.actualNotional} (${ndelta.toFixed(0)}bps) — step-rounding or minQty floor`);
   757→  it.lat = {
   758→    detectMs: it.lat.detectMs ?? null,
   759→    execMs: tFill - tSent,
   760→    totalMs: it.createdAt0 !== undefined ? tFill - it.createdAt0 : null,
   761→    slipBps: fillPrice && it.cardEntry > 0 ? ((it.direction === "LONG" ? fillPrice - it.cardEntry : it.cardEntry - fillPrice) / it.cardEntry) * 10_000 : null,
   762→  };
   763→  it.events.push({ at: tFill, ev: "OPEN", note: `${it.mode} qty=${qty} mult=×${mult} fill=${fillPrice}` });
   764→  S.metrics.push({ serial: it.serial, symbol: it.symbol, direction: it.direction, mode: it.mode, detectMs: it.lat.detectMs, execMs: it.lat.execMs, totalMs: it.lat.totalMs, slipBps: it.lat.slipBps, fillPrice, cardEntry: it.cardEntry, at: tFill });
   765→  if (S.metrics.length > METRICS_WINDOW) S.metrics = S.metrics.slice(-METRICS_WINDOW);
   766→}

   768→// resolve the real average fill price of a market order (best effort)
   769→async function resolveFill(tbSymbol: string, orderId: string | null, fallback: number): Promise<number> {
   770→  if (orderId) {
   771→    for (let i = 0; i < 3; i++) {
   772→      await new Promise((r) => setTimeout(r, 350));
   773→      const r2 = await signed<{ status?: string; orderStatus?: string; avgPrice?: string; price?: string; executedQty?: string; cumQty?: string }>("GET", "/api/v1/futures/order", { orderId });
   774→      const d = r2.data as Record<string, unknown> | null;
   775→      const st = String(d?.status ?? d?.orderStatus ?? "").toUpperCase();
   776→      const avg = Number(d?.avgPrice ?? d?.price ?? 0);
   777→      if (avg > 0) return avg;
   778→      if (/CANCELED|REJECTED|EXPIRED/.test(st)) break;
   779→      if (st === "FILLED" || i === 2) break;
   780→    }
   781→  }
   782→  return fallback;
   783→}

   785→// v1.4.3 LIVE PnL TELEMETRY — exchange-verified realized PnL for an intent's
   786→// lifetime (fills since openedAt; realizedPnL arrives on closing fills).
   787→// This is the money-truth behind «چرا هیچ سیگنالی در سود بسته نشده».
   788→async function fetchRealizedPnl(it: Intent): Promise<number | null> {
   789→  try {
   790→    // v1.4.5 — a NEVER-EXECUTED intent (no openedAt) owns ZERO exchange fills;
   791→    // with since=0 the symbol's whole fill history was once re-summed into it
   792→    // (phantom PnL: ENA -0.7445 = morning -0.6893-0.0552, BTW -3.2605, PEPE -0.1066).
   793→    if (!it.openedAt) return null;
   794→    const r = await signed<Array<{ time?: string; realizedPnl?: string; commission?: string; makerRebate?: string }>>("GET", "/api/v1/futures/userTrades", { symbol: it.tbSymbol, limit: 100 });
   795→    const rows = Array.isArray(r.data) ? r.data : [];
   796→    const since = it.openedAt;
   797→    let pnl = 0, seen = false;
   798→    for (const f of rows) {
   799→      if (Number(f.time ?? 0) < since) continue;
   800→      seen = true;
   801→      pnl += Number(f.realizedPnl ?? 0) - Number(f.commission ?? 0) + Number(f.makerRebate ?? 0);
   802→    }
   803→    if (!seen) return null;
   804→    return Number(pnl.toFixed(4));
   805→  } catch { return null; }
   806→}
   807→// revert the card-status transition so the next poll re-fires the failed
   808→// action (v1.4.3 — transitions are no longer one-shot on transient rejections;
   809→// capped so a permanently-rejected action cannot loop forever).
   810→function revertTransition(it: Intent, what: string, err: string): void {
   811→  const attempts = (it.actionAttempts ?? 0) + 1;
   812→  it.actionAttempts = attempts;
   813→  if (attempts <= 20 && it.prevStatusSeen) {

   816→    audit("EXEC-RETRY-ARMED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   817→  } else {
   818→    it.events.push({ at: Date.now(), ev: `${what}_ABANDONED`, note: `attempts=${attempts} — will not retry: ${err.slice(0, 100)}` });
   819→    audit("EXEC-RETRY-ABANDONED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   820→  }
   821→  persistState();
   822→}

   824→async function doBank(it: Intent, fracOfOrig: number, reason: string): Promise<void> {
   825→  if (it.status !== "OPEN" || it.qtyRemaining <= 0) return;
   826→  let qty = floorStep(Math.min(it.qtyOrig * fracOfOrig, it.qtyRemaining), instruments.get(it.tbSymbol)?.stepSize ?? 0);
   827→  if (qty <= 0) return;
   828→  if (qty >= it.qtyRemaining) return doClose(it, reason);
   829→  if (it.mode === "LIVE") {
   830→    const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   831→    // v1.4.3 — clientOrderId is mandatory on Toobit (pre-1.4.3 every bank was
   832→    // rejected -1004 → no signal ever closed in profit).
   833→    const r = await placeOrder(it.tbSymbol, side, qty, { clientOrderId: `NX${it.serial}${reason}${Date.now() % 100000}` });
   834→    audit("EXEC-BANK", { serial: it.serial, reason, qty, ok: r.ok, err: r.err });
   835→    if (!r.ok) { it.events.push({ at: Date.now(), ev: "BANK_REJECTED", note: (r.err ?? "").slice(0, 160) }); revertTransition(it, "BANK", r.err ?? "bank order rejected"); return; }
   836→  } else {
   837→    // v1.4.0 — PAPER سیو سود was invisible in audit.jsonl (only in state.json);
   838→    // the paper fill rides the REAL toobit book so it deserves the same trail.
   839→    // v1.5.0 — paper money truth: the banked slice is priced at the LIVE taker
   840→    // side and books REAL PnL into the intent + fixed-$3 account.
   841→    const tick = await tbTicker(it.tbSymbol);
   842→    const exit = tick ? (it.direction === "LONG" ? tick.bid : tick.ask) : null;
   843→    const entry = it.fillPrice ?? it.cardEntry;
   844→    const sign = it.direction === "LONG" ? 1 : -1;
   845→    const mult = it.contractMult ?? instruments.get(it.tbSymbol)?.mult ?? 1;
   846→    const pnl = exit !== null && entry > 0 ? (exit - entry) * qty * sign * mult : 0;
   847→    it.realizedPnlUsdt = Number(((it.realizedPnlUsdt ?? 0) + pnl).toFixed(6));
   848→    audit("EXEC-BANK-PAPER", { serial: it.serial, reason, qty, exit: exit ?? null, pnl: Number(pnl.toFixed(6)), note: "paper fill on real book (v1.5.0 priced)" });
   849→    if (pnl !== 0) paperBook(it, pnl, `${reason} bank`);
   850→  }
   851→  it.qtyRemaining = Number((it.qtyRemaining - qty).toFixed(10));
   852→  if (reason === "TP1") it.qtyBankedTp1 = qty;
   853→  if (reason === "TP2") it.qtyBankedTp2 = qty;
   854→  it.actionAttempts = 0;
   855→  it.events.push({ at: Date.now(), ev: reason, note: `banked ${qty} (سیو سود)` });
   856→}

   858→async function doClose(it: Intent, reason: string): Promise<void> {
   859→  if (it.status === "CLOSED" || it.status === "CLOSING") return;
   860→  it.status = "CLOSING";
   861→  if (it.mode === "LIVE" && it.qtyRemaining > 0) {
   862→    const flash = await flashClose(it.tbSymbol, it.direction);
   863→    audit("EXEC-CLOSE", { serial: it.serial, reason, flash: flash.ok, err: flash.err });
   864→    if (!flash.ok) {
   865→      // fallback: explicit MARKET close of the remaining qty
   866→      const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   867→      // v1.4.3 — clientOrderId mandatory (pre-1.4.3 the SL-close fallback was
   868→      // rejected -1004 and the transition was lost forever).
   869→      const r = await placeOrder(it.tbSymbol, side, it.qtyRemaining, { clientOrderId: `NX${it.serial}X${Date.now() % 100000}` });
   870→      audit("EXEC-CLOSE-FALLBACK", { serial: it.serial, reason, ok: r.ok, err: r.err });
   871→      if (!r.ok) {
   872→        it.status = "OPEN";
   873→        it.events.push({ at: Date.now(), ev: "CLOSE_REJECTED", note: (r.err ?? "").slice(0, 160) });
   874→        revertTransition(it, "CLOSE", r.err ?? "close order rejected");
   875→        return;
   876→      }
   877→    }
   878→  }
   879→  // v1.5.0 — PAPER close books the remaining slice at the LIVE taker side and
   880→  // settles the intent's full realized PnL into the fixed-$3 account.
   881→  if (it.mode === "PAPER" && it.status === "CLOSING") {
   882→    const tick = await tbTicker(it.tbSymbol);
   883→    const exit = tick ? (it.direction === "LONG" ? tick.bid : tick.ask) : null;
   884→    const entry = it.fillPrice ?? it.cardEntry;
   885→    const sign = it.direction === "LONG" ? 1 : -1;
   886→    const mult = it.contractMult ?? instruments.get(it.tbSymbol)?.mult ?? 1;
   887→    const pnl = exit !== null && entry > 0 ? (exit - entry) * it.qtyRemaining * sign * mult : 0;
   888→    it.realizedPnlUsdt = Number(((it.realizedPnlUsdt ?? 0) + pnl).toFixed(6));
   889→    audit("EXEC-PNL-PAPER", { serial: it.serial, reason, exit: exit ?? null, closePnl: Number(pnl.toFixed(6)), totalPnl: it.realizedPnlUsdt });
   890→    if (pnl !== 0) paperBook(it, pnl, `close:${reason}`);
   891→    S.paper.closedN++;
   892→    if ((it.realizedPnlUsdt ?? 0) > 0) S.paper.wins++; else S.paper.losses++;
   893→  }
   894→  it.qtyRemaining = 0;
   895→  it.status = "CLOSED";
   896→  it.closedAt = Date.now();
   897→  it.closeReason = reason;
   898→  it.actionAttempts = 0;
   899→  if (it.mode === "LIVE") {
   900→    // v1.4.5 — DETECTed-but-never-sent intents (exec gate off / gates skip) own
   901→    // no exchange fills: closing them must NOT sum the symbol's fill history.
   902→    const neverExecuted = !it.orderId && !it.openedAt;
   903→    if (neverExecuted) {
   904→      it.closeReason = `${reason} (بدون اجرا)`;
   905→      audit("EXEC-PNL-SKIP", { serial: it.serial, reason: it.closeReason, why: "intent never executed — no exchange fills belong to it" });
   906→    } else {
   907→      const pnl = await fetchRealizedPnl(it); // v1.4.3 — money truth of this close
   908→      if (pnl !== null) it.realizedPnlUsdt = pnl;
   909→      audit("EXEC-PNL", { serial: it.serial, reason, realizedPnlUsdt: pnl });
   910→    }
   911→  }
   912→  it.events.push({ at: it.closedAt, ev: "CLOSED", note: reason });
   913→  persistState();
   914→}

   917→let pollBusy = false;
   918→async function pollCards(): Promise<void> {
   919→  if (pollBusy) return;
   920→  pollBusy = true;
   921→  try {
   922→    if (instrumentsAt === 0 || Date.now() - instrumentsAt > 10 * 60_000) await refreshInstruments();
   923→    const cards = fetchRecentCards(60);
   924→    if (cards.length === 0) return;
   925→    const newest = Math.max(...cards.map((c) => parseTs(c.createdAt)));
   926→    if (newest > S.lastCreatedAtMs) S.lastCreatedAtMs = newest;
   927→
   928→    // ⓪ v1.4.0 RETRY WINDOW — a freed capacity slot inside the ENTRY-DELAY
   929→    // window re-opens a MAX-CONCURRENT-skipped card (was: skipped forever).
   930→    if (retryList.length > 0 && !killFile() && S.cfg.execEnabled) {
   931→      const openCount0 = Object.values(S.intents).filter((x) => x.status === "OPEN" || x.status === "OPENING").length;
   932→      if (S.cfg.maxConcurrent > 0 && openCount0 < S.cfg.maxConcurrent) {
   933→        const nowMs = Date.now();
   934→        retryList = retryList.filter((r) => {
   935→          const it = S.intents[r.cardId];
   936→          if (!it || it.status !== "SKIP") return false; // resolved otherwise
   937→          const age = nowMs - (it.createdAt0 ?? r.createdAt0);
   938→          if (age > S.cfg.maxEntryAgeMs) { audit("RETRY-EXPIRED", { serial: it.serial, ageSec: Math.round(age / 1000) }); return false; }
   939→          it.status = "OPENING";
   940→          it.events.push({ at: nowMs, ev: "RETRY", note: "capacity freed inside entry window" });
   941→          pushAction({ kind: "open", cardId: r.cardId, reason: "RETRY-MAX-CONCURRENT" });
   942→          audit("RETRY-OPEN", { serial: it.serial, symbol: it.symbol });
   943→          return false;
   944→        });
   945→      }
   946→    }
   947→
   948→    // ① brand-new cards → intent
   949→    for (const c of cards) {
   950→      if (S.intents[c.id]) continue;
   951→      if (c.status !== "ACTIVE") continue; // only live cards are executed
   952→      const created = parseTs(c.createdAt);
   953→      const age = Date.now() - created;
   954→      const tbSymbol = toobitSymbol(c.symbol);
   955→      const known = instruments.has(tbSymbol);
   956→      const openCount = Object.values(S.intents).filter((x) => x.status === "OPEN" || x.status === "OPENING").length;
   957→      let skip: string | null = null;
   958→      if (age > STALE_SKIP_MS) skip = "STALE-CARD";
   959→      else if (!known) skip = "SYMBOL-ABSENT-ON-TOOBIT";
   960→      else if (!killFile() && S.cfg.execEnabled && S.cfg.maxConcurrent > 0 && openCount >= S.cfg.maxConcurrent) skip = "MAX-CONCURRENT";
   961→      if (skip) countSkip(skip); // v1.3.0 doctrine funnel
   962→      if (skip === "MAX-CONCURRENT") retryList.push({ cardId: c.id, createdAt0: created }); // v1.4.0 — eligible for the retry window

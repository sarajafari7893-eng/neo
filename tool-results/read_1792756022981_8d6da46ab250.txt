   357→// ── weight table (Binance USDT-M futures documented weights) ────────────────
   358→// klines ~2 (limit≤100) · depth 2-5-10 by limit · aggTrades 20 · trades 20 ·
   359→// ticker/24hr 1 per-symbol / 40 full-universe · premiumIndex 1 per-symbol /
   360→// 10 all-markers · futures/data/* 1 (openInterestHist, L/S ratios, basis…) ·
   361→// exchangeInfo 1 · ping/time 1 · unknown default 5 (conservative).
   362→function resolveWeight(path: string): number {
   363→  const qIdx = path.indexOf("?");
   364→  const ep = (qIdx === -1 ? path : path.slice(0, qIdx)).replace(/^\//, "").replace(/\/+$/, "");
   365→  const query = qIdx === -1 ? "" : path.slice(qIdx + 1);
   366→  const limit = (): number | null => {
   367→    const m = /[?&]limit=(\d+)/.exec(query);
   368→    return m ? Number(m[1]) : null;
   369→  };
   370→  switch (ep) {
   371→    case "fapi/v1/ping":
   372→    case "fapi/v1/time":
   373→    case "fapi/v1/exchangeInfo":
   374→      return 1;
   375→    case "fapi/v1/klines": {
   376→      const l = limit() ?? 500;
   377→      return l <= 100 ? 2 : l <= 499 ? 5 : 10;
   378→    }
   379→    case "fapi/v1/depth": {
   380→      const l = limit() ?? 100;
   381→      return l <= 100 ? 2 : l <= 500 ? 5 : 10;
   382→    }
   383→    case "fapi/v1/aggTrades":
   384→    case "fapi/v1/trades":
   385→      return 20;
   386→    case "fapi/v1/ticker/24hr":
   387→      return /[?&]symbol=/.test(query) ? 1 : 40;
   388→    case "fapi/v1/premiumIndex":
   389→      return /[?&]symbol=/.test(query) ? 1 : 10;
   390→    default:
   391→      if (ep.startsWith("futures/data/")) return 1; // openInterestHist / L-S ratios / basis family
   392→      return 5; // unknown → conservative default (covers fapi/v1/fundingRate etc.)
   393→  }
   394→}

   396→function pruneWindow(now: number): void {
   397→  const cutoff = now - WINDOW_MS;
   398→  while (state.spend.length > 0 && state.spend[0]!.at <= cutoff) {
   399→    const e = state.spend.shift()!;
   400→    state.windowSum -= e.w;
   401→  }
   402→  if (state.spend.length > SPEND_CAP) {
   403→    // defensive: drop the oldest half rather than grow unbounded
   404→    const keep = state.spend.slice(Math.floor(state.spend.length / 2));
   405→    let sum = 0;
   406→    for (const e of keep) sum += e.w;
   407→    state.spend = keep;
   408→    state.windowSum = sum;
   409→  }
   410→}

   412→/** Reserve `weight` from the rolling window. Returns the spend entry (for refunds) or null. */
   413→function takeWeight(weight: number): SpendEntry | null {
   414→  const now = Date.now();
   415→  pruneWindow(now);
   416→  if (state.windowSum + weight > state.budgetPer60s) return null;
   417→  const entry: SpendEntry = { at: now, w: weight };
   418→  state.spend.push(entry);
   419→  state.windowSum += weight;
   420→  return entry;
   421→}

   423→/** 418/429 responses carry no weight cost — give the reservation back. */
   424→function refundWeight(entry: SpendEntry | null): void {
   425→  if (!entry) return;
   426→  const idx = state.spend.indexOf(entry);
   427→  if (idx === -1) return; // already pruned out of the window (its cost is gone too)
   428→  state.spend.splice(idx, 1);
   429→  state.windowSum -= entry.w;
   430→}

   433→  if (until > state.banUntil) {
   434→    state.banUntil = until;
   435→    state.stats.lastBanUntil = until;
   436→  }

   439→/** Record a 418/429 body. Returns { ip, until } when the body NAMES the banned egress IP. */
   440→function noteBanBody(status: number, body: string): { ip: string; until: number } | null {
   441→  const now = Date.now();
   442→  const named = /IP\(([\d.]+)\)\s*banned until (\d+)/i.exec(body);
   443→  if (named) {
   444→    const ip = named[1]!;
   445→    // v12.1 TASK-49: the LEDGER carries Binance's REAL end (sanity-capped 24h) —
   446→    // the cosmetic +35min cap only applies to the global silence window below.
   447→    // Capping the ledger made the early probe fire at a fake expiry and the
   448→    // resulting 418 contact re-escalated the real ban (11:58→12:34→13:09 spiral).
   449→    const realUntil = Math.min(Number(named[2]), now + 24 * 60_000);
   450→    const until = Math.min(realUntil, now + BAN_CAP_MS);
   451→    state.ipBans.set(ip, realUntil);
   452→    state.lastBannedIp = ip;
   453→    persistBans(); // v12.0 — named-IP ban ledger survives restarts (30s debounce)
   454→    if (state.ipBans.size > IP_BAN_MAP_CAP) {
   455→      // evict the soonest-expiring record
   456→      let oldestKey: string | null = null;
   457→      let oldestVal = Infinity;
   458→      for (const [k, v] of state.ipBans) if (v < oldestVal) { oldestVal = v; oldestKey = k; }
   459→      if (oldestKey) state.ipBans.delete(oldestKey);
   460→    }
   461→    state.recentBanEvents.push({ at: now, ip });
   462→    if (state.recentBanEvents.length > 64) state.recentBanEvents.splice(0, state.recentBanEvents.length - 64);
   463→    return { ip, until };
   464→  }
   465→  const untilMatch = /banned until (\d+)/i.exec(body);
   466→  if (untilMatch) {
   467→    armGlobalBan(Math.min(Number(untilMatch[1]), now + BAN_CAP_MS)); // unknown scope — assume it's us
   468→    return null;
   469→  }
   470→  // unparseable body — conservative defaults (taxonomy precedent for 418)
   471→  armGlobalBan(now + (status === 429 ? BAN_429_DEFAULT_MS : BAN_418_DEFAULT_MS));
   472→  return null;
   473→}

   475→/** Every retry landed on a (named) banned IP — escalate the global silence window. */
   476→function escalatePoolBlock(worstNamedUntil: number): void {
   477→  const now = Date.now();
   478→  state.allBanStreak++;
   479→  if (state.allBanStreak >= POOL_STREAK_FOR_LONG_BLOCK) {
   480→    // repeated independent evidence — the egress pool is saturated: full
   481→    // zero-network window, capped at 35min, then a fresh probe self-heals.
   482→    armGlobalBan(Math.min(worstNamedUntil, now + BAN_CAP_MS));
   483→    state.allBanStreak = 0;
   484→  } else {
   485→    armGlobalBan(now + POOL_BLOCK_SHORT_MS);
   486→  }
   487→}

   499→/**
   500→ * THE gate body. Resolves parsed JSON, or null on: live IP ban, weight-budget
   501→ * shed, HTTP failure, timeout, or parse error. Never throws, never queues.
   502→ */
   503→async function executeGated<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   504→  const lane: FapiGatePriority = opts?.priority ?? "telemetry";
   505→  const weight = Math.max(1, Math.round(opts?.weight ?? resolveWeight(path)));
   506→  const timeoutMs = opts?.timeoutMs ?? DEFAULT_TIMEOUT_MS;
   507→
   508→  let entry = takeWeight(weight);
   509→  if (!entry) {
   510→    if (lane !== "critical") {
   511→      state.stats.droppedBudget++;
   512→      return null; // telemetry: drop instead of queueing a storm
   513→    }
   514→    await delay(CRITICAL_RETRY_MS); // one small negative retry
   515→    entry = takeWeight(weight);
   516→    if (!entry) {
   517→      state.stats.droppedBudget++;
   518→      return null;
   519→    }
   520→  }
   521→
   522→  const host = opts?.host ?? FAPI_HOST;
   523→  let attempt = 0;
   524→
   525→  while (true) {
   526→    const ctrl = new AbortController();
   527→    const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   528→    let res: Response;
   529→    try {
   530→      // v12.6.1 CONNECTION-RECYCLE (measured 2026-10-20 19:41Z, bun runtime):
   531→      // a keep-alive fetch POOLS one TCP flow ⇒ ONE egress IP for its whole
   532→      // life — a pinned-banned connection turned 231 gate calls into 0 allowed
   533→      // while 6/10 fresh connections succeeded. `Connection: close` = a fresh
   534→      // NAT lottery draw per request; the TLS cost (~60ms) is trivial next to
   535→      // the starvation it prevents.
   536→      res = await fetch(`${host}${path}`, { signal: ctrl.signal, cache: "no-store", headers: { Connection: "close" } });
   537→    } catch {
   538→      return null; // timeout/abort/network — weight may have been consumed, keep the reservation
   539→    } finally {
   540→      clearTimeout(timer);
   541→    }
   542→
   543→    if (res.ok) {
   544→      const text = await res.text().catch(() => "");
   545→      if (!text) return null; // empty 200 — nothing to invent
   546→      state.stats.allowed++;
   547→      state.allBanStreak = 0; // a clean 2xx proves the pool still serves us
   548→      try {
   549→        return JSON.parse(text) as T;
   550→      } catch {
   551→        return null;
   552→      }
   553→    }
   554→
   555→    if (res.status === 418 || res.status === 429) {
   556→      const body = await res.text().catch(() => "");
   557→      refundWeight(entry); // banned responses carry no weight cost
   558→      entry = null;
   559→      const named = noteBanBody(res.status, body);
   560→      if (opts?.singleProbe) return null; // v12.0 early probe — single attempt, window stays armed
   561→      if (!named) return null; // unknown-scope ban — global zero-network window already armed
   562→      if (attempt >= BAN_RETRIES) {
   563→        escalatePoolBlock(named.until);
   564→        return null;
   565→      }
   566→      attempt++;
   567→      state.stats.retriesAfterBan++;
   568→      await delay(BAN_RETRY_DELAYS_MS[Math.min(attempt - 1, BAN_RETRY_DELAYS_MS.length - 1)]!); // span the NAT rotation window
   569→      entry = takeWeight(weight);
   570→      if (!entry) {
   571→        state.stats.droppedBudget++;
   572→        return null;
   573→      }
   574→      continue;
   575→    }
   576→
   577→    return null; // other non-ok status — honest null, no retry
   578→  }
   579→}

   587→// v12.1 TASK-49 — SHARED LEDGER REFRESH: trade-service / price-ws write their

   599→    let unknownMax = 0;

   602→      if (isUnknownScopeKey(key)) {
   603→        if (until > unknownMax) unknownMax = until;
   604→        continue;
   605→      }
   606→      if (until > (state.ipBans.get(key) ?? 0)) state.ipBans.set(key, until);

   612→    }

   624→export async function gateFetch<T>(path: string, opts?: GateFetchOpts): Promise<T | null> {
   625→  state.stats.attempted++;
   626→  const now = Date.now();
   627→
   628→  // v12.1 TASK-49 — LAZY LEDGER GUARD: the instant our global window expires,
   629→  // re-check the shared ledger (throttled to one disk read / 15s) BEFORE any
   630→  // network call — closes the poke-gap where the window died but the refresh
   631→  // interval hadn't fired yet. While the pool is ledger-banned, zero pokes.
   632→  if (now >= state.banUntil && now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   633→  if (now < state.banUntil) state.stats.droppedBan++; // ledger re-armed the window — shed instantly
   634→
   635→  // GLOBAL BAN AWARENESS — while a global ban window is live, zero network calls
   636→  // …except the v12.0 ROTATION-AWARE EARLY PROBE: one critical-lane attempt per
   637→  // PROBE_COOLDOWN_MS when the sampled egress IP is clean — the pool rotates,
   638→  // so the window can be LIFTED the moment a clean draw appears instead of
   639→  // waiting out the full 35min cap. A failed probe keeps the window armed.
   640→  if (now < state.banUntil) {
   641→    state.stats.droppedBan++;
   642→    const canProbe = (opts?.priority ?? "telemetry") === "critical" && now - state.lastProbeAt >= PROBE_COOLDOWN_MS;
   643→    if (!canProbe) return null;
   644→    // v12.1 TASK-49 — LEDGER FRESHNESS BEFORE PROBE: state may hold a cosmetic
   645→    // cap while the disk carries a longer REAL end another process discovered.
   646→    // A probe fired on stale state pokes a still-banned pool and re-escalates.
   647→    // Re-read the ledger (15s throttle) and re-arm BEFORE the probe decision.
   648→    if (now - lastLedgerCheckAt >= 15_000) refreshLedgerOnce();
   649→    if (now < state.banUntil) return null; // ledger extended the silence — no probe
   650→    state.lastProbeAt = now;
   651→    const ip = await sampleEgressIp();
   652→    // v12.1 TASK-49: expiry-aware check — a recorded ban whose real end passed
   653→    // no longer blocks the probe (the old .has() kept ghost entries armed).
   654→    if (!ip || isIpBanned(ip, now)) return null; // sampled draw is banned/unknown — stay silent
   655→    state.stats.probesIssued++;
   656→    const probeResult = await executeGated<T>(path, { ...opts, singleProbe: true });
   657→    if (probeResult !== null) {
   658→      state.banUntil = 0; // clean draw proved the pool rotated — lift instantly
   659→      state.allBanStreak = 0;
   660→      state.stats.probeLifts++;
   661→      console.log(`[fapi-gate] v12.0 ban window LIFTED early — egress rotated to clean IP ${ip}`);
   662→    }
   663→    return probeResult;
   664→  }
   665→
   666→  // SINGLE-FLIGHT per (host,path): share the request while it is in flight and
   667→  // for a 2s result window — concurrent module askers pay the weight once.
   668→  const key = `${opts?.host ?? FAPI_HOST}${path}`;
   669→  const existing = state.inflight.get(key);
   670→  if (existing) {
   671→    const age = now - existing.startedAt;
   672→    if (!existing.settled || age < SF_RESULT_MS) {
   673→      state.stats.singleFlightShared++;
   674→      return existing.promise as Promise<T | null>;
   675→    }
   676→    state.inflight.delete(key);
   677→  }
   678→  if (state.inflight.size >= SF_MAP_CAP) pruneInflight(now);
   679→
   680→  const promise = executeGated<T>(path, opts);
   681→  const entry = { startedAt: now, settled: false, promise };
   682→  state.inflight.set(key, entry);
   683→  void promise
   684→    .catch(() => null) // executeGated never rejects — belt & suspenders
   685→    .finally(() => {
   686→      entry.settled = true;
   687→    });
   688→  return promise;
   689→}

   113→async function apiCall<T>(method: string, body: unknown, timeoutMs = 4500): Promise<{ ok: boolean; result?: T; error?: string }> {
   114→  const ctrl = new AbortController();
   115→  const timer = setTimeout(() => ctrl.abort(), timeoutMs);
   116→  try {
   117→    const res = await fetch(`https://api.telegram.org/bot${TOKEN}/${method}`, {
   118→      method: "POST",
   119→      headers: { "content-type": "application/json" },
   120→      body: JSON.stringify(body),
   121→      signal: ctrl.signal,
   122→      cache: "no-store",
   123→    });
   124→    const json = (await res.json()) as { ok: boolean; result?: T; description?: string };
   125→    return json.ok ? { ok: true, result: json.result } : { ok: false, error: json.description ?? `HTTP ${res.status}` };
   126→  } catch (e: unknown) {
   127→    return { ok: false, error: e instanceof Error ? (e.name === "AbortError" ? `timeout ${timeoutMs}ms` : e.message) : String(e) };
   128→  } finally {
   129→    clearTimeout(timer);
   130→  }
   131→}

   133→/** Multipart photo upload (sendPhoto) — FormData/Blob are global in Node 18+. */
   134→async function apiSendPhoto(chat: string, caption: string, photo: Buffer, filename: string): Promise<{ ok: boolean; error?: string }> {
   135→  const ctrl = new AbortController();
   136→  const timer = setTimeout(() => ctrl.abort(), 9000);
   137→  try {
   138→    const isPng = filename.toLowerCase().endsWith(".png");
   139→    const form = new FormData();
   140→    form.append("chat_id", chat);
   141→    form.append("caption", caption.slice(0, 1024));
   142→    form.append("parse_mode", "HTML");
   143→    form.append("photo", new Blob([new Uint8Array(photo)], { type: isPng ? "image/png" : "image/jpeg" }), filename);
   144→    const res = await fetch(`https://api.telegram.org/bot${TOKEN}/sendPhoto`, { method: "POST", body: form, signal: ctrl.signal, cache: "no-store" });
   145→    const json = (await res.json()) as { ok: boolean; description?: string };
   146→    return json.ok ? { ok: true } : { ok: false, error: json.description ?? `HTTP ${res.status}` };
   147→  } catch (e: unknown) {
   148→    return { ok: false, error: e instanceof Error ? (e.name === "AbortError" ? "timeout 9000ms" : e.message) : String(e) };
   149→  } finally {
   150→    clearTimeout(timer);
   151→  }
   152→}

   248→async function buildCard(type: string, symbol: string | null, engineMessage: string, signalId: string | null): Promise<RelayCard> {
   249→  const head = HEADLINE[type] ?? { emoji: "📡", title: type };
   250→  const sig = signalId ? await db.signal.findUnique({ where: { id: signalId } }).catch(() => null) : null;
   251→
   252→  if (!sig) {
   253→    // honest fallback — engine line only
   254→    return { text: `${head.emoji} <b>${APP_NAME} — ${esc(head.title)}</b>\n${symbol ? `📊 <b>${esc(symbol)}</b>\n` : ""}\n<blockquote>${esc(engineMessage)}</blockquote>` };
   255→  }
   256→
   257→  const text = composeSignalText(type, sig, engineMessage);
   258→  const banner = BANNER_STATUS[type];
   259→  if (!banner) return { text };
   260→  // CREATED cards are direction-aware: the red bear artwork ships for SHORT
   261→  // deployments, the rocket/bull for LONG.
   262→  const kind: BannerStatus = type === "CREATED" && sig.direction === "SHORT" ? "SHORT" : banner;
   263→
   264→  const serial = `NX-${String(sig.serial).padStart(4, "0")}`;
   265→  const pnl = sig.pnlPercent ?? 0;
   266→  const pnlLine = `${pnl >= 0 ? "+" : ""}${pnl.toFixed(1)}% NET`;
   267→  const reached = sig.reachedTp ?? 0;
   268→  const line2 = type === "SL" ? `RISK CONTAINED · LADDER ${reached}/3` : type === "TTL_EXPIRED" ? `TTL ${sig.ttlMinutes}M ELAPSED · LADDER ${reached}/3` : type === "EARLY_EXIT" ? `EARLY CUT (R2/R7) · LADDER ${reached}/3` : type === "TTL_EXTENDED" ? `+${sig.ttlExtensions} EXT · STILL LIVE` : type === "AFTERLIFE" ? `POST-EXPIRY ${((sig.afterlife ?? "TRACKING").replace(/_/g, " "))} · MFE ${((sig.maxPostPnlPercent ?? 0) >= 0 ? "+" : "")}${(sig.maxPostPnlPercent ?? 0).toFixed(1)}%` : type === "CREATED" ? `ENTRY ${fmtPrice(sig.entryPrice)} · RR 1:${sig.riskReward.toFixed(1)}` : `LADDER ${reached}/3 BANKED · RR 1:${sig.riskReward.toFixed(1)}`;
   269→  const spec = {
   270→    status: kind,
   271→    title: `${sig.symbol} ${sig.direction}`,
   272→    line1: `${serial} · ${pnlLine}`,
   273→    line2,
   274→    ts: tehranStamp(),
   275→  };
   276→  // v8.1: professional composite (AI art + coin logo + live data) first; the
   277→  // operator JPG set and the pixel-font banner remain as fallbacks.
   278→  try {
   279→    const photo = await renderSignalBannerV2(spec);
   280→    return { text, photo, photoName: `${kind.toLowerCase()}-${serial}.png` };
   281→  } catch { /* sharp failure → legacy chain below */ }
   282→  const art = await loadOperatorArt(kind);
   283→  if (art) return { text, photo: art, photoName: ART_FILES[kind] };
   284→  return { text, photo: renderStatusBanner(spec), photoName: `${kind.toLowerCase()}-${serial}.png` };
   285→}

   287→/** Fire-and-forget relay — NEVER await this from the engine hot path. */
   288→export function relaySignalEvent(type: string, symbol: string | null, message: string, signalId?: string | null): void {
   289→  if (!relayArmed() || !RELAYABLE_TYPES.has(type)) return;
   290→  void (async () => {
   291→    const chat = await resolveChat();
   292→    if (!chat) return; // honest pending state — not an error, no doomed send
   293→    // v9.1 RAW-CAL SPAM GUARD → v10.4 REBALANCE (operator mandate: «گزارشات
   294→    // کامل و دقیق به تلگرام ارسال نمیشه، مثل کارت های سیگنال جدید، استاپ و
   295→    // دیگر کارتها»): RAW-CAL CREATED cards no longer VANISH — they ship as
   296→    // compact TEXT cards (structured HTML body, photo suppressed) so the
   297→    // capture run stays fully visible without flooding the chat with ~5 heavy
   298→    // photo posters/hour. Outcome cards (SL / EARLY_EXIT / TTL_EXPIRED / TP*)
   299→    // keep their full photo treatment — that is where the detail matters.
   300→    let rawCalTextOnly = false;
   301→    if (type === "CREATED" && signalId) {
   302→      const sig = await db.signal.findUnique({ where: { id: signalId }, select: { tags: true } }).catch(() => null);
   303→      if (sig?.tags?.includes("RAW-CAL")) rawCalTextOnly = true;
   304→    }
   305→    let card: RelayCard;
   306→    try {
   307→      card = await buildCard(type, symbol, message, signalId ?? null);
   308→    } catch {
   309→      card = { text: `📡 ${APP_NAME} ${type}${symbol ? ` · ${esc(symbol)}` : ""}\n${esc(message)}`.slice(0, 3800) };
   310→    }
   311→    if (rawCalTextOnly) card = { text: card.text }; // v10.4: strip photo, keep the full HTML body
   312→    let r: { ok: boolean; error?: string };
   313→    if (card.photo) {
   314→      r = await apiSendPhoto(chat, card.text, card.photo, card.photoName ?? "flowradar-signal.png");
   315→      if (!r.ok) {
   316→        // photo failed (size/network) — retry once as plain HTML text so the
   317→        // alert still lands; banner is presentation, the message is the payload
   318→        r = await apiCall<unknown>("sendMessage", { chat_id: chat, text: card.text.slice(0, 3800), parse_mode: "HTML", disable_web_page_preview: true });
   319→      }
   320→    } else {
   321→      r = await apiCall<unknown>("sendMessage", { chat_id: chat, text: card.text.slice(0, 3800), parse_mode: "HTML", disable_web_page_preview: true });
   322→    }
   323→    if (r.ok) {
   324→      sentCount++;
   325→      lastSentAt = Date.now();
   326→    } else {
   327→      failCount++;
   328→      pushErr(`${type}: ${r.error}`);
   329→      console.error(`[NOTIFY] telegram relay send failed (${type}): ${r.error}`);
   330→    }
   331→  })();
   332→}

   341→/** Operator-initiated proof: send an arbitrary line to the resolved chat. */
   342→export async function sendDirect(text: string): Promise<{ ok: boolean; error?: string; chatId?: string }> {
   343→  const chat = await resolveChat();
   344→  if (!TOKEN) return { ok: false, error: "TELEGRAM_BOT_TOKEN unset — relay dormant" };
   345→  if (!chat) {
   346→    if (botId !== null && (chatOverride ?? ENV_CHAT) === String(botId)) {
   347→      return { ok: false, error: `TELEGRAM_CHAT_ID (${ENV_CHAT}) is the bot's own id — Telegram rejects bot→bot sends. Message the bot once from your account, then run discover.` };
   348→    }
   349→    return { ok: false, error: chatOverride || ENV_CHAT ? "chat unresolved (getMe pending or override pending)" : "TELEGRAM_CHAT_ID unset and no discovered chat yet" };
   350→  }
   351→  const r = await apiCall<unknown>("sendMessage", { chat_id: chat, text, disable_web_page_preview: true });
   352→  if (r.ok) { sentCount++; lastSentAt = Date.now(); } else { failCount++; pushErr(`direct: ${r.error}`); }
   353→  return { ok: r.ok, error: r.error, chatId: chat };
   354→}

   476→export function ensureAutoDiscover(): void {
   477→  if (autoDiscoverTimer || !relayArmed()) return;
   478→  // v8.3 BRIDGE COEXISTENCE: the chat id is already persisted (EngineSetting
   479→  // telegram.chatId) since v6.2.2 — discovery is DONE. The legacy 30s
   480→  // getUpdates probe now 409-terminates the v8.3 bridge long-poller, so the
   481→  // loop may only start after a DB-backed override check FAILS, and each
   482→  // tick re-checks the override BEFORE touching getUpdates.
   483→  void (async () => {
   484→    try {
   485→      await ensureBotIdentity();
   486→      await loadOverride();
   487→    } catch {
   488→      /* identity check is best-effort */
   489→    }
   490→    if (chatResolvedNow()) {
   491→      console.log("[NOTIFY] auto-discover SKIPPED — chat already resolved (v8.3 bridge owns getUpdates)");
   492→      return;
   493→    }
   494→    const timer: ReturnType<typeof setInterval> = setInterval(() => {
   495→      if (chatResolvedNow()) {
   496→        clearInterval(timer);
   497→        autoDiscoverTimer = null;
   498→        return;
   499→      }
   500→      void (async () => {
   501→        try {
   502→          await ensureBotIdentity();
   503→          await loadOverride();
   504→        } catch {
   505→          /* best-effort */
   506→        }
   507→        if (chatResolvedNow()) {
   508→          clearInterval(timer);
   509→          autoDiscoverTimer = null;
   510→          return;
   511→        }
   512→        const r = await discoverChat();
   513→        if (r.ok) {
   514→          clearInterval(timer);
   515→          autoDiscoverTimer = null;
   516→          const proof = await sendDirect(
   517→            `🟢 ${APP_NAME} relay ARMED — operator chat discovered and persisted.\nFrom now on CREATED / TP1-3 / SL / TTL_EXPIRED / TTL_EXTENDED / AFTERLIFE events mirror here as structured cards (WIN / STOP / EXPIRED outcomes also arrive as rendered banner images).`
   518→          );
   519→          console.log(`[NOTIFY] auto-discover complete — confirmation ${proof.ok ? "delivered" : `NOT delivered (${proof.error ?? "unknown"})`}`);
   520→        }
   521→        // ok:false → "no messages waiting" (benign, keep polling) or a
   522→        // transient network error (silent retry on the next 30s pass).
   523→      })();
   524→    }, 30_000);
   525→    autoDiscoverTimer = timer;
   526→    console.log("[NOTIFY] auto-discover loop started — polling getUpdates every 30s until the operator messages the bot");
   527→  })();

   530→/** Learn the operator chat from the bot's update stream — the operator sends
   531→ * ANY message to the bot in Telegram, we read getUpdates and persist the chat. */
   532→export async function discoverChat(): Promise<{ ok: boolean; error?: string; chatId?: string; from?: string; text?: string }> {
   533→  if (!TOKEN) return { ok: false, error: "TELEGRAM_BOT_TOKEN unset — relay dormant" };
   534→  await ensureBotIdentity();
   535→  const r = await apiCall<Array<{ message?: { chat: { id: number; first_name?: string; username?: string }; text?: string } }>>("getUpdates", { limit: 10, timeout: 0 }, 8000);
   536→  if (!r.ok || !r.result) return { ok: false, error: r.error ?? "getUpdates returned no payload" };
   537→  const withMsg = r.result.filter((u) => u.message?.chat?.id !== undefined);
   538→  if (withMsg.length === 0) return { ok: false, error: "no messages waiting — open the bot in Telegram, press START and send any message (e.g. «سلام»), then retry discover" };
   539→  const last = withMsg[withMsg.length - 1].message!;
   540→  const chatId = String(last.chat.id);
   541→  chatOverride = chatId;
   542→  overrideLoaded = true;
   543→  await db.engineSetting.upsert({ where: { key: CHAT_OVERRIDE_KEY }, create: { key: CHAT_OVERRIDE_KEY, value: chatId }, update: { value: chatId } });
   544→  console.log(`[NOTIFY] telegram chat DISCOVERED — id ${chatId} (${last.chat.first_name ?? last.chat.username ?? "unknown"}), persisted to ${CHAT_OVERRIDE_KEY}; relay fully ARMED`);
   545→  return { ok: true, chatId, from: last.chat.first_name ?? last.chat.username, text: last.text };
   546→}

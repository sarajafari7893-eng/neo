   126→const SAMPLE_CAP = 8000;
   127→const BLOCK_CAP = 800;
   128→const DEPLOY_CAP = 300;
   129→const BOUNDARY_DEDUP_MS = 60_000; // per symbol|direction — boundary state persists tick after tick
   130→const REJECT_DEDUP_MS = 150_000;
   131→const PASS_DEDUP_MS = 60_000; // PASS verdicts change slowly too — keep the ring dense with boundary data
   132→const BLOCK_DEDUP_MS = 120_000;

   134→const DATA_DIR = path.join(process.cwd(), ".data");
   135→const FILE = path.join(DATA_DIR, "gate-journal.jsonl");
   136→const FILE_MAX_BYTES = 8 * 1024 * 1024;
   137→
   138→const samples: GateSample[] = [];
   139→const blocks: GateBlock[] = [];
   140→const deploys: GateDeploy[] = [];
   141→const lastAccepted = new Map<string, number>();
   142→let persisted = 0;
   143→let hydrating: Promise<void> | null = null;
   144→// v8.2.1 SPLIT-BRAIN COUNTERMEASURE — Turbopack dev compiles one module
   145→// instance PER ROUTE: the engine tick's instance writes the JSONL while a
   146→// freshly-compiled read route starts with EMPTY rings and (before this fix)
   147→// never hydrated, because hydration was only triggered from write paths.
   148→// readPos tracks how many bytes of the JSONL this instance has already
   149→// ingested, so reads can TAIL the file incrementally no matter which
   150→// instance is doing the writing (including our own — persist() advances
   151→// readPos so records we wrote ourselves are never re-ingested).
   152→let readPos = 0;
   153→let hydratedAt = 0; // epoch ms of the last successful hydrate/refresh — 0 = cold rings
   154→let refreshing: Promise<void> | null = null;
   155→let lastRefreshAt = 0;
   156→const REFRESH_MIN_INTERVAL_MS = 10_000;
   157→const REFRESH_MAX_BITE = 512 * 1024; // bounded bite per refresh
   158→const seenSerials = new Set<number>(); // deploy serials ingested (hydrate + refresh) — cross-instance deploy dedupe
   159→
   160→async function hydrate(): Promise<void> {
   161→  if (hydrating) return hydrating;
   162→  hydrating = (async () => {
   163→    try {
   164→      const buf = await readFile(FILE, "utf8").catch(() => "");
   165→      if (!buf) return;
   166→      const lines = buf.split("\n").filter(Boolean);
   167→      const tail = lines.slice(-6000); // bounded restore
   168→      for (const line of tail) {
   169→        try {
   170→          const rec = JSON.parse(line) as { k: "s" | "b" | "d"; v: GateSample | GateBlock | GateDeploy };
   171→          if (rec.k === "s" && samples.length < SAMPLE_CAP) samples.push(rec.v as GateSample);
   172→          else if (rec.k === "b" && blocks.length < BLOCK_CAP) blocks.push(rec.v as GateBlock);
   173→          else if (rec.k === "d" && deploys.length < DEPLOY_CAP) deploys.push(rec.v as GateDeploy);
   174→        } catch {
   175→          /* skip torn line */
   176→        }
   177→      }
   178→      const st = await stat(FILE).catch(() => null);
   179→      persisted = st?.size ?? 0;
   180→      // the whole file has been consumed (tail kept) — mark the read position
   181→      readPos = st?.size ?? 0;
   182→      hydratedAt = Date.now();
   183→      // prime the dedup map from the hydrated history so a freshly-compiled
   184→      // instance does not instantly re-accept everything another instance
   185→      // journaled seconds ago (dedup continuity across instance swaps)
   186→      for (const s of samples) {
   187→        const key = `${s.symbol}|${s.direction}|${s.band}`;
   188→        const prev = lastAccepted.get(key) ?? 0;
   189→        if (s.at > prev) lastAccepted.set(key, s.at);
   190→      }
   191→      for (const b of blocks) {
   192→        const key = `${b.gate}|${b.symbol ?? "*"}`;
   193→        const prev = lastAccepted.get(key) ?? 0;
   194→        if (b.at > prev) lastAccepted.set(key, b.at);
   195→      }
   196→      for (const d of deploys) if (d.serial > 0) seenSerials.add(d.serial);
   197→      if ((st?.size ?? 0) > FILE_MAX_BYTES) {
   198→        await rename(FILE, `${FILE}.old`).catch(() => undefined);
   199→        persisted = 0; // (was a stale-size bug: rotation must reset the counter)
   200→        readPos = 0;
   201→      }
   202→    } catch {
   203→      /* fresh start — journaling must never break the engine */
   204→    }
   205→  })();
   206→  return hydrating;
   207→}

   209→async function persist(rec: { k: "s" | "b" | "d"; v: unknown }): Promise<void> {
   210→  try {
   211→    const line = JSON.stringify(rec) + "\n";
   212→    await appendFile(FILE, line, "utf8");
   213→    persisted += line.length;
   214→    readPos += line.length; // our own bytes are already in the rings — never re-ingest
   215→    if (persisted > FILE_MAX_BYTES) {
   216→      await rename(FILE, `${FILE}.old`).catch(() => undefined);
   217→      persisted = 0;
   218→      readPos = 0;
   219→    }
   220→  } catch {
   221→    /* non-fatal */
   222→  }
   223→}

   234→/** v8.2.1 READ-PATH LIVENESS — tail-ingest whatever another module instance
   235→ * wrote to the JSONL since our last read. Throttled (one stat+read per
   236→ * REFRESH_MIN_INTERVAL_MS), single-flight, rotation-safe, bounded bite.
   237→ * Cheap no-op when this instance wrote the bytes itself (readPos already
   238→ * advanced by persist()). READ ROUTES MUST CALL THIS — it is what keeps the
   239→ * gauge live across Turbopack's per-route module split-brain. */
   240→export async function gateJournalRefresh(maxAgeMs = REFRESH_MIN_INTERVAL_MS): Promise<void> {
   241→  if (Date.now() - lastRefreshAt < maxAgeMs) return;
   242→  if (refreshing) return refreshing;
   243→  refreshing = (async () => {
   244→    try {
   245→      // VERDICT-CORE audit 2026-11-05 — COLD-GENERATION READ FIX: hydrate() only
   246→      // ran on WRITE paths, so a freshly-compiled read generation started with
   247→      // readPos=0 and tailed the file's OLDEST bytes (11-03 history) — 12h-window
   248→      // stats came back EMPTY and the core wrote "journal is empty" (observed live
   249→      // on cycle #1412's first POST). Hydrate cold generations here (once, single-
   250→      // flight, readPos=file-size) so the first refresh already sees the freshest
   251→      // 6000-line history. Warm generations (hydratedAt>0) are untouched — re-running
   252→      // hydrate on them would re-push tail records into the rings.
   253→      if (hydratedAt === 0) await hydrate();
   254→      const st = await stat(FILE).catch(() => null);
   255→      if (!st) return;
   256→      if (st.size < readPos) readPos = 0; // rotated — restart the tail on the fresh file
   257→      if (st.size === readPos) return;
   258→      const len = Math.min(st.size - readPos, REFRESH_MAX_BITE);
   259→      const fh = await open(FILE, "r").catch(() => null);
   260→      if (!fh) return;
   261→      try {
   262→        const buf = Buffer.alloc(len);
   263→        const { bytesRead } = await fh.read(buf, 0, len, readPos);
   264→        let text = buf.toString("utf8", 0, bytesRead);
   265→        const lastNl = text.lastIndexOf("\n");
   266→        if (lastNl === -1) return; // partial line at the tail — wait for the writer
   267→        text = text.slice(0, lastNl + 1); // only consume COMPLETE lines
   268→        readPos += Buffer.byteLength(text, "utf8");
   269→        for (const line of text.split("\n")) {
   270→          if (!line) continue;
   271→          try {
   272→            const rec = JSON.parse(line) as { k: "s" | "b" | "d"; v: GateSample | GateBlock | GateDeploy };
   273→            if (rec.k === "s") {
   274→              const s = rec.v as GateSample;
   275→              // v8.2.1 ingest discipline — a SECOND module instance may write the
   276→              // same verdict; apply the exact same dedup rule as recordGateSamples
   277→              // so rings/stats stay honest no matter the writer
   278→              const key = `${s.symbol}|${s.direction}|${s.band}`;
   279→              const interval = s.band === "BOUNDARY" ? BOUNDARY_DEDUP_MS : s.band === "REJECT" ? REJECT_DEDUP_MS : PASS_DEDUP_MS;
   280→              const last = lastAccepted.get(key) ?? 0;
   281→              if (interval && s.at - last < interval) continue;
   282→              lastAccepted.set(key, s.at);
   283→              samples.push(s);
   284→              if (samples.length > SAMPLE_CAP) samples.splice(0, samples.length - SAMPLE_CAP);
   285→            } else if (rec.k === "b") {
   286→              const b = rec.v as GateBlock;
   287→              const key = `${b.gate}|${b.symbol ?? "*"}`;
   288→              if (b.at - (lastAccepted.get(key) ?? 0) < BLOCK_DEDUP_MS) continue;
   289→              lastAccepted.set(key, b.at);
   290→              blocks.push(b);
   291→              if (blocks.length > BLOCK_CAP) blocks.splice(0, blocks.length - BLOCK_CAP);
   292→            } else if (rec.k === "d") {
   293→              const d = rec.v as GateDeploy;
   294→              if (d.serial > 0 && seenSerials.has(d.serial)) continue; // duplicate deploy from another instance
   295→              if (d.serial > 0) seenSerials.add(d.serial);
   296→              deploys.push(d);
   297→              if (deploys.length > DEPLOY_CAP) deploys.splice(0, deploys.length - DEPLOY_CAP);
   298→            }
   299→          } catch {
   300→            /* torn line */
   301→          }
   302→        }
   303→        hydratedAt = Date.now();
   304→      } finally {
   305→        await fh.close().catch(() => undefined);
   306→      }
   307→    } catch {
   308→      /* non-fatal — reads must never break */
   309→    } finally {
   310→      lastRefreshAt = Date.now();
   311→    }
   312→  })().finally(() => {
   313→    refreshing = null;
   314→  });
   315→  return refreshing;
   316→}

   320→function bandOf(gap: number, passed: boolean): GateBand {
   321→  if (passed) return "PASS";
   322→  return gap >= -8 ? "BOUNDARY" : "REJECT";

   325→/** Journal one scan batch of threshold verdicts (engine calls this once per
   326→ * deploy-signal() evaluation pass, before the early returns). Dedup keeps
   327→ * persisting boundary/reject churn from flooding rings + disk. */
   328→export async function recordGateSamples(
   329→  batch: Array<Omit<GateSample, "at" | "best" | "band" | "gap"> & { passed: boolean }>,
   330→): Promise<void> {
   331→  if (batch.length === 0) return;
   332→  await hydrate();
   333→  const now = Date.now();
   334→  let bestGap = Infinity;
   335→  let bestIdx = -1;
   336→  const enriched = batch.map((b, i) => {
   337→    const gap = Math.abs(b.score) - b.effMin;
   338→    if (Math.abs(gap) < bestGap) {
   339→      bestGap = Math.abs(gap);
   340→      bestIdx = i;
   341→    }
   342→    return { ...b, at: now, gap, band: bandOf(gap, b.passed) };
   343→  });
   344→
   345→  for (let i = 0; i < enriched.length; i++) {
   346→    const s = enriched[i]!;
   347→    const key = `${s.symbol}|${s.direction}|${s.band}`;
   348→    const interval = s.band === "BOUNDARY" ? BOUNDARY_DEDUP_MS : s.band === "REJECT" ? REJECT_DEDUP_MS : PASS_DEDUP_MS;
   349→    const last = lastAccepted.get(key) ?? 0;
   350→    if (interval && now - last < interval) continue;
   351→    lastAccepted.set(key, now);
   352→    const rec: GateSample = {
   353→      at: s.at,
   354→      symbol: s.symbol,
   355→      direction: s.direction,
   356→      strategy: s.strategy,
   357→      score: s.score,
   358→      effMin: s.effMin,
   359→      baseMin: s.baseMin,
   360→      gap: s.gap,
   361→      band: s.band,
   362→      thr: s.thr,
   363→      confirms: s.confirms,
   364→      opposes: s.opposes,
   365→      notes: s.notes,
   366→      source: s.source,
   367→      best: i === bestIdx,
   368→    };
   369→    samples.push(rec);
   370→    if (samples.length > SAMPLE_CAP) samples.splice(0, samples.length - SAMPLE_CAP);
   371→    void persist({ k: "s", v: rec });
   372→  }
   373→}

   375→export async function recordGateBlock(b: Omit<GateBlock, "at">): Promise<void> {
   376→  await hydrate();
   377→  const now = Date.now();
   378→  const key = `${b.gate}|${b.symbol ?? "*"}`;
   379→  if (now - (lastAccepted.get(key) ?? 0) < BLOCK_DEDUP_MS) return;
   380→  lastAccepted.set(key, now);
   381→  const rec: GateBlock = { ...b, at: now };
   382→  blocks.push(rec);
   383→  if (blocks.length > BLOCK_CAP) blocks.splice(0, blocks.length - BLOCK_CAP);
   384→  void persist({ k: "b", v: rec });
   385→}

   397→/** Live gauge roster: latest verdict per symbol|direction, nearest-to-gate
   398→ * first (BOUNDARY → PASS → REJECT, each by |gap|). */
   399→export function gateLive(limit = 16): GateSample[] {
   400→  const latest = new Map<string, GateSample>();
   401→  for (const s of samples) latest.set(`${s.symbol}|${s.direction}`, s); // rings are append-ordered
   402→  const rank = (b: GateBand) => (b === "BOUNDARY" ? 0 : b === "PASS" ? 1 : 2);
   403→  return [...latest.values()]
   404→    .sort((a, b) => rank(a.band) - rank(b.band) || Math.abs(a.gap) - Math.abs(b.gap))
   405→    .slice(0, limit);
   406→}

   423→  samples: number;
   424→  byBand: { PASS: number; BOUNDARY: number; REJECT: number };

   426→  confirmReasons: GateReasonAgg[];
   427→  opposeReasons: GateReasonAgg[];
   428→  thrModifiers: GateThrAgg[];

   433→  // the block ring caps at BLOCK_CAP; with heavy churn (~8-11 blocks/min in chop)
   434→  // it holds only ~60-90 min of accepted records, so per-gate counts are a
   435→  // RECENT-CHURN SNAPSHOT (and dedup-topology-skewed: single-key gates like
   436→  // MEMORY compress, multi-symbol gates like REGIME fan out) — never a true
   437→  // window census. Consumers (AI core narrative) MUST disclose this.
   438→  blocksRingSize: number;
   439→  blocksRingSaturated: boolean;

   442→export function gateStats(windowHours = 6): GateStats {
   443→  const since = Date.now() - windowHours * 3600_000;
   444→  const win = samples.filter((s) => s.at >= since);
   445→  const byBand = { PASS: 0, BOUNDARY: 0, REJECT: 0 };
   446→  for (const s of win) byBand[s.band]++;
   447→
   448→  const agg = (list: GateSample[], pick: (s: GateSample) => GateFactor[]): Map<string, { n: number; sum: number; last: string | null }> => {
   449→    const m = new Map<string, { n: number; sum: number; last: string | null }>();
   450→    for (const s of list)
   451→      for (const f of pick(s)) {
   452→        const e = m.get(f.factor) ?? { n: 0, sum: 0, last: null };
   453→        e.n++;
   454→        e.sum += f.vote;
   455→        if (f.detail) e.last = f.detail;
   456→        m.set(f.factor, e);
   457→      }
   458→    return m;
   459→  };
   460→  const toAgg = (m: Map<string, { n: number; sum: number; last: string | null }>): GateReasonAgg[] =>
   461→    [...m.entries()]
   462→      .map(([factor, e]) => ({ factor, count: e.n, avgVote: Number((e.sum / e.n).toFixed(2)), lastDetail: e.last }))
   463→      .sort((a, b) => b.count - a.count);
   464→
   465→  const confirmAgg = toAgg(agg(win, (s) => s.confirms));
   466→  const opposeAgg = toAgg(agg(win, (s) => s.opposes));
   467→
   468→  const thrNames: Array<keyof ThrBreakdown> = ["regime", "spill", "chase", "mtf", "liq", "lesson"];
   469→  const thrModifiers: GateThrAgg[] = thrNames
   470→    .map((name) => {
   471→      const vals = win.map((s) => s.thr[name]).filter((v): v is number => v !== undefined && v !== 0);
   472→      return { name, count: vals.length, avg: vals.length ? Number((vals.reduce((a, b) => a + b, 0) / vals.length).toFixed(2)) : 0 };
   473→    })
   474→    .filter((t) => t.count > 0);
   475→
   476→  const blockMap = new Map<GateName, { count: number; lastReason: string }>();
   477→  for (const b of blocks) {
   478→    if (b.at < since) continue;
   479→    const e = blockMap.get(b.gate) ?? { count: 0, lastReason: b.reason };
   480→    e.count++;
   481→    e.lastReason = b.reason;
   482→    blockMap.set(b.gate, e);
   483→  }
   484→
   485→  const deploysWin = deploys.filter((d) => d.at >= since);
   486→  const boundaryN = byBand.BOUNDARY;
   487→  return {
   488→    windowHours,
   489→    samples: win.length,
   490→    byBand,
   491→    boundary:
   492→      boundaryN > 0
   493→        ? {
   494→            avgGap: Number((win.filter((s) => s.band === "BOUNDARY").reduce((a, s) => a + s.gap, 0) / boundaryN).toFixed(2)),
   495→            worstGap: Math.max(...win.filter((s) => s.band === "BOUNDARY").map((s) => s.gap)),
   496→            sharePct: Number(((boundaryN / Math.max(1, win.length)) * 100).toFixed(1)),
   497→          }
   498→        : null,
   499→    confirmReasons: confirmAgg.slice(0, 6),
   500→    opposeReasons: opposeAgg.slice(0, 6),
   501→    thrModifiers,
   502→    blocks: [...blockMap.entries()].map(([gate, e]) => ({ gate, count: e.count, lastReason: e.lastReason })).sort((a, b) => b.count - a.count),
   503→    deploys: deploysWin.length,
   504→    avgDeployScore: deploysWin.length ? Number((deploysWin.reduce((a, d) => a + Math.abs(d.score), 0) / deploysWin.length).toFixed(1)) : null,
   505→    blocksRingSize: blocks.length,
   506→    blocksRingSaturated: blocks.length >= BLOCK_CAP,
   507→  };
   508→}

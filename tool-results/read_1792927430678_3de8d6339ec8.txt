   258→async function pubGet<T>(path: string, params: Record<string, string | number> = {}): Promise<T | null> {
   259→  try {
   260→    const qs = new URLSearchParams(Object.entries(params).map(([k, v]) => [k, String(v)])).toString();
   261→    const r = await fetch(`${BASE}${path}${qs ? `?${qs}` : ""}`, { signal: AbortSignal.timeout(6500) });
   262→    if (!r.ok) return null;
   263→    return (await r.json()) as T;
   264→  } catch { return null; }
   265→}

   267→/** Signed v1 request — ALL params in the form body (query string empty),
   268→    signature = HMAC_SHA256(secret, totalParams) appended as body param. */
   269→async function signed<T>(method: "GET" | "POST", path: string, params: Record<string, string | number> = {}): Promise<{ ok: boolean; status: number; data: T | null; err: string | null }> {
   270→  if (!API_KEY || !API_SECRET) return { ok: false, status: 0, data: null, err: "no API credentials" };
   271→  const full: Record<string, string> = {};
   272→  for (const [k, v] of Object.entries(params)) full[k] = String(v);
   273→  full.timestamp = String(Date.now() + clockOffset);
   274→  full.recvWindow = "5000";
   275→  const qs = Object.entries(full).map(([k, v]) => `${k}=${encodeURIComponent(v)}`).join("&");
   276→  const sig = hmac(API_SECRET, qs);
   277→  try {
   278→    const init: RequestInit = method === "GET"
   279→      ? { signal: AbortSignal.timeout(8000) }
   280→      : { method: "POST", headers: { "Content-Type": "application/x-www-form-urlencoded", "X-BB-APIKEY": API_KEY }, body: `${qs}&signature=${sig}`, signal: AbortSignal.timeout(8000) };
   281→    if (method === "GET") (init as Record<string, unknown>).headers = { "X-BB-APIKEY": API_KEY };
   282→    const url = method === "GET" ? `${BASE}${path}?${qs}&signature=${sig}` : `${BASE}${path}`;
   283→    const r = await fetch(url, init);
   284→    const text = await r.text();
   285→    let data: unknown = null;
   286→    try { data = JSON.parse(text); } catch { /* non-json */ }
   287→    return { ok: r.ok, status: r.status, data: (data ?? null) as T | null, err: r.ok ? null : text.slice(0, 240) };
   288→  } catch (e) {
   289→    return { ok: false, status: 0, data: null, err: String(e) };
   290→  }
   291→}

   469→async function doOpen(it: Intent): Promise<void> {
   470→  const liveMode = isLiveMode();
   471→  it.mode = liveMode ? "LIVE" : "PAPER";
   472→  const tick = await tbTicker(it.tbSymbol);
   473→  if (!tick) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_TICKER", note: it.tbSymbol }); audit("EXEC-FAIL", { serial: it.serial, why: "no toobit ticker" }); return; }
   474→  const instr = instruments.get(it.tbSymbol);
   475→  if (!instr) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_INSTRUMENT" }); audit("EXEC-FAIL", { serial: it.serial, why: `symbol absent on toobit: ${it.tbSymbol}` }); return; }
   476→  const refPrice = it.direction === "LONG" ? tick.ask : tick.bid;
   477→  // v1.1.0 FIXED-$5 MARGIN (margin never varies per signal) + v1.2.0 PER-CARD
   478→  // LEVERAGE (operator mandate trace 1a15ddbdbabf84a9): notional = fixed margin
   479→  // × THE CARD'S leverage — size and risk mirror the signal card exactly.
   480→  const notional = S.cfg.marginUsdt * it.leverage;
   481→  it.expectedNotional = Number(notional.toFixed(6));
   482→  let qty = notional / refPrice;
   483→
   484→  // v1.3.0 ENTRY GATES (operator mandate trace 1a15f7d945d27ead):
   485→  // ① ENTRY-DELAY — فاصله زمانی با تأخیر: a card whose entry window has passed
   486→  //    is no longer the signal the card described; chasing it trades a stale edge.
   487→  const ageAtEntry = Date.now() - (it.createdAt0 ?? Date.now());
   488→  it.entryAgeMs = ageAtEntry;
   489→  if (ageAtEntry > S.cfg.maxEntryAgeMs) {
   490→    it.status = "SKIP";
   491→    it.events.push({ at: Date.now(), ev: "ENTRY-DELAY", note: `age=${Math.round(ageAtEntry / 1000)}s > cap ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s` });
   492→    countSkip("ENTRY-DELAY");
   493→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-delay: card age ${Math.round(ageAtEntry / 1000)}s exceeds ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s window` });
   494→    return;
   495→  }
   496→  // ② ENTRY-DRIFT — اختلاف قیمتی بالا: if the book has already run away from
   497→  //    the card's entry, the fill is not the card's trade — skip honestly.
   498→  const driftBps = Math.abs(refPrice - it.cardEntry) / it.cardEntry * 10_000;
   499→  it.entryDriftBps = Number(driftBps.toFixed(2));
   500→  if (it.cardEntry > 0 && driftBps > S.cfg.maxEntryDriftBps) {
   501→    it.status = "SKIP";
   502→    it.events.push({ at: Date.now(), ev: "ENTRY-DRIFT", note: `drift=${driftBps.toFixed(0)}bps > cap ${S.cfg.maxEntryDriftBps}bps (ref=${refPrice} vs card=${it.cardEntry})` });
   503→    countSkip("ENTRY-DRIFT");
   504→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-drift: ${driftBps.toFixed(0)}bps from card entry exceeds ${S.cfg.maxEntryDriftBps}bps cap` });
   505→    return;
   506→  }
   507→  // $5-fixed mandate: a symbol whose exchange floor (minNotional / minQty×price)
   508→  // cannot fit inside the mandated margin notional is SKIPPED honestly —
   509→  // silently upsizing would breach the fixed-margin contract.
   510→  const floorNotional = Math.max(instr.minNotional, instr.minQty * refPrice, S.floorLearn?.[it.tbSymbol] ?? 0);
   511→  if (floorNotional > notional) {
   512→    it.status = "SKIP";
   513→    it.events.push({ at: Date.now(), ev: "MIN_FLOOR_SKIP", note: `${it.tbSymbol} floor=${floorNotional.toFixed(2)} > marginNotional=${notional}${S.floorLearn?.[it.tbSymbol] ? " (learned)" : ""} ` });
   514→    countSkip("MIN-FLOOR");
   515→    audit("EXEC-SKIP", { serial: it.serial, why: `exchange floor ${floorNotional.toFixed(2)} USDT exceeds fixed-margin notional ${notional} USDT` });
   516→    return;
   517→  }
   518→  qty = floorStep(Math.min(qty, instr.maxQty), instr.stepSize);
   519→  if (qty < instr.minQty) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "QTY_BELOW_MIN" }); audit("EXEC-FAIL", { serial: it.serial, why: `qty ${qty} < minQty ${instr.minQty}` }); return; }
   520→

   535→        S.floorLearn[it.tbSymbol] = Math.max(prevFloor, notional);
   536→        it.status = "SKIP";

   547→  } else {

   549→  }
   550→  const tFill = Date.now();
   551→  it.status = "OPEN";
   552→  it.qtyOrig = qty;
   553→  it.qtyRemaining = qty;
   554→  it.orderId = orderId;
   555→  it.fillPrice = fillPrice;
   556→  it.openedAt = tFill;

   564→  if (ndelta > 2000) doctrineViolation(it.serial, "NOTIONAL-DELTA", `exp=${it.expectedNotional} act=${it.actualNotional} (${ndelta.toFixed(0)}bps) — step-rounding or minQty floor`);

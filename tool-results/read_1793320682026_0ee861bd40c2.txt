   659→async function drainQ(): Promise<void> {
   660→  if (qBusy) return;
   661→  qBusy = true;
   662→  try {
   663→    while (actionQ.length > 0) {
   664→      const a = actionQ.shift()!;
   665→      const it = S.intents[a.cardId];
   666→      if (!it) continue;

   670→        else await doClose(it, a.reason);
   671→      } catch (e) {
   672→        audit("EXEC-ERROR", { cardId: a.cardId, kind: a.kind, err: String(e) });
   673→      }
   674→      persistState();
   675→      flushState();
   676→      await new Promise((r) => setTimeout(r, 250)); // pacing — 60 orders / 2s limit
   677→    }
   678→  } finally { qBusy = false; }

   681→// v1.4.4 RISK-CAP — shared sizing truth for doOpen + doctrineScan: notional is
   682→// the v1.1.0 mandate ceiling (margin × cardLev), scaled DOWN when
   683→// notional × stopDistance would exceed maxRiskPct% of WALLET equity
   684→// (trace 1a17357988d00515 «حساب لایو داره از بین میره»). Never scaled up.
   685→function riskCappedNotional(lev: number, slDistPct: number, liveMode: boolean): { base: number; eff: number; capped: boolean } {
   686→  const base = S.cfg.marginUsdt * lev;
   687→  if (!liveMode) return { base, eff: base, capped: false };
   688→  const equity = walletUsdt ?? balanceUsdt ?? 0;
   689→  const maxRiskUsdt = equity * (S.cfg.maxRiskPct / 100);
   690→  if (maxRiskUsdt > 0 && base * slDistPct > maxRiskUsdt) {
   691→    return { base, eff: Math.floor((maxRiskUsdt / slDistPct) * 100) / 100, capped: true };
   692→  }
   693→  return { base, eff: base, capped: false };
   694→}

   696→async function doOpen(it: Intent): Promise<void> {
   697→  const liveMode = isLiveMode();
   698→  it.mode = liveMode ? "LIVE" : "PAPER";
   699→  const tick = await tbTicker(it.tbSymbol);
   700→  if (!tick) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_TICKER", note: it.tbSymbol }); audit("EXEC-FAIL", { serial: it.serial, why: "no toobit ticker" }); return; }
   701→  const instr = instruments.get(it.tbSymbol);
   702→  if (!instr) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "NO_INSTRUMENT" }); audit("EXEC-FAIL", { serial: it.serial, why: `symbol absent on toobit: ${it.tbSymbol}` }); return; }
   703→  const refPrice = it.direction === "LONG" ? tick.ask : tick.bid;
   704→  // v1.1.0 FIXED-$5 MARGIN (margin never varies per signal) + v1.2.0 PER-CARD
   705→  // LEVERAGE (operator mandate trace 1a15ddbdbabf84a9): notional = fixed margin
   706→  // × THE CARD'S leverage — size and risk mirror the signal card exactly.
   707→  // v1.4.3 REAL SIZING — qty is in CONTRACTS; the real notional the book sees is
   708→  // qty × price × contractMultiplier (0.1 on SUI/FARTCOIN …). Pre-1.4.3 the
   709→  // multiplier was ignored → NX-3126/NX-3140 opened at $0.5 margin instead of $5.
   710→  const mult = instr.mult > 0 ? instr.mult : 1;
   711→  it.contractMult = mult;
   712→  it.sizingRev = 4;
   713→  // v1.4.4 RISK-CAP — equity-aware sizing (trace 1a17357988d00515
   714→  // «حساب لایو داره از بین میره»): the card's stop distance defines the real
   715→  // money at risk = notional × slDistPct; on a small LIVE account a ~5% stop
   716→  // on a $100 notional burns ~14% of equity in ONE SL hit. The per-trade risk
   717→  // budget is capped at maxRiskPct% of wallet equity by scaling notional DOWN.
   718→  const slDistPct = it.stopLoss > 0 && it.cardEntry > 0
   719→    ? Math.abs(it.cardEntry - it.stopLoss) / it.cardEntry
   720→    : 0.05; // no usable stop on the card → conservative 5% assumption
   721→  const cap = riskCappedNotional(it.leverage, slDistPct, liveMode);
   722→  const notional = cap.eff;
   723→  const effMargin = notional / it.leverage;
   724→  if (cap.capped) {
   725→    it.events.push({ at: Date.now(), ev: "RISK-CAP", note: `notional ${cap.base.toFixed(2)}→${notional.toFixed(2)}$ (SLdist=${(slDistPct * 100).toFixed(2)}% × wallet ${(walletUsdt ?? balanceUsdt ?? 0).toFixed(2)}$ × cap ${S.cfg.maxRiskPct}%)` });
   726→    audit("EXEC-RISK-CAP", { serial: it.serial, baseNotional: Number(cap.base.toFixed(4)), cappedNotional: Number(notional.toFixed(4)), slDistPct: Number((slDistPct * 100).toFixed(3)), equity: Number((walletUsdt ?? balanceUsdt ?? 0).toFixed(4)), maxRiskPct: S.cfg.maxRiskPct });
   727→  }
   728→  it.expectedNotional = Number(notional.toFixed(6));
   729→  let qty = notional / (refPrice * mult);
   730→
   731→  // v1.3.0 ENTRY GATES (operator mandate trace 1a15f7d945d27ead):
   732→  // ① ENTRY-DELAY — فاصله زمانی با تأخیر: a card whose entry window has passed
   733→  //    is no longer the signal the card described; chasing it trades a stale edge.
   734→  const ageAtEntry = Date.now() - (it.createdAt0 ?? Date.now());
   735→  it.entryAgeMs = ageAtEntry;
   736→  if (ageAtEntry > S.cfg.maxEntryAgeMs) {
   737→    it.status = "SKIP";
   738→    it.events.push({ at: Date.now(), ev: "ENTRY-DELAY", note: `age=${Math.round(ageAtEntry / 1000)}s > cap ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s` });
   739→    countSkip("ENTRY-DELAY");
   740→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-delay: card age ${Math.round(ageAtEntry / 1000)}s exceeds ${Math.round(S.cfg.maxEntryAgeMs / 1000)}s window` });
   741→    return;
   742→  }
   743→  // ② ENTRY-DRIFT — اختلاف قیمتی بالا: if the book has already run away from
   744→  //    the card's entry, the fill is not the card's trade — skip honestly.
   745→  const driftBps = Math.abs(refPrice - it.cardEntry) / it.cardEntry * 10_000;
   746→  it.entryDriftBps = Number(driftBps.toFixed(2));
   747→  if (it.cardEntry > 0 && driftBps > S.cfg.maxEntryDriftBps) {
   748→    it.status = "SKIP";
   749→    it.events.push({ at: Date.now(), ev: "ENTRY-DRIFT", note: `drift=${driftBps.toFixed(0)}bps > cap ${S.cfg.maxEntryDriftBps}bps (ref=${refPrice} vs card=${it.cardEntry})` });
   750→    countSkip("ENTRY-DRIFT");
   751→    audit("EXEC-SKIP", { serial: it.serial, why: `entry-drift: ${driftBps.toFixed(0)}bps from card entry exceeds ${S.cfg.maxEntryDriftBps}bps cap` });
   752→    return;
   753→  }
   754→  // ③ BALANCE-CAP (v1.4.3) — the fixed margin must fit the LIVE balance; when
   755→  // the account cannot fund another isolated position, skip honestly instead of
   756→  // throwing an insufficient-margin order at the book.
   757→  if (liveMode) {
   758→    await refreshBalance();
   759→    const avail = balanceUsdt ?? 0;
   760→    if (avail < effMargin) {
   761→      it.status = "SKIP";
   762→      it.events.push({ at: Date.now(), ev: "BALANCE-CAP", note: `margin ${effMargin.toFixed(2)}$ > available ${avail.toFixed(2)}$` });
   763→      countSkip("BALANCE-CAP");
   764→      audit("EXEC-SKIP", { serial: it.serial, why: `balance-cap: margin ${effMargin.toFixed(2)} USDT exceeds available ${avail.toFixed(2)} USDT` });
   765→      return;
   766→    }
   767→    // v1.4.4 — a risk-capped margin below $1 is exchange dust; skip honestly.
   768→    if (cap.capped && effMargin < 1) {
   769→      it.status = "SKIP";
   770→      it.events.push({ at: Date.now(), ev: "RISK-CAP-DUST", note: `margin ${effMargin.toFixed(2)}$ < $1 after risk cap` });
   771→      countSkip("RISK-CAP");
   772→      audit("EXEC-SKIP", { serial: it.serial, why: `risk-cap dust: margin ${effMargin.toFixed(2)} USDT below $1 floor` });
   773→      return;
   774→    }
   775→  }
   776→  // $5-fixed mandate: a symbol whose exchange floor (minNotional / minQty×price)
   777→  // cannot fit inside the mandated margin notional is SKIPPED honestly —
   778→  // silently upsizing would breach the fixed-margin contract.
   779→  // v1.4.3 — floors are REAL notional (multiplier-aware).
   780→  const floorNotional = Math.max(instr.minNotional, instr.minQty * refPrice * mult, S.floorLearn?.[it.tbSymbol] ?? 0);
   781→  if (floorNotional > notional) {
   782→    it.status = "SKIP";
   783→    it.events.push({ at: Date.now(), ev: "MIN_FLOOR_SKIP", note: `${it.tbSymbol} floor=${floorNotional.toFixed(2)} > marginNotional=${notional}${S.floorLearn?.[it.tbSymbol] ? " (learned)" : ""} ` });
   784→    countSkip("MIN-FLOOR");
   785→    audit("EXEC-SKIP", { serial: it.serial, why: `exchange floor ${floorNotional.toFixed(2)} USDT exceeds fixed-margin notional ${notional} USDT` });
   786→    return;
   787→  }
   788→  qty = floorStep(Math.min(qty, instr.maxQty), instr.stepSize);
   789→  if (qty < instr.minQty) { it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "QTY_BELOW_MIN" }); audit("EXEC-FAIL", { serial: it.serial, why: `qty ${qty} < minQty ${instr.minQty}` }); return; }
   790→
   791→  const tSent = Date.now();
   792→  let fillPrice: number | null = null;
   793→  let orderId: string | null = null;
   794→  if (liveMode) {
   795→    await ensureLeverage(it.tbSymbol, it.leverage);
   796→    const side = it.direction === "LONG" ? "BUY_OPEN" : "SELL_OPEN";
   797→    const r = await placeOrder(it.tbSymbol, side, qty, { takeProfit: it.tp3, stopLoss: it.stopLoss, clientOrderId: it.clientOrderId });
   798→    audit("EXEC-OPEN", { serial: it.serial, symbol: it.tbSymbol, side, qty, lev: it.leverage, tp3: it.tp3, sl: it.stopLoss, ok: r.ok, orderId: r.orderId, err: r.err });
   799→    if (!r.ok) {
   800→      // v1.4.2 — server-side real floor: exchangeInfo hides it (minNotional=0)
   801→      // but the book rejects with -1199/-1202 "quantity too small". Teach the
   802→      // floor, reclassify as an honest MIN-FLOOR SKIP, keep the ledger truthful.
   803→      if (/quantity too small/i.test(r.err ?? "")) {
   804→        const prevFloor = S.floorLearn?.[it.tbSymbol] ?? 0;
   805→        S.floorLearn[it.tbSymbol] = Math.max(prevFloor, notional);
   806→        it.status = "SKIP";
   807→        it.events.push({ at: Date.now(), ev: "MIN_FLOOR_LEARNED", note: `${it.tbSymbol} real floor ≥ ${notional} USDT (server -1199/-1202; exchangeInfo said ${floorNotional.toFixed(2)})` });
   808→        countSkip("MIN-FLOOR");
   809→        audit("EXEC-SKIP", { serial: it.serial, why: `server floor learned: ${it.tbSymbol} ≥ ${notional} USDT notional (rejected quantity-too-small; exchangeInfo floor ${floorNotional.toFixed(2)})` });
   810→        persistState();
   811→        return;
   812→      }
   813→      it.status = "FAILED"; it.events.push({ at: Date.now(), ev: "OPEN_REJECTED", note: (r.err ?? "").slice(0, 160) }); return;
   814→    }
   815→    orderId = r.orderId;
   816→    fillPrice = await resolveFill(it.tbSymbol, orderId, refPrice);
   817→  } else {
   818→    fillPrice = refPrice; // paper fill on the real toobit book (taker side)
   819→  }
   820→  const tFill = Date.now();
   821→  it.status = "OPEN";
   822→  it.qtyOrig = qty;
   823→  it.qtyRemaining = qty;
   824→  it.orderId = orderId;
   825→  it.fillPrice = fillPrice;
   826→  it.openedAt = tFill;
   827→  // v1.3.0 doctrine bookkeeping — the truth of what the book actually saw:
   828→  // step-size floor rounding makes actualNotional ≤ expectedNotional by design.
   829→  // v1.4.3 — actual notional is REAL: qty(contracts) × fill × contractMultiplier.
   830→  it.actualNotional = fillPrice ? Number((qty * fillPrice * mult).toFixed(6)) : null;
   831→  it.protected = it.mode === "LIVE" ? true : true; // LIVE: attached TP/SL in same order + recon re-check; PAPER: sim active
   832→  S.doctrine.opens++;
   833→  const ndelta = it.expectedNotional && it.actualNotional ? Math.abs(it.actualNotional - it.expectedNotional) / it.expectedNotional * 10_000 : 0;
   834→  audit("EXEC-OPEN-DOCTRINE", { serial: it.serial, lev: it.leverage, levSrc: it.levSrc ?? "card", margin: Number(effMargin.toFixed(4)), riskCapped: cap.capped, contractMult: mult, expectedNotional: it.expectedNotional, actualNotional: it.actualNotional, notionalDeltaBps: Number(ndelta.toFixed(1)), entryAgeMs: it.entryAgeMs, entryDriftBps: it.entryDriftBps });
   835→  if (ndelta > 2000) doctrineViolation(it.serial, "NOTIONAL-DELTA", `exp=${it.expectedNotional} act=${it.actualNotional} (${ndelta.toFixed(0)}bps) — step-rounding or minQty floor`);
   836→  it.lat = {
   837→    detectMs: it.lat.detectMs ?? null,
   838→    execMs: tFill - tSent,
   839→    totalMs: it.createdAt0 !== undefined ? tFill - it.createdAt0 : null,
   840→    slipBps: fillPrice && it.cardEntry > 0 ? ((it.direction === "LONG" ? fillPrice - it.cardEntry : it.cardEntry - fillPrice) / it.cardEntry) * 10_000 : null,
   841→  };
   842→  it.events.push({ at: tFill, ev: "OPEN", note: `${it.mode} qty=${qty} mult=×${mult} fill=${fillPrice}` });
   843→  S.metrics.push({ serial: it.serial, symbol: it.symbol, direction: it.direction, mode: it.mode, detectMs: it.lat.detectMs, execMs: it.lat.execMs, totalMs: it.lat.totalMs, slipBps: it.lat.slipBps, fillPrice, cardEntry: it.cardEntry, at: tFill });
   844→  if (S.metrics.length > METRICS_WINDOW) S.metrics = S.metrics.slice(-METRICS_WINDOW);
   845→}

   848→async function resolveFill(tbSymbol: string, orderId: string | null, fallback: number): Promise<number> {
   849→  if (orderId) {
   850→    for (let i = 0; i < 3; i++) {
   851→      await new Promise((r) => setTimeout(r, 350));
   852→      const r2 = await signed<{ status?: string; orderStatus?: string; avgPrice?: string; price?: string; executedQty?: string; cumQty?: string }>("GET", "/api/v1/futures/order", { orderId });
   853→      const d = r2.data as Record<string, unknown> | null;
   854→      const st = String(d?.status ?? d?.orderStatus ?? "").toUpperCase();
   855→      const avg = Number(d?.avgPrice ?? d?.price ?? 0);
   856→      if (avg > 0) return avg;
   857→      if (/CANCELED|REJECTED|EXPIRED/.test(st)) break;
   858→      if (st === "FILLED" || i === 2) break;
   859→    }
   860→  }
   861→  return fallback;

   864→// v1.4.3 LIVE PnL TELEMETRY — exchange-verified realized PnL for an intent's
   865→// lifetime (fills since openedAt; realizedPnL arrives on closing fills).
   866→// This is the money-truth behind «چرا هیچ سیگنالی در سود بسته نشده».
   867→async function fetchRealizedPnl(it: Intent): Promise<number | null> {
   868→  try {
   869→    // v1.4.5 — a NEVER-EXECUTED intent (no openedAt) owns ZERO exchange fills;
   870→    // with since=0 the symbol's whole fill history was once re-summed into it
   871→    // (phantom PnL: ENA -0.7445 = morning -0.6893-0.0552, BTW -3.2605, PEPE -0.1066).
   872→    if (!it.openedAt) return null;
   873→    const r = await signed<Array<{ time?: string; realizedPnl?: string; commission?: string; makerRebate?: string }>>("GET", "/api/v1/futures/userTrades", { symbol: it.tbSymbol, limit: 100 });
   874→    const rows = Array.isArray(r.data) ? r.data : [];
   875→    const since = it.openedAt;
   876→    let pnl = 0, seen = false;
   877→    for (const f of rows) {
   878→      if (Number(f.time ?? 0) < since) continue;
   879→      seen = true;
   880→      pnl += Number(f.realizedPnl ?? 0) - Number(f.commission ?? 0) + Number(f.makerRebate ?? 0);
   881→    }
   882→    if (!seen) return null;
   883→    return Number(pnl.toFixed(4));
   884→  } catch { return null; }
   885→}
   886→// revert the card-status transition so the next poll re-fires the failed
   887→// action (v1.4.3 — transitions are no longer one-shot on transient rejections;
   888→// capped so a permanently-rejected action cannot loop forever).
   889→function revertTransition(it: Intent, what: string, err: string): void {
   890→  const attempts = (it.actionAttempts ?? 0) + 1;
   891→  it.actionAttempts = attempts;
   892→  if (attempts <= 20 && it.prevStatusSeen) {
   893→    it.lastStatusSeen = it.prevStatusSeen;
   894→    it.events.push({ at: Date.now(), ev: `${what}_RETRY-ARMED`, note: err.slice(0, 140) });
   895→    audit("EXEC-RETRY-ARMED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   896→  } else {
   897→    it.events.push({ at: Date.now(), ev: `${what}_ABANDONED`, note: `attempts=${attempts} — will not retry: ${err.slice(0, 100)}` });
   898→    audit("EXEC-RETRY-ABANDONED", { serial: it.serial, what, attempts, err: err.slice(0, 140) });
   899→  }
   900→  persistState();
   901→}

   903→async function doBank(it: Intent, fracOfOrig: number, reason: string): Promise<void> {
   904→  if (it.status !== "OPEN" || it.qtyRemaining <= 0) return;
   905→  let qty = floorStep(Math.min(it.qtyOrig * fracOfOrig, it.qtyRemaining), instruments.get(it.tbSymbol)?.stepSize ?? 0);
   906→  if (qty <= 0) return;
   907→  if (qty >= it.qtyRemaining) return doClose(it, reason);
   908→  if (it.mode === "LIVE") {
   909→    const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   910→    // v1.4.3 — clientOrderId is mandatory on Toobit (pre-1.4.3 every bank was
   911→    // rejected -1004 → no signal ever closed in profit).
   912→    const r = await placeOrder(it.tbSymbol, side, qty, { clientOrderId: `NX${it.serial}${reason}${Date.now() % 100000}` });
   913→    audit("EXEC-BANK", { serial: it.serial, reason, qty, ok: r.ok, err: r.err });
   914→    if (!r.ok) { it.events.push({ at: Date.now(), ev: "BANK_REJECTED", note: (r.err ?? "").slice(0, 160) }); revertTransition(it, "BANK", r.err ?? "bank order rejected"); return; }
   915→  } else {
   916→    // v1.4.0 — PAPER سیو سود was invisible in audit.jsonl (only in state.json);
   917→    // the paper fill rides the REAL toobit book so it deserves the same trail.
   918→    // v1.5.0 — paper money truth: the banked slice is priced at the LIVE taker
   919→    // side and books REAL PnL into the intent + fixed-$3 account.
   920→    const tick = await tbTicker(it.tbSymbol);
   921→    const exit = tick ? (it.direction === "LONG" ? tick.bid : tick.ask) : null;
   922→    const entry = it.fillPrice ?? it.cardEntry;
   923→    const sign = it.direction === "LONG" ? 1 : -1;
   924→    const mult = it.contractMult ?? instruments.get(it.tbSymbol)?.mult ?? 1;
   925→    const pnl = exit !== null && entry > 0 ? (exit - entry) * qty * sign * mult : 0;
   926→    it.realizedPnlUsdt = Number(((it.realizedPnlUsdt ?? 0) + pnl).toFixed(6));
   927→    audit("EXEC-BANK-PAPER", { serial: it.serial, reason, qty, exit: exit ?? null, pnl: Number(pnl.toFixed(6)), note: "paper fill on real book (v1.5.0 priced)" });
   928→    if (pnl !== 0) paperBook(it, pnl, `${reason} bank`);
   929→  }
   930→  it.qtyRemaining = Number((it.qtyRemaining - qty).toFixed(10));
   931→  if (reason === "TP1") it.qtyBankedTp1 = qty;
   932→  if (reason === "TP2") it.qtyBankedTp2 = qty;
   933→  it.actionAttempts = 0;
   934→  it.events.push({ at: Date.now(), ev: reason, note: `banked ${qty} (سیو سود)` });
   935→}

   937→async function doClose(it: Intent, reason: string): Promise<void> {
   938→  if (it.status === "CLOSED" || it.status === "CLOSING") return;
   939→  it.status = "CLOSING";
   940→  if (it.mode === "LIVE" && it.qtyRemaining > 0) {
   941→    const flash = await flashClose(it.tbSymbol, it.direction);
   942→    audit("EXEC-CLOSE", { serial: it.serial, reason, flash: flash.ok, err: flash.err });
   943→    if (!flash.ok) {
   944→      // fallback: explicit MARKET close of the remaining qty
   945→      const side = it.direction === "LONG" ? "SELL_CLOSE" : "BUY_CLOSE";
   946→      // v1.4.3 — clientOrderId mandatory (pre-1.4.3 the SL-close fallback was
   947→      // rejected -1004 and the transition was lost forever).

   987→    // no exchange fills: closing them must NOT sum the symbol's fill history.

   996→    }
   997→  }
   998→  it.events.push({ at: it.closedAt, ev: "CLOSED", note: reason });
   999→  persistState();

  1281→  try {
  1282→    const stat = statSync(AUDIT);
  1283→    const READ = 256 * 1024;
  1284→    const start = Math.max(0, stat.size - READ);
  1285→    const fh = openSync(AUDIT, "r");
  1286→    const len = stat.size - start;
  1287→    const buf = Buffer.alloc(len);
  1288→    readSync(fh, buf, 0, len, start);
  1289→    closeSync(fh);
  1290→    const lines = buf.toString("utf8").split("\n").filter((l) => l.trim().length > 0);
  1291→    const rows: Array<Record<string, unknown>> = [];
  1292→    for (let i = lines.length - 1; i >= 0 && rows.length < maxRows; i--) {
  1293→      try { rows.push(JSON.parse(lines[i]!) as Record<string, unknown>); } catch { /* torn/partial line — skip */ }
  1294→    }
  1295→    return rows;
  1296→  } catch { return []; }

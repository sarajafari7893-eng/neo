   296→function ingestVerdict(v: ShadowVerdict): boolean {
   297→  if (!v || typeof v.id !== "string") return false;

   300→  // outcome sweep closes it, the orphan sweep stamps it). First-record-wins
   301→  // hydration silently DROPPED those upgrades — every restart reverted linked-
   302→  // but-unsettled verdicts back to unlinked (the real-veto 17→12 regression
   303→  // after the 2026-09-21 restarts). Duplicate ids now MERGE their upgrade
   304→  // fields into the already-ingested record, order-independent.
   305→  if (seenIds.has(v.id)) {
   306→    const ex = verdicts.find((x) => x.id === v.id);
   307→    if (ex) {
   308→      if (ex.serial === null && v.serial !== null) ex.serial = v.serial;
   309→      if (!ex.outcome && v.outcome) ex.outcome = v.outcome;
   310→      if (ex.sweptAt === undefined && v.sweptAt !== undefined) ex.sweptAt = v.sweptAt;
   311→    }
   312→    archiveVerdict(v, true); // TASK-40: upgrade records also update the archive
   313→    return false;
   314→  }
   315→  v.audit = v.audit ?? null; // normalize legacy records
   316→  // collapse cross-instance duplicate adjudications of the same candidate —
   317→  // scan backwards to the candidate's latest record; inside the window = dupe
   318→  for (let i = verdicts.length - 1; i >= 0; i--) {
   319→    const x = verdicts[i]!;
   320→    if (x.symbol === v.symbol && x.direction === v.direction) {
   321→      if (Math.abs(x.at - v.at) < INGEST_COLLAPSE_MS) return false;
   322→      break;
   323→    }
   324→  }
   325→  seenIds.add(v.id);
   326→  verdicts.push(v);
   327→  archiveVerdict(v); // TASK-40: every ingested verdict is archived (trim-proof)
   328→  trimVerdicts(); // Task 34-2: evidence-safety eviction (was: blind oldest-first splice)
   329→  return true;

   416→async function hydrate(): Promise<void> {
   417→  if (hydrating) return hydrating;
   418→  hydrating = (async () => {
   419→    try {
   420→    // TASK-40 (2026-09-27, operator bug «بازوها پر میشوند ولی مجدد داده‌ها از
   421→    // دست میرود و کاهش پیدا می‌کند»): hydrate previously ingested ONLY the last
   422→    // 3000 journal lines. With the live journal at ~5.1k lines, every restart
   423→    // dropped the older verdicts — and crucially the OUTCOME records ("o"/"pc")
   424→    // whose base record had scrolled out of the window — so the REAL arms
   425→    // collapsed after each boot (observed live: real-keep 30→2, real-veto
   426→    // 30→1 after the 09-25..09-27 restart waves; offline replay of the FULL
   427→    // journal proves both arms hold a complete rolling-30 cohort). The journal
   428→    // is rotation-capped at 6MB, so FULL ingestion is bounded; the .old
   429→    // rotation ring (when present) is ingested FIRST so upgrade-merge
   430→    // (ingestVerdict) and outcome application (applyOutcomeRecord /
   431→    // applyPaperClose) see correct oldest→newest chronology.
   432→    // MEMORY (Task-39 lesson): the first cut did one readFile + one giant
   433→    // split — a multi-MB transient spike exactly during the boot+compile
   434→    // window, which pushed the 09:46 boot over the edge into a kernel
   435→    // OOM-kill. Ingestion is now STREAMED line-by-line (64KB readahead):
   436→    // peak memory = one journal line, event loop yields naturally between
   437→    // stream chunks, and evidence completeness is unchanged.
   438→    const ingestJournalLine = (line: string): void => {
   439→      if (!line) return;
   440→      try {
   441→        const rec = JSON.parse(line) as { k: string; v: unknown };
   442→        if (rec.k === "v") ingestVerdict(rec.v as ShadowVerdict);
   443→        else if (rec.k === "p") ingestPaper(rec.v as PaperTrade);
   444→        else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   445→        else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });
   446→      } catch {
   447→        /* torn line */
   448→      }
   449→    };
   450→    const ingestJournalStreamed = async (filePath: string): Promise<void> => {
   451→      try {
   452→        const rl = createInterface({ input: createReadStream(filePath, { encoding: "utf8" }), crlfDelay: Infinity });
   453→        for await (const line of rl) ingestJournalLine(line);
   454→      } catch {
   455→        /* missing ring (no rotation yet) or unreadable — cold for this ring only */
   456→      }
   457→    };
   458→    await ingestJournalStreamed(`${FILE}.old`); // rotation ring — OLDER records first
   459→    await ingestJournalStreamed(FILE);
   460→    if (verdicts.length || papers.length) {
   461→      // v8.2.1 — seq MUST derive from the MAX id on disk (length-derived seq
   462→      // reused ids when two module instances wrote with independent counters)
   463→      seq = Math.max(maxIdSeq(verdicts), maxIdSeq(papers));
   464→    }
   465→    const st = await stat(FILE).catch(() => null);
   466→    persisted = st?.size ?? 0;
   467→    readPos = st?.size ?? 0;
   468→    hydratedAt = Date.now();
   469→    // dedup continuity across instance swaps — a fresh instance must not
   470→    // instantly re-adjudicate what another instance just adjudicated
   471→    for (const v of verdicts) {
   472→      const key = `${v.symbol}|${v.direction}`;
   473→      const prev = lastAdjAt.get(key) ?? 0;
   474→      if (v.at > prev) lastAdjAt.set(key, v.at);
   475→    }
   476→    if ((st?.size ?? 0) > FILE_MAX_BYTES) {
   477→      // v10.6.1 persist-guard (operator APPROVE 2026-09-18): rotation is the
   478→      // destructive rewrite — if a fresher boot just rotated, resync our
   479→      // counters from disk instead of renaming over its rotation.
   480→      const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");
   481→      await serializePersist("shadow-veto-journal", async () => {
   482→        if (!(await persistGuardOk("shadow-veto-journal"))) {
   483→          const fresh = await stat(FILE).catch(() => null);
   484→          persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   485→          readPos = fresh?.size ?? 0;
   486→          return;
   487→        }
   488→        await rename(FILE, `${FILE}.old`).catch(() => undefined);
   489→        await persistWatermark("shadow-veto-journal");
   490→        persisted = 0;
   491→        readPos = 0;
   492→      });
   493→    }
   494→  } catch (e) {
   495→    // TASK-40: journaling must never break the engine, but a SILENT hydrate
   496→    // failure used to leave empty rings that later persists could not repair
   497→    // (shadow-veto only appends). Make the failure observable in dev.log.

   499→  }
   500→  })();

   504→async function persist(rec: { k: string; v: unknown }): Promise<void> {
   505→  try {
   506→    const line = JSON.stringify(rec) + "\n";
   507→    await appendFile(FILE, line, "utf8");
   508→    persisted += line.length;
   509→    readPos += line.length; // our own bytes are already in the rings
   510→    if (persisted > FILE_MAX_BYTES) {
   511→      // v10.6.1 persist-guard: serialize rotation + re-stat before renaming so
   512→      // concurrent writers (or a stale generation) never clobber a fresh file.
   513→      const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");
   514→      await serializePersist("shadow-veto-journal", async () => {
   515→        const cur = await stat(FILE).catch(() => null);
   516→        if ((cur?.size ?? 0) <= FILE_MAX_BYTES) {
   517→          // someone else already rotated — just resync our counters
   518→          persisted = Math.min(cur?.size ?? 0, FILE_MAX_BYTES);
   519→          readPos = cur?.size ?? 0;
   520→          return;
   521→        }
   522→        if (!(await persistGuardOk("shadow-veto-journal"))) {
   523→          const fresh = await stat(FILE).catch(() => null);
   524→          persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   525→          readPos = fresh?.size ?? 0;
   526→          return;
   527→        }
   528→        await rename(FILE, `${FILE}.old`).catch(() => undefined);
   529→        await persistWatermark("shadow-veto-journal");
   530→        persisted = 0;
   531→        readPos = 0;
   532→      });
   533→    }
   534→  } catch {
   535→    /* non-fatal */
   536→  }

   548→/** v8.2.1 READ-PATH LIVENESS — tail-ingest another instance's writes (see
   549→ * gate-journal.gateJournalRefresh for the full rationale). Throttled,
   550→ * single-flight, rotation-safe, bounded bite. */
   551→export async function shadowVetoRefresh(maxAgeMs = REFRESH_MIN_INTERVAL_MS): Promise<void> {
   552→  if (Date.now() - lastRefreshAt < maxAgeMs) return;
   553→  if (refreshing) return refreshing;
   554→  refreshing = (async () => {
   555→    try {
   556→      const st = await stat(FILE).catch(() => null);
   557→      if (!st) return;
   558→      if (st.size < readPos) readPos = 0; // rotated
   559→      if (st.size === readPos) return;
   560→      const len = Math.min(st.size - readPos, REFRESH_MAX_BITE);
   561→      const fh = await open(FILE, "r").catch(() => null);
   562→      if (!fh) return;
   563→      try {
   564→        const buf = Buffer.alloc(len);
   565→        const { bytesRead } = await fh.read(buf, 0, len, readPos);
   566→        let text = buf.toString("utf8", 0, bytesRead);
   567→        const lastNl = text.lastIndexOf("\n");
   568→        if (lastNl === -1) return; // partial line — wait for the writer
   569→        text = text.slice(0, lastNl + 1);
   570→        readPos += Buffer.byteLength(text, "utf8");
   571→        for (const line of text.split("\n")) {
   572→          if (!line) continue;
   573→          try {
   574→            const rec = JSON.parse(line) as { k: string; v: unknown };
   575→            if (rec.k === "v") {
   576→              const v = rec.v as ShadowVerdict;
   577→              const m = /^SV-(\d+)$/.exec(v.id ?? "");
   578→              if (ingestVerdict(v) && m) seq = Math.max(seq, Number(m[1]));
   579→            } else if (rec.k === "p") {
   580→              ingestPaper(rec.v as PaperTrade);
   581→            } else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   582→            else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });
   583→          } catch {
   584→            /* torn line */
   585→          }
   586→        }
   587→        hydratedAt = Date.now();
   588→      } finally {
   589→        await fh.close().catch(() => undefined);
   590→      }
   591→    } catch {
   592→      /* non-fatal */
   593→    } finally {
   594→      lastRefreshAt = Date.now();
   595→    }
   596→  })().finally(() => {
   597→    refreshing = null;
   598→  });
   599→  return refreshing;
   600→}

   668→/** Wilson score LOWER bound at 95% (z=1.96) — the A/B evidence statistic. */
   669→export function wilsonLB(wins: number, n: number): number | null {
   670→  if (n <= 0) return null;
   671→  const z = 1.96;
   672→  const p = wins / n;
   673→  const denom = 1 + (z * z) / n;
   674→  const centre = p + (z * z) / (2 * n);
   675→  const spread = z * Math.sqrt((p * (1 - p) + (z * z) / (4 * n)) / n);
   676→  return Math.max(0, ((centre - spread) / denom) * 100);
   677→}
   678→
   679→/** v11.0 — Wilson score UPPER bound at 95%: the OPTIMISTIC reading of a cohort's
   680→ * win rate. PATTERN-VETO uses it as the strict arm of its proof: even granting
   681→ * the would-veto cohort every favorable doubt, its ceiling must sit BELOW the
   682→ * keep cohort's OBSERVED win rate before veto authority flows. */
   683→export function wilsonUB(wins: number, n: number): number | null {
   684→  if (n <= 0) return null;
   685→  const z = 1.96;
   686→  const p = wins / n;
   687→  const denom = 1 + (z * z) / n;
   688→  const centre = p + (z * z) / (2 * n);
   689→  const spread = z * Math.sqrt((p * (1 - p) + (z * z) / (4 * n)) / n);
   690→  return Math.min(100, ((centre + spread) / denom) * 100);
   691→}

   727→/** Offer a scan batch's fence candidates to the adjudicator. Fire-and-forget,
   728→ * self-budgeted, deduped — never throws, never blocks the scan. */
   729→export async function offerFenceAdjudication(batch: FenceCandidate[]): Promise<void> {
   730→  try {
   731→    const cfg = await getShadowVetoConfig();
   732→    if (cfg.mode === "off" || batch.length === 0 || pendingCount >= MAX_PENDING) return;
   733→    await shadowVetoReady();
   734→    const now = Date.now();
   735→    // closest-to-gate first so the budget buys the most informative verdicts
   736→    const fence = batch
   737→      .map((c) => ({ c, gap: Math.abs(c.score) - c.effMin }))
   738→      .filter(({ c, gap }) => (c.passed ? gap >= 0 && gap <= cfg.windowPass : gap < 0 && gap >= cfg.windowReject))
   739→      .sort((a, b) => Math.abs(a.gap) - Math.abs(b.gap));
   740→    for (const { c, gap } of fence) {
   741→      const key = `${c.symbol}|${c.direction}`;
   742→      const last = lastAdjAt.get(key) ?? 0;
   743→      if (now - last < cfg.dedupMs) continue;
   744→      const cached = verdictCache.get(key);
   745→      if (cached && now - cached.at < cfg.cacheTtlMs) continue; // fresh verdict already in place
   746→      lastAdjAt.set(key, now);
   747→      pendingCount++;
   748→      void runAdjudication(c, c.passed ? "PASS" : "REJECT", gap)
   749→        .catch(() => undefined)
   750→        .finally(() => {
   751→          pendingCount = Math.max(0, pendingCount - 1);
   752→        });
   753→    }
   754→  } catch {
   755→    /* never break the scan */
   756→  }
   757→}

   766→}

   819→async function adjudicateCore(c: FenceCandidate, side: FenceSide, gap: number, cfg: ShadowVetoConfig, tag = "", origin: "capture" | "fence" = "fence"): Promise<void> {
   820→  const { aiShadowAdjudication } = await import("@/lib/ai");
   821→  const res = await aiShadowAdjudication({
   822→    symbol: c.symbol,
   823→    direction: c.direction,
   824→    strategy: c.strategy,
   825→    side,
   826→    gap,
   827→    score: c.score,
   828→    effMin: c.effMin,
   829→    thr: c.thr as unknown as Record<string, number>,
   830→    confirms: c.confirms,
   831→    opposes: c.opposes,
   832→    notes: c.notes,
   833→  });
   834→  seq++;
   835→  const rec: ShadowVerdict = {
   836→    id: `SV-${String(seq).padStart(4, "0")}`,
   837→    at: Date.now(),
   838→    symbol: c.symbol,

   842→    gap,

   845→    thr: c.thr,
   846→    verdict: res.verdict,
   847→    confidence: res.confidence,
   848→    reason: `${res.reason}${tag}`,
   849→    label: res.label,

   856→  };
   857→  verdicts.push(rec);
   858→  seenIds.add(rec.id); // id-discipline: this instance's live verdicts are also "seen"

   862→
   863→  // v8.2.1 — an audited HALLUCINATION is an operator-visible event (rate-bounded
   864→  // by the adjudication budget itself; this path only fires on VIOLATION)
   865→  if (res.audit?.verdict === "VIOLATION") {
   866→    await import("@/lib/engine")
   867→      .then((m) => m.fireEvent(null, c.symbol, "ENGINE", "warn", `SHADOW-VETO ${rec.id}: model verdict DISCARDED — grounding violation (${res.audit!.unmatched.length} citations not in evidence bundle). Heuristic adjudication used instead.`))
   868→      .catch(() => undefined);
   869→  }
   870→
   871→  // REJECT-side TRADE verdict = the core disagrees with the numeric gate →
   872→  // open the counterfactual paper trade (the would-have-won ledger).
   873→  if (side === "REJECT" && res.verdict === "TRADE") void openPaperTrade(rec, cfg.paperMarginUsd).catch(() => undefined);
   874→
   875→  // rate-limited audit event so the operator sees the core speaking
   876→  if ((side === "REJECT" && res.verdict === "TRADE") || (side === "PASS" && res.verdict === "VETO")) {
   877→    const pos = side === "PASS" ? "hair-thin pass" : `fence reject gap ${gap.toFixed(1)}`;

   879→    await import("@/lib/engine")
   880→      .then((m) => m.fireEvent(null, c.symbol, "ENGINE", "ai", msg))
   881→      .catch(() => undefined);
   882→  }
   883→}

  1033→ * PASS verdict older than 24h could NEVER find its deploy again — the
  1034→ * "125 PASS verdicts older than 24h never linked" telemetry alert was
  1035→ * structurally permanent. Fixes, evidence-honesty preserved:
  1036→ *   · match window widened to ±90min (verdict born up to 90min before or after
  1037→ *     the deploy it describes — nearest-Δt greedy still picks the tightest
  1038→ *     pair);
  1039→ *   · the trade lookback now covers the OLDEST unlinked verdict (minus the
  1040→ *     90min pre-warm margin), bounded at 7 days, so the stale backlog drains
  1041→ *     instead of aging out.
  1042→ * Identity constraints UNCHANGED: same symbol|direction, one-to-one (a serial
  1043→ * binds at most one verdict and vice versa), never touches REJECT-side
  1044→ * verdicts or already-linked ones. Outcome closure stays with the existing
  1045→ * closeRealOutcomes sweep. Invoked on the engine's existing tick-%40 cadence
  1046→ * (same block as closeRealOutcomes / maybeAutoPromote), so the backlog drains
  1047→ * automatically without any new scheduler. */
  1048→export async function retroLinkRealDeploys(): Promise<number> {
  1049→  try {
  1050→    await shadowVetoReady();
  1051→    const unlinked = verdicts.filter((v) => v.side === "PASS" && v.serial === null);
  1052→    if (unlinked.length === 0) return 0;
  1053→    const taken = new Set(verdicts.filter((v) => v.serial !== null).map((v) => v.serial!));
  1054→    // v10.16 — look back far enough to reach the OLDEST unlinked verdict's
  1055→    // deploy window (±90min), bounded at 7 days so the query stays cheap.
  1056→    const oldestUnlinked = unlinked.reduce((a, v) => Math.min(a, v.at), Date.now());
  1057→    const sinceMs = Math.max(Date.now() - 7 * 24 * 3600_000, oldestUnlinked - 90 * 60_000);
  1058→    const sigs = await db.signal.findMany({
  1059→      where: { createdAt: { gte: new Date(sinceMs) } }, // serial is non-nullable on Signal — no extra filter needed
  1060→      select: { serial: true, symbol: true, direction: true, createdAt: true },
  1061→      orderBy: { createdAt: "desc" },
  1062→      take: 400,
  1063→    });
  1064→    let bound = 0;
  1065→    // v10.19 ORPHAN SWEEP — coverage bounds of the fetch (take-400 desc): when
  1066→    // the query returns its FULL take, the oldest fetched row bounds coverage;
  1067→    // otherwise the query reaches sinceMs. A verdict is only stampable as an
  1068→    // orphan when its ENTIRE ±90min fence window lies inside covered ground —
  1069→    // absence is then PROOF no deploy ever existed, not a fetch artifact.
  1070→    const coverageLower = sigs.length === 400 && sigs.length > 0 ? sigs[sigs.length - 1]!.createdAt.getTime() : sinceMs;
  1071→    let swept = 0;
  1072→    for (const v of unlinked) {
  1073→      let anyInWindow = false; // v10.19 — a matching deploy exists (even if claimed)
  1074→      let best: { serial: number; dt: number } | null = null;
  1075→      for (const s of sigs) {
  1076→        if (s.symbol !== v.symbol || s.direction !== v.direction) continue;
  1077→        const dt = v.at - s.createdAt.getTime(); // <0 = verdict pre-warmed the deploy, >0 = fence batch after it
  1078→        // v10.16 — match window widened 15min→90min (both sides) so edge-born
  1079→        // verdicts and the >24h stale backlog can bind their own trades.
  1080→        if (dt < -90 * 60_000 || dt > 90 * 60_000) continue;
  1081→        anyInWindow = true;
  1082→        if (taken.has(s.serial!)) continue;
  1083→        if (!best || Math.abs(dt) < Math.abs(best.dt)) best = { serial: s.serial!, dt };
  1084→      }
  1085→      if (best) {
  1086→        v.serial = best.serial;
  1087→        taken.add(best.serial);
  1088→        void persist({ k: "v", v });
  1089→        bound++;
  1090→      } else if (
  1091→        v.sweptAt === undefined &&
  1092→        v.at + 90 * 60_000 < Date.now() && // fence window fully closed — a late deploy can no longer arrive
  1093→        v.at - 90 * 60_000 >= coverageLower && // window fully inside the query reach
  1094→        !anyInWindow // no same-symbol|direction deploy existed at all → structural orphan, promotion-inert
  1095→      ) {
  1096→        v.sweptAt = Date.now();
  1097→        void persist({ k: "v", v });
  1098→        swept++;
  1099→      }
  1100→    }
  1101→    if (swept > 0) console.log(`[shadow-veto] retro-link: bound=${bound} swept-orphans=${swept}`);
  1102→    return bound;
  1103→  } catch (e) {
  1104→    // Task 34-2: silent fail-open hid the retro-link stall (28 unlinked PASS
  1105→    // in 24h with bound=0) — same observability lesson as closeRealOutcomes.
  1106→    console.error("[shadow-veto] retroLinkRealDeploys failed", e);
  1107→    return 0; // evidence plumbing must never throw into the tick
  1108→  }
  1109→}

  1172→/** Base geometry for paper trades: slAtrMult × ATR stop (no regime/vol/flow
  1173→ * drivers — those are deploy-time only), levered-risk-capped leverage, base RR.
  1174→ * Direction-aware. Documented simplification: the paper arm measures
  1175→ * DIRECTIONAL edge at the fence, not perfect PnL replication. */
  1176→async function paperGeometry(symbol: string, direction: "LONG" | "SHORT", strategy: string, entryOverride: number | null): Promise<PaperGeometry> {
  1177→  const gate = await getGateSettings();
  1178→  const candles = await fetchKlines(symbol, "15m", 160).catch(() => null);
  1179→  const snap = candles ? buildSnapshot(symbol, candles) : null;
  1180→  const entry = entryOverride ?? snap?.price ?? 0;
  1181→  if (!entry) throw new Error("no price");
  1182→  const atrPct = snap?.atrPct ?? 1.5;
  1183→  const slDistPct = Math.min(5, Math.max(0.7, gate.slAtrMult * atrPct)) / 100;
  1184→  const levRaw = Math.round(Math.min(20, Math.max(5, 60 / (slDistPct * 100))));
  1185→  const leverage = gate.leveredRiskCap > 0 && slDistPct * 100 * levRaw > gate.leveredRiskCap ? Math.max(5, Math.floor(gate.leveredRiskCap / (slDistPct * 100))) : levRaw;
  1186→  const rr = RR_BASE[strategy] ?? 2.6;
  1187→  const tp2Mult = Math.min(2.2, Math.max(1.5, 1.2 + rr * 0.22));
  1188→  const [ttlMin, ttlMax] = TTL_BY_STRATEGY[strategy] ?? [90, 180];
  1189→  const isLong = direction === "LONG";
  1190→  const risk = entry * slDistPct;
  1191→  const fx = (v: number) => Number(v.toFixed(entry > 100 ? 2 : 6));
  1192→  return {
  1193→    entry: fx(entry),
  1194→    initialStop: fx(isLong ? entry - risk : entry + risk),
  1195→    stop: fx(isLong ? entry - risk : entry + risk),
  1196→    tp1: fx(isLong ? entry + risk : entry - risk),
  1197→    tp2: fx(isLong ? entry + risk * tp2Mult : entry - risk * tp2Mult),
  1198→    tp3: fx(isLong ? entry + risk * rr : entry - risk * rr),
  1199→    leverage,
  1200→    slDistPct,
  1201→    rr,
  1202→    ttlMinutes: Math.round(ttlMin + Math.random() * (ttlMax - ttlMin)),
  1203→  };
  1204→}

  1268→export async function shadowPaperSweep(priceMap: Map<string, number>): Promise<number> {
  1269→  try {

  1271→    const now = Date.now();
  1272→    let closed = 0;
  1273→    for (const p of papers) {
  1274→      if (p.outcome) continue;
  1275→      const price = priceMap.get(p.symbol);
  1276→      if (!price) continue;
  1277→      const isLong = p.direction === "LONG";
  1278→      const hitSl = isLong ? price <= p.stop : price >= p.stop;
  1279→      const hitTp1 = isLong ? price >= p.tp1 : price <= p.tp1;
  1280→      const hitTp2 = isLong ? price >= p.tp2 : price <= p.tp2;
  1281→      const hitTp3 = isLong ? price >= p.tp3 : price <= p.tp3;
  1282→
  1283→      if (hitSl) {
  1284→        closePaper(p, "SL", movePctAt(p, p.stop), now, `stopped @ ${p.stop} (trailed)`);
  1285→        closed++;
  1286→      } else if (hitTp3) {
  1287→        p.reachedTp = 3;
  1288→        closePaper(p, "TP3", movePctAt(p, p.tp3), now, `final target ${p.rr.toFixed(1)}R secured`);
  1289→        closed++;
  1290→      } else if (hitTp2 && p.reachedTp < 2) {
  1291→        p.reachedTp = 2;
  1292→        p.status = "TP2";
  1293→        p.stop = p.tp1; // trail to TP1 — +1R locked (engine semantics)
  1294→      } else if (hitTp1 && p.reachedTp < 1) {
  1295→        p.reachedTp = Math.max(p.reachedTp, 1);
  1296→        p.status = "TP1";
  1297→        p.stop = p.entry; // breakeven trail (engine semantics)
  1298→      } else if (now > p.at + p.ttlMinutes * 60_000) {
  1299→        closePaper(p, "EXPIRED", movePctAt(p, price), now, `TTL ${p.ttlMinutes}m elapsed at mark`);
  1300→        closed++;
  1301→      }
  1302→      // track MFE
  1303→      p.maxPnlPercent = Math.max(p.maxPnlPercent, Number((movePctAt(p, price) * p.leverage).toFixed(2)));
  1304→    }
  1305→    return closed;
  1306→  } catch {
  1307→    return 0;
  1308→  }
  1309→}

  1312→function closePaper(p: PaperTrade, status: PaperTrade["status"], exitMovePct: number, closedAt: number, note: string): void {
  1313→  const rung = (px: number) => movePctAt(p, px);
  1314→  const grossPct = p.reachedTp >= 1 ? ladderWeightedMove([{ weight: LADDER_TRANCHES.tp1, movePct: rung(p.tp1) }, ...(p.reachedTp >= 2 ? [{ weight: LADDER_TRANCHES.tp2, movePct: rung(p.tp2) }] : []), { weight: LADDER_TRANCHES.runner, movePct: status === "TP3" ? rung(p.tp3) : exitMovePct }]) : exitMovePct;
  1315→  const heldMin = Math.max(0, (closedAt - p.at) / 60_000);
  1316→  const fb = feeBreakdown(p.marginUsd, p.leverage, grossPct, heldMin);
  1317→  p.status = status;
  1318→  p.pnlPercent = Number(fb.netPct.toFixed(2));
  1319→  p.pnlUsd = Number(fb.netUsd.toFixed(3));

  1321→    status,
  1322→    pnlPercent: p.pnlPercent,
  1323→    pnlUsd: p.pnlUsd,
  1324→    win: p.pnlPercent > 0 || p.reachedTp >= 1,
  1325→    closedAt,
  1326→  };

  1328→  void persist({ k: "pc", v: { id: p.id, status: p.status, reachedTp: p.reachedTp, pnlPercent: p.pnlPercent, pnlUsd: p.pnlUsd, stop: p.stop, outcome: p.outcome, closeNote: note } });
  1329→}

  1454→const PATTERN_MIN_N = 8; // per side — below this a cell cannot speak
  1455→const PATTERN_MIN_SEP_PP = 15; // observed WR separation (keep − veto)
  1456→
  1457→/** PATTERN-VETO evidence: per (strategy|direction) cell over the settled
  1458→ * fence-origin REAL pass-side archive. The proof test (ALL must hold):
  1459→ *   · n≥8 on BOTH sides (minimum sample per cell),
  1460→ *   · observed WR separation keep−veto ≥ 15pp,
  1461→ *   · Wilson-UB95(veto) < WR(keep) — even the OPTIMISTIC bound of the
  1462→ *     would-veto cohort cannot reach the keep cohort's observed win rate,
  1463→ *   · would-veto cohort net < 0 (the vetoes must have been losing money).
  1464→ * REAL-only by design: paper rows measure fence REJECTS with base geometry —
  1465→ * a different population; mixing them would contaminate the proof. Legacy
  1466→ * archive rows without pattern dimensions are inert (never vote). */
  1467→export function patternVetoEvidence(): { rows: PatternEvidenceRow[]; proven: string[] } {
  1468→  const cells = new Map<string, { keep: Array<{ win: boolean; usd: number }>; veto: Array<{ win: boolean; usd: number }> }>();
  1469→  for (const e of settledArchive.values()) {
  1470→    if (e.kind !== "real" || e.side !== "PASS" || !e.epoch || e.capture || !e.outcome) continue;
  1471→    if (!e.strategy || !e.direction) continue; // legacy row — pattern-inert
  1472→    const key = `${e.strategy}|${e.direction}`;
  1473→    if (!cells.has(key)) cells.set(key, { keep: [], veto: [] });
  1474→    const cell = cells.get(key)!;
  1475→    const row = { win: e.outcome.win, usd: e.outcome.pnlUsd };
  1476→    if (e.verdict === "TRADE") cell.keep.push(row);
  1477→    else if (e.verdict === "VETO") cell.veto.push(row);
  1478→  }
  1479→  const rows: PatternEvidenceRow[] = [];
  1480→  for (const [pattern, cell] of [...cells.entries()].sort((a, b) => a[0].localeCompare(b[0]))) {
  1481→    const nK = cell.keep.length;
  1482→    const nV = cell.veto.length;
  1483→    const wK = cell.keep.filter((r) => r.win).length;
  1484→    const wV = cell.veto.filter((r) => r.win).length;
  1485→    const wrK = nK ? (100 * wK) / nK : null;
  1486→    const wrV = nV ? (100 * wV) / nV : null;
  1487→    const netK = Number(cell.keep.reduce((a, r) => a + r.usd, 0).toFixed(2));
  1488→    const netV = Number(cell.veto.reduce((a, r) => a + r.usd, 0).toFixed(2));
  1489→    const ubV = wilsonUB(wV, nV);
  1490→    const okN = nK >= PATTERN_MIN_N && nV >= PATTERN_MIN_N;
  1491→    const okSep = wrK !== null && wrV !== null && wrK - wrV >= PATTERN_MIN_SEP_PP;
  1492→    const okUB = ubV !== null && wrK !== null && ubV < wrK;
  1493→    const okNet = netV < 0;
  1494→    const proven = okN && okSep && okUB && okNet;
  1495→    const f1 = (x: number | null): string => (x === null ? "n/a" : `${x.toFixed(1)}%`);
  1496→    rows.push({
  1497→      pattern,
  1498→      nKeep: nK,
  1499→      wrKeep: wrK === null ? null : Number(wrK.toFixed(1)),
  1500→      netKeep: netK,
  1501→      nVeto: nV,
  1502→      wrVeto: wrV === null ? null : Number(wrV.toFixed(1)),
  1503→      netVeto: netV,
  1504→      ubVeto: ubV === null ? null : Number(ubV.toFixed(1)),
  1505→      proven,
  1506→      detail: `n veto/keep ${nV}/${nK} · wr ${f1(wrV)}/${f1(wrK)} · UB95 ${f1(ubV)} · net $${netV}/$${netK} · ${okN ? "✓n" : "✗n"}${okSep ? " ✓sep" : " ✗sep"}${okUB ? " ✓ub" : " ✗ub"}${okNet ? " ✓net" : " ✗net"}`,
  1507→    });
  1508→  }
  1509→  return { rows, proven: rows.filter((r) => r.proven).map((r) => r.pattern) };
  1510→}

  1619→ *           waiting for 30 adds latency without adding certainty).

  1622→export function promotionEvidence(arms: ArmRow[]): PromotionEvidence {

  1628→  // alpha-spending). It computes on the SAME arms the fixed gate sees (frozen
  1629→  // when a COHORT-FREEZE exists): a PROMOTE-EARLY state satisfies promotion
  1630→  // WITHOUT waiting for n=30 (min-n guard 12/arm inside), ABORT-EARLY vetoes
  1631→  // the hypothesis outright, CONTINUE leaves the fixed gate as sole path.
  1632→  const sequential = seqStopEvaluate(
  1633→    { wins: realKeep.wins, n: realKeep.n, netUsd: realKeep.netUsd },
  1634→    { wins: realVeto.wins, n: realVeto.n, netUsd: realVeto.netUsd },
  1635→  );
  1636→  const okN = (a: ArmRow): boolean => a.n >= 30 || (a.n >= 20 && (a.wilsonLB ?? 0) >= 50);
  1637→  const nDetail = (role: string, a: ArmRow): string => {
  1638→    const fast = a.n >= 20 && a.n < 30 && (a.wilsonLB ?? 0) >= 50;
  1639→    return `rolling-30 ${role} cohort ${a.n}/30${a.n >= 20 && a.n < 30 ? ` (fast path n≥20 + LB≥50%: ${fast ? "MET" : "unmet — LB " + (a.wilsonLB !== null ? a.wilsonLB.toFixed(1) : "n/a") + "%"})` : ""}`;
  1640→  };
  1641→  const criteria = [

   171→function maxIdSeq(list: Array<{ id: string }>): number {
   172→  let max = 0;
   173→  for (const r of list) {
   174→    const m = /^S[VP]-(\d+)$/.exec(r.id);
   175→    if (m) max = Math.max(max, Number(m[1]));
   176→  }
   177→  return max;

   180→function ingestVerdict(v: ShadowVerdict): boolean {
   181→  if (!v || typeof v.id !== "string") return false;
   182→  if (seenIds.has(v.id)) return false;
   183→  v.audit = v.audit ?? null; // normalize legacy records
   184→  // collapse cross-instance duplicate adjudications of the same candidate —
   185→  // scan backwards to the candidate's latest record; inside the window = dupe
   186→  for (let i = verdicts.length - 1; i >= 0; i--) {
   187→    const x = verdicts[i]!;
   188→    if (x.symbol === v.symbol && x.direction === v.direction) {
   189→      if (Math.abs(x.at - v.at) < INGEST_COLLAPSE_MS) return false;
   190→      break;
   191→    }
   192→  }
   193→  seenIds.add(v.id);
   194→  verdicts.push(v);
   195→  if (verdicts.length > VERDICT_CAP) verdicts.splice(0, verdicts.length - VERDICT_CAP);
   196→  return true;
   197→}

   199→function ingestPaper(p: PaperTrade): boolean {
   200→  if (!p || typeof p.id !== "string") return false;
   201→  if (seenIds.has(p.id)) return false;
   202→  seenIds.add(p.id);
   203→  papers.push(p);
   204→  trimPapers();
   205→  return true;
   206→}

   208→/** v10.7 FIFO cap with evidence-safety — evict oldest SETTLED papers first;
   209→ * only when the ring is all-OPEN (degenerate burst) fall back to oldest-first.
   210→ * A settled record can always be re-derived from the rotated journal, but a
   211→ * live counterfactual evicted mid-flight is lost evidence forever. */
   212→function trimPapers(): void {
   213→  if (papers.length <= PAPER_CAP) return;
   214→  let overflow = papers.length - PAPER_CAP;
   215→  for (let i = 0; i < papers.length && overflow > 0; ) {
   216→    if (papers[i]!.outcome) {
   217→      papers.splice(i, 1);
   218→      overflow--;
   219→    } else i++;
   220→  }
   221→  if (papers.length > PAPER_CAP) papers.splice(0, papers.length - PAPER_CAP);
   222→}

   251→async function hydrate(): Promise<void> {
   252→  if (hydrating) return hydrating;
   253→  hydrating = (async () => {
   254→    try {
   255→      const buf = await readFile(FILE, "utf8").catch(() => "");
   256→      if (buf) {
   257→        const lines = buf.split("\n").filter(Boolean);
   258→        for (const line of lines.slice(-3000)) {
   259→          try {
   260→            const rec = JSON.parse(line) as { k: string; v: unknown };
   261→            if (rec.k === "v") ingestVerdict(rec.v as ShadowVerdict);
   262→            else if (rec.k === "p") ingestPaper(rec.v as PaperTrade);
   263→            else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   264→            else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });
   265→          } catch {
   266→            /* torn line */
   267→          }
   268→        }
   269→        // v8.2.1 — seq MUST derive from the MAX id on disk (length-derived seq
   270→        // reused ids when two module instances wrote with independent counters)
   271→        seq = Math.max(maxIdSeq(verdicts), maxIdSeq(papers));
   272→      }
   273→      const st = await stat(FILE).catch(() => null);
   274→      persisted = st?.size ?? 0;
   275→      readPos = st?.size ?? 0;
   276→      hydratedAt = Date.now();
   277→      // dedup continuity across instance swaps — a fresh instance must not
   278→      // instantly re-adjudicate what another instance just adjudicated
   279→      for (const v of verdicts) {
   280→        const key = `${v.symbol}|${v.direction}`;
   281→        const prev = lastAdjAt.get(key) ?? 0;
   282→        if (v.at > prev) lastAdjAt.set(key, v.at);
   283→      }
   284→      if ((st?.size ?? 0) > FILE_MAX_BYTES) {
   285→        // v10.6.1 persist-guard (operator APPROVE 2026-09-18): rotation is the
   286→        // destructive rewrite — if a fresher boot just rotated, resync our
   287→        // counters from disk instead of renaming over its rotation.
   288→        const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");
   289→        await serializePersist("shadow-veto-journal", async () => {
   290→          if (!(await persistGuardOk("shadow-veto-journal"))) {
   291→            const fresh = await stat(FILE).catch(() => null);
   292→            persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   293→            readPos = fresh?.size ?? 0;
   294→            return;
   295→          }
   296→          await rename(FILE, `${FILE}.old`).catch(() => undefined);
   297→          await persistWatermark("shadow-veto-journal");
   298→          persisted = 0;
   299→          readPos = 0;
   300→        });
   301→      }
   302→    } catch {
   303→      /* journaling must never break the engine */
   304→    }
   305→  })();
   306→  return hydrating;
   307→}

   309→async function persist(rec: { k: string; v: unknown }): Promise<void> {
   310→  try {
   311→    const line = JSON.stringify(rec) + "\n";
   312→    await appendFile(FILE, line, "utf8");
   313→    persisted += line.length;
   314→    readPos += line.length; // our own bytes are already in the rings
   315→    if (persisted > FILE_MAX_BYTES) {
   316→      // v10.6.1 persist-guard: serialize rotation + re-stat before renaming so
   317→      // concurrent writers (or a stale generation) never clobber a fresh file.
   318→      const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");
   319→      await serializePersist("shadow-veto-journal", async () => {
   320→        const cur = await stat(FILE).catch(() => null);
   321→        if ((cur?.size ?? 0) <= FILE_MAX_BYTES) {
   322→          // someone else already rotated — just resync our counters
   323→          persisted = Math.min(cur?.size ?? 0, FILE_MAX_BYTES);
   324→          readPos = cur?.size ?? 0;
   325→          return;
   326→        }
   327→        if (!(await persistGuardOk("shadow-veto-journal"))) {
   328→          const fresh = await stat(FILE).catch(() => null);
   329→          persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   330→          readPos = fresh?.size ?? 0;
   331→          return;
   332→        }
   333→        await rename(FILE, `${FILE}.old`).catch(() => undefined);
   334→        await persistWatermark("shadow-veto-journal");
   335→        persisted = 0;
   336→        readPos = 0;
   337→      });
   338→    }
   339→  } catch {
   340→    /* non-fatal */
   341→  }
   342→}

   353→/** v8.2.1 READ-PATH LIVENESS — tail-ingest another instance's writes (see
   354→ * gate-journal.gateJournalRefresh for the full rationale). Throttled,
   355→ * single-flight, rotation-safe, bounded bite. */
   356→export async function shadowVetoRefresh(maxAgeMs = REFRESH_MIN_INTERVAL_MS): Promise<void> {
   357→  if (Date.now() - lastRefreshAt < maxAgeMs) return;
   358→  if (refreshing) return refreshing;
   359→  refreshing = (async () => {
   360→    try {
   361→      const st = await stat(FILE).catch(() => null);
   362→      if (!st) return;
   363→      if (st.size < readPos) readPos = 0; // rotated
   364→      if (st.size === readPos) return;
   365→      const len = Math.min(st.size - readPos, REFRESH_MAX_BITE);
   366→      const fh = await open(FILE, "r").catch(() => null);
   367→      if (!fh) return;
   368→      try {
   369→        const buf = Buffer.alloc(len);
   370→        const { bytesRead } = await fh.read(buf, 0, len, readPos);
   371→        let text = buf.toString("utf8", 0, bytesRead);
   372→        const lastNl = text.lastIndexOf("\n");
   373→        if (lastNl === -1) return; // partial line — wait for the writer
   374→        text = text.slice(0, lastNl + 1);
   375→        readPos += Buffer.byteLength(text, "utf8");
   376→        for (const line of text.split("\n")) {
   377→          if (!line) continue;
   378→          try {
   379→            const rec = JSON.parse(line) as { k: string; v: unknown };
   380→            if (rec.k === "v") {
   381→              const v = rec.v as ShadowVerdict;
   382→              const m = /^SV-(\d+)$/.exec(v.id ?? "");
   383→              if (ingestVerdict(v) && m) seq = Math.max(seq, Number(m[1]));
   384→            } else if (rec.k === "p") {
   385→              ingestPaper(rec.v as PaperTrade);
   386→            } else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   387→            else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });
   388→          } catch {
   389→            /* torn line */
   390→          }
   391→        }
   392→        hydratedAt = Date.now();
   393→      } finally {
   394→        await fh.close().catch(() => undefined);
   395→      }
   396→    } catch {
   397→      /* non-fatal */
   398→    } finally {
   399→      lastRefreshAt = Date.now();
   400→    }
   401→  })().finally(() => {
   402→    refreshing = null;
   403→  });
   404→  return refreshing;
   405→}
   406→
   407→function applyOutcomeRecord(rec: { id: string; outcome: ShadowOutcome }): void {
   408→  const v = [...verdicts].reverse().find((x) => x.id === rec.id);
   409→  if (v && !v.outcome) v.outcome = rec.outcome;
   410→}
   411→
   412→function applyPaperClose(rec: { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string }): void {
   413→  const p = [...papers].reverse().find((x) => x.id === rec.id);
   414→  if (p && !p.outcome) {
   415→    p.status = rec.status;
   416→    p.reachedTp = rec.reachedTp;
   417→    p.pnlPercent = rec.pnlPercent;
   418→    p.pnlUsd = rec.pnlUsd;
   419→    p.stop = rec.stop;
   420→    p.outcome = rec.outcome;
   421→    p.closeNote = rec.closeNote;
   422→  }
   423→}

   468→/** Wilson score LOWER bound at 95% (z=1.96) — the A/B evidence statistic. */
   469→export function wilsonLB(wins: number, n: number): number | null {
   470→  if (n <= 0) return null;
   471→  const z = 1.96;
   472→  const p = wins / n;
   473→  const denom = 1 + (z * z) / n;
   474→  const centre = p + (z * z) / (2 * n);
   475→  const spread = z * Math.sqrt((p * (1 - p) + (z * z) / (4 * n)) / n);
   476→  return Math.max(0, ((centre - spread) / denom) * 100);
   477→}

   513→/** Offer a scan batch's fence candidates to the adjudicator. Fire-and-forget,
   514→ * self-budgeted, deduped — never throws, never blocks the scan. */
   515→export async function offerFenceAdjudication(batch: FenceCandidate[]): Promise<void> {
   516→  try {
   517→    const cfg = await getShadowVetoConfig();
   518→    if (cfg.mode === "off" || batch.length === 0 || pendingCount >= MAX_PENDING) return;
   519→    await shadowVetoReady();
   520→    const now = Date.now();
   521→    // closest-to-gate first so the budget buys the most informative verdicts
   522→    const fence = batch
   523→      .map((c) => ({ c, gap: Math.abs(c.score) - c.effMin }))
   524→      .filter(({ c, gap }) => (c.passed ? gap >= 0 && gap <= cfg.windowPass : gap < 0 && gap >= cfg.windowReject))
   525→      .sort((a, b) => Math.abs(a.gap) - Math.abs(b.gap));
   526→    for (const { c, gap } of fence) {
   527→      const key = `${c.symbol}|${c.direction}`;
   528→      const last = lastAdjAt.get(key) ?? 0;
   529→      if (now - last < cfg.dedupMs) continue;
   530→      const cached = verdictCache.get(key);
   531→      if (cached && now - cached.at < cfg.cacheTtlMs) continue; // fresh verdict already in place
   532→      lastAdjAt.set(key, now);
   533→      pendingCount++;
   534→      void runAdjudication(c, c.passed ? "PASS" : "REJECT", gap)
   535→        .catch(() => undefined)
   536→        .finally(() => {
   537→          pendingCount = Math.max(0, pendingCount - 1);
   538→        });
   539→    }
   540→  } catch {
   541→    /* never break the scan */
   542→  }
   543→}

   823→    // v10.16 — look back far enough to reach the OLDEST unlinked verdict's
   824→    // deploy window (±90min), bounded at 7 days so the query stays cheap.
   825→    const oldestUnlinked = unlinked.reduce((a, v) => Math.min(a, v.at), Date.now());

   827→    const sigs = await db.signal.findMany({

   831→      take: 400,

   844→      }

   846→        v.serial = best.serial;
   847→        taken.add(best.serial);

   849→        bound++;
   850→      }
   851→    }
   852→    return bound;
   853→  } catch {
   854→    return 0; // evidence plumbing must never throw into the tick

   913→ * DIRECTIONAL edge at the fence, not perfect PnL replication. */
   914→async function paperGeometry(symbol: string, direction: "LONG" | "SHORT", strategy: string, entryOverride: number | null): Promise<PaperGeometry> {
   915→  const gate = await getGateSettings();
   916→  const candles = await fetchKlines(symbol, "15m", 160).catch(() => null);
   917→  const snap = candles ? buildSnapshot(symbol, candles) : null;
   918→  const entry = entryOverride ?? snap?.price ?? 0;
   919→  if (!entry) throw new Error("no price");
   920→  const atrPct = snap?.atrPct ?? 1.5;
   921→  const slDistPct = Math.min(5, Math.max(0.7, gate.slAtrMult * atrPct)) / 100;
   922→  const levRaw = Math.round(Math.min(20, Math.max(5, 60 / (slDistPct * 100))));
   923→  const leverage = gate.leveredRiskCap > 0 && slDistPct * 100 * levRaw > gate.leveredRiskCap ? Math.max(5, Math.floor(gate.leveredRiskCap / (slDistPct * 100))) : levRaw;
   924→  const rr = RR_BASE[strategy] ?? 2.6;
   925→  const tp2Mult = Math.min(2.2, Math.max(1.5, 1.2 + rr * 0.22));
   926→  const [ttlMin, ttlMax] = TTL_BY_STRATEGY[strategy] ?? [90, 180];
   927→  const isLong = direction === "LONG";
   928→  const risk = entry * slDistPct;
   929→  const fx = (v: number) => Number(v.toFixed(entry > 100 ? 2 : 6));
   930→  return {
   931→    entry: fx(entry),
   932→    initialStop: fx(isLong ? entry - risk : entry + risk),
   933→    stop: fx(isLong ? entry - risk : entry + risk),
   934→    tp1: fx(isLong ? entry + risk : entry - risk),
   935→    tp2: fx(isLong ? entry + risk * tp2Mult : entry - risk * tp2Mult),
   936→    tp3: fx(isLong ? entry + risk * rr : entry - risk * rr),
   937→    leverage,
   938→    slDistPct,
   939→    rr,
   940→    ttlMinutes: Math.round(ttlMin + Math.random() * (ttlMax - ttlMin)),
   941→  };
   942→}

  1003→/** Per-tick sweep — engine passes the fresh price map. Mirrors the real
  1004→ * resolver: rung ladders with trailing stops, tranche-weighted closes, fees
  1005→ * via feeBreakdown, TTL expiry. In-memory, cheap, never throws. */
  1006→export async function shadowPaperSweep(priceMap: Map<string, number>): Promise<number> {
  1007→  try {
  1008→    if (papers.length === 0) return 0;
  1009→    const now = Date.now();
  1010→    let closed = 0;
  1011→    for (const p of papers) {
  1012→      if (p.outcome) continue;
  1013→      const price = priceMap.get(p.symbol);
  1014→      if (!price) continue;
  1015→      const isLong = p.direction === "LONG";
  1016→      const hitSl = isLong ? price <= p.stop : price >= p.stop;
  1017→      const hitTp1 = isLong ? price >= p.tp1 : price <= p.tp1;
  1018→      const hitTp2 = isLong ? price >= p.tp2 : price <= p.tp2;
  1019→      const hitTp3 = isLong ? price >= p.tp3 : price <= p.tp3;
  1020→
  1021→      if (hitSl) {
  1022→        closePaper(p, "SL", movePctAt(p, p.stop), now, `stopped @ ${p.stop} (trailed)`);
  1023→        closed++;
  1024→      } else if (hitTp3) {
  1025→        p.reachedTp = 3;
  1026→        closePaper(p, "TP3", movePctAt(p, p.tp3), now, `final target ${p.rr.toFixed(1)}R secured`);
  1027→        closed++;
  1028→      } else if (hitTp2 && p.reachedTp < 2) {
  1029→        p.reachedTp = 2;
  1030→        p.status = "TP2";
  1031→        p.stop = p.tp1; // trail to TP1 — +1R locked (engine semantics)
  1032→      } else if (hitTp1 && p.reachedTp < 1) {
  1033→        p.reachedTp = Math.max(p.reachedTp, 1);
  1034→        p.status = "TP1";
  1035→        p.stop = p.entry; // breakeven trail (engine semantics)
  1036→      } else if (now > p.at + p.ttlMinutes * 60_000) {
  1037→        closePaper(p, "EXPIRED", movePctAt(p, price), now, `TTL ${p.ttlMinutes}m elapsed at mark`);
  1038→        closed++;
  1039→      }
  1040→      // track MFE
  1041→      p.maxPnlPercent = Math.max(p.maxPnlPercent, Number((movePctAt(p, price) * p.leverage).toFixed(2)));
  1042→    }
  1043→    return closed;
  1044→  } catch {
  1045→    return 0;
  1046→  }
  1047→}

  1050→function closePaper(p: PaperTrade, status: PaperTrade["status"], exitMovePct: number, closedAt: number, note: string): void {
  1051→  const rung = (px: number) => movePctAt(p, px);
  1052→  const grossPct = p.reachedTp >= 1 ? ladderWeightedMove([{ weight: LADDER_TRANCHES.tp1, movePct: rung(p.tp1) }, ...(p.reachedTp >= 2 ? [{ weight: LADDER_TRANCHES.tp2, movePct: rung(p.tp2) }] : []), { weight: LADDER_TRANCHES.runner, movePct: status === "TP3" ? rung(p.tp3) : exitMovePct }]) : exitMovePct;
  1053→  const heldMin = Math.max(0, (closedAt - p.at) / 60_000);
  1054→  const fb = feeBreakdown(p.marginUsd, p.leverage, grossPct, heldMin);
  1055→  p.status = status;
  1056→  p.pnlPercent = Number(fb.netPct.toFixed(2));
  1057→  p.pnlUsd = Number(fb.netUsd.toFixed(3));
  1058→  p.outcome = {
  1059→    status,
  1060→    pnlPercent: p.pnlPercent,
  1061→    pnlUsd: p.pnlUsd,
  1062→    win: p.pnlPercent > 0 || p.reachedTp >= 1,
  1063→    closedAt,
  1064→  };
  1065→  p.closeNote = note;
  1066→  void persist({ k: "pc", v: { id: p.id, status: p.status, reachedTp: p.reachedTp, pnlPercent: p.pnlPercent, pnlUsd: p.pnlUsd, stop: p.stop, outcome: p.outcome, closeNote: note } });
  1067→}

  1265→/** v10.7 REAL-ONLY promotion evidence → v10.10 ROLLING-30 EVIDENCE WINDOW →
  1266→ * v10.16 FENCE-ORIGIN COHORTS (audit 4-f). Cohort qualification, per real arm
  1267→ * (over the rolling-30 settled window — see shadowArms(); since v10.16 the
  1268→ * arms themselves are computed over fence-origin verdicts ONLY — the capture-
  1269→ * observe path stamps origin:"capture" and is filtered in
  1270→ * liveRealArmAggregates, so the n/WR/net below are fence-pure and a fence arm
  1271→ * below its minimum simply fails okN → no promotion):
  1272→ *   FULL  — n ≥ 30 settled (original bar), or
  1273→ *   FAST  — n ≥ 20 AND Wilson-LB95 ≥ 50% (guardrail min-sample practice: the
  1274→ *           lower bound of the win rate already clears coin-flip odds, so
  1275→ *           waiting for 30 adds latency without adding certainty).
  1276→ * Wilson separation + negative would-veto net are UNCHANGED. Epoch is a
  1277→ * hygiene tag only — the evidence window is the rolling-30 (v10.10). */
  1278→export function promotionEvidence(arms: ArmRow[]): PromotionEvidence {
  1279→  const realKeep = arms[0]!;
  1280→  const realVeto = arms[1]!;
  1281→  const paperVeto = arms[3]!;
  1282→  const both = realKeep.n >= 30 && realVeto.n >= 30;
  1283→  // v10.11 SEQ-STOP — the sequential early-decision layer (always-valid, OF
  1284→  // alpha-spending). It computes on the SAME arms the fixed gate sees (frozen
  1285→  // when a COHORT-FREEZE exists): a PROMOTE-EARLY state satisfies promotion
  1286→  // WITHOUT waiting for n=30 (min-n guard 12/arm inside), ABORT-EARLY vetoes
  1287→  // the hypothesis outright, CONTINUE leaves the fixed gate as sole path.
  1288→  const sequential = seqStopEvaluate(
  1289→    { wins: realKeep.wins, n: realKeep.n, netUsd: realKeep.netUsd },
  1290→    { wins: realVeto.wins, n: realVeto.n, netUsd: realVeto.netUsd },
  1291→  );

   188→function maxIdSeq(list: Array<{ id: string }>): number {
   189→  let max = 0;
   190→  for (const r of list) {
   191→    const m = /^S[VP]-(\d+)$/.exec(r.id);
   192→    if (m) max = Math.max(max, Number(m[1]));

   197→function ingestVerdict(v: ShadowVerdict): boolean {
   198→  if (!v || typeof v.id !== "string") return false;
   199→  // v10.19 LINK-DELTA REBUILD (restored lost v10.17): the JSONL re-persists a
   200→  // verdict whenever its state UPGRADES (linkRealDeploy binds the serial, the
   201→  // outcome sweep closes it, the orphan sweep stamps it). First-record-wins
   202→  // hydration silently DROPPED those upgrades — every restart reverted linked-
   203→  // but-unsettled verdicts back to unlinked (the real-veto 17→12 regression
   204→  // after the 2026-09-21 restarts). Duplicate ids now MERGE their upgrade
   205→  // fields into the already-ingested record, order-independent.
   206→  if (seenIds.has(v.id)) {
   207→    const ex = verdicts.find((x) => x.id === v.id);
   208→    if (ex) {
   209→      if (ex.serial === null && v.serial !== null) ex.serial = v.serial;
   210→      if (!ex.outcome && v.outcome) ex.outcome = v.outcome;
   211→      if (ex.sweptAt === undefined && v.sweptAt !== undefined) ex.sweptAt = v.sweptAt;
   212→    }
   213→    return false;
   214→  }
   215→  v.audit = v.audit ?? null; // normalize legacy records
   216→  // collapse cross-instance duplicate adjudications of the same candidate —
   217→  // scan backwards to the candidate's latest record; inside the window = dupe
   218→  for (let i = verdicts.length - 1; i >= 0; i--) {
   219→    const x = verdicts[i]!;
   220→    if (x.symbol === v.symbol && x.direction === v.direction) {
   221→      if (Math.abs(x.at - v.at) < INGEST_COLLAPSE_MS) return false;
   222→      break;
   223→    }
   224→  }
   225→  seenIds.add(v.id);
   226→  verdicts.push(v);
   227→  if (verdicts.length > VERDICT_CAP) verdicts.splice(0, verdicts.length - VERDICT_CAP);
   228→  return true;
   229→}

   240→/** v10.7 FIFO cap with evidence-safety — evict oldest SETTLED papers first;
   241→ * only when the ring is all-OPEN (degenerate burst) fall back to oldest-first.
   242→ * A settled record can always be re-derived from the rotated journal, but a
   243→ * live counterfactual evicted mid-flight is lost evidence forever. */
   244→function trimPapers(): void {
   245→  if (papers.length <= PAPER_CAP) return;
   246→  let overflow = papers.length - PAPER_CAP;
   247→  for (let i = 0; i < papers.length && overflow > 0; ) {
   248→    if (papers[i]!.outcome) {
   249→      papers.splice(i, 1);
   250→      overflow--;
   251→    } else i++;
   252→  }
   253→  if (papers.length > PAPER_CAP) papers.splice(0, papers.length - PAPER_CAP);
   254→}

   284→  if (hydrating) return hydrating;
   285→  hydrating = (async () => {
   286→    try {
   287→      const buf = await readFile(FILE, "utf8").catch(() => "");
   288→      if (buf) {
   289→        const lines = buf.split("\n").filter(Boolean);
   290→        for (const line of lines.slice(-3000)) {
   291→          try {
   292→            const rec = JSON.parse(line) as { k: string; v: unknown };
   293→            if (rec.k === "v") ingestVerdict(rec.v as ShadowVerdict);
   294→            else if (rec.k === "p") ingestPaper(rec.v as PaperTrade);
   295→            else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   296→            else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });

   299→          }

   302→        // reused ids when two module instances wrote with independent counters)

   315→      }
   316→      if ((st?.size ?? 0) > FILE_MAX_BYTES) {
   317→        // v10.6.1 persist-guard (operator APPROVE 2026-09-18): rotation is the
   318→        // destructive rewrite — if a fresher boot just rotated, resync our

   320→        const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");

   322→          if (!(await persistGuardOk("shadow-veto-journal"))) {

   324→            persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   325→            readPos = fresh?.size ?? 0;
   326→            return;
   327→          }
   328→          await rename(FILE, `${FILE}.old`).catch(() => undefined);
   329→          await persistWatermark("shadow-veto-journal");
   330→          persisted = 0;
   331→          readPos = 0;
   332→        });
   333→      }
   334→    } catch {
   335→      /* journaling must never break the engine */
   336→    }
   337→  })();
   338→  return hydrating;
   339→}

   341→async function persist(rec: { k: string; v: unknown }): Promise<void> {
   342→  try {
   343→    const line = JSON.stringify(rec) + "\n";
   344→    await appendFile(FILE, line, "utf8");
   345→    persisted += line.length;
   346→    readPos += line.length; // our own bytes are already in the rings
   347→    if (persisted > FILE_MAX_BYTES) {
   348→      // v10.6.1 persist-guard: serialize rotation + re-stat before renaming so
   349→      // concurrent writers (or a stale generation) never clobber a fresh file.
   350→      const { persistGuardOk, persistWatermark, serializePersist } = await import("./persist-guard");
   351→      await serializePersist("shadow-veto-journal", async () => {
   352→        const cur = await stat(FILE).catch(() => null);
   353→        if ((cur?.size ?? 0) <= FILE_MAX_BYTES) {
   354→          // someone else already rotated — just resync our counters
   355→          persisted = Math.min(cur?.size ?? 0, FILE_MAX_BYTES);
   356→          readPos = cur?.size ?? 0;
   357→          return;
   358→        }
   359→        if (!(await persistGuardOk("shadow-veto-journal"))) {
   360→          const fresh = await stat(FILE).catch(() => null);
   361→          persisted = Math.min(fresh?.size ?? 0, FILE_MAX_BYTES);
   362→          readPos = fresh?.size ?? 0;
   363→          return;
   364→        }
   365→        await rename(FILE, `${FILE}.old`).catch(() => undefined);
   366→        await persistWatermark("shadow-veto-journal");
   367→        persisted = 0;
   368→        readPos = 0;
   369→      });
   370→    }
   371→  } catch {
   372→    /* non-fatal */
   373→  }
   374→}

   385→/** v8.2.1 READ-PATH LIVENESS — tail-ingest another instance's writes (see
   386→ * gate-journal.gateJournalRefresh for the full rationale). Throttled,
   387→ * single-flight, rotation-safe, bounded bite. */
   388→export async function shadowVetoRefresh(maxAgeMs = REFRESH_MIN_INTERVAL_MS): Promise<void> {
   389→  if (Date.now() - lastRefreshAt < maxAgeMs) return;
   390→  if (refreshing) return refreshing;
   391→  refreshing = (async () => {
   392→    try {
   393→      const st = await stat(FILE).catch(() => null);
   394→      if (!st) return;
   395→      if (st.size < readPos) readPos = 0; // rotated
   396→      if (st.size === readPos) return;
   397→      const len = Math.min(st.size - readPos, REFRESH_MAX_BITE);
   398→      const fh = await open(FILE, "r").catch(() => null);
   399→      if (!fh) return;
   400→      try {
   401→        const buf = Buffer.alloc(len);
   402→        const { bytesRead } = await fh.read(buf, 0, len, readPos);
   403→        let text = buf.toString("utf8", 0, bytesRead);
   404→        const lastNl = text.lastIndexOf("\n");
   405→        if (lastNl === -1) return; // partial line — wait for the writer
   406→        text = text.slice(0, lastNl + 1);
   407→        readPos += Buffer.byteLength(text, "utf8");
   408→        for (const line of text.split("\n")) {
   409→          if (!line) continue;
   410→          try {
   411→            const rec = JSON.parse(line) as { k: string; v: unknown };
   412→            if (rec.k === "v") {
   413→              const v = rec.v as ShadowVerdict;
   414→              const m = /^SV-(\d+)$/.exec(v.id ?? "");
   415→              if (ingestVerdict(v) && m) seq = Math.max(seq, Number(m[1]));
   416→            } else if (rec.k === "p") {
   417→              ingestPaper(rec.v as PaperTrade);
   418→            } else if (rec.k === "o") applyOutcomeRecord(rec.v as { id: string; outcome: ShadowOutcome });
   419→            else if (rec.k === "pc") applyPaperClose(rec.v as { id: string; status: PaperTrade["status"]; reachedTp: number; pnlPercent: number; pnlUsd: number; stop: number; outcome: ShadowOutcome; closeNote: string });
   420→          } catch {
   421→            /* torn line */
   422→          }
   423→        }
   424→        hydratedAt = Date.now();
   425→      } finally {
   426→        await fh.close().catch(() => undefined);
   427→      }
   428→    } catch {
   429→      /* non-fatal */
   430→    } finally {
   431→      lastRefreshAt = Date.now();
   432→    }
   433→  })().finally(() => {
   434→    refreshing = null;
   435→  });
   436→  return refreshing;
   437→}

   500→/** Wilson score LOWER bound at 95% (z=1.96) — the A/B evidence statistic. */
   501→export function wilsonLB(wins: number, n: number): number | null {
   502→  if (n <= 0) return null;
   503→  const z = 1.96;
   504→  const p = wins / n;
   505→  const denom = 1 + (z * z) / n;
   506→  const centre = p + (z * z) / (2 * n);
   507→  const spread = z * Math.sqrt((p * (1 - p) + (z * z) / (4 * n)) / n);
   508→  return Math.max(0, ((centre - spread) / denom) * 100);
   509→}

   545→/** Offer a scan batch's fence candidates to the adjudicator. Fire-and-forget,
   546→ * self-budgeted, deduped — never throws, never blocks the scan. */
   547→export async function offerFenceAdjudication(batch: FenceCandidate[]): Promise<void> {
   548→  try {
   549→    const cfg = await getShadowVetoConfig();
   550→    if (cfg.mode === "off" || batch.length === 0 || pendingCount >= MAX_PENDING) return;
   551→    await shadowVetoReady();
   552→    const now = Date.now();
   553→    // closest-to-gate first so the budget buys the most informative verdicts
   554→    const fence = batch
   555→      .map((c) => ({ c, gap: Math.abs(c.score) - c.effMin }))
   556→      .filter(({ c, gap }) => (c.passed ? gap >= 0 && gap <= cfg.windowPass : gap < 0 && gap >= cfg.windowReject))
   557→      .sort((a, b) => Math.abs(a.gap) - Math.abs(b.gap));
   558→    for (const { c, gap } of fence) {
   559→      const key = `${c.symbol}|${c.direction}`;
   560→      const last = lastAdjAt.get(key) ?? 0;
   561→      if (now - last < cfg.dedupMs) continue;
   562→      const cached = verdictCache.get(key);
   563→      if (cached && now - cached.at < cfg.cacheTtlMs) continue; // fresh verdict already in place
   564→      lastAdjAt.set(key, now);
   565→      pendingCount++;
   566→      void runAdjudication(c, c.passed ? "PASS" : "REJECT", gap)
   567→        .catch(() => undefined)
   568→        .finally(() => {
   569→          pendingCount = Math.max(0, pendingCount - 1);
   570→        });
   571→    }
   572→  } catch {
   573→    /* never break the scan */
   574→  }
   575→}

   901→    }
   902→    if (swept > 0) console.log(`[shadow-veto] retro-link: bound=${bound} swept-orphans=${swept}`);
   903→    return bound;
   904→  } catch {
   905→    return 0; // evidence plumbing must never throw into the tick
   906→  }

   961→/** Base geometry for paper trades: slAtrMult × ATR stop (no regime/vol/flow
   962→ * drivers — those are deploy-time only), levered-risk-capped leverage, base RR.
   963→ * Direction-aware. Documented simplification: the paper arm measures
   964→ * DIRECTIONAL edge at the fence, not perfect PnL replication. */
   965→async function paperGeometry(symbol: string, direction: "LONG" | "SHORT", strategy: string, entryOverride: number | null): Promise<PaperGeometry> {
   966→  const gate = await getGateSettings();
   967→  const candles = await fetchKlines(symbol, "15m", 160).catch(() => null);
   968→  const snap = candles ? buildSnapshot(symbol, candles) : null;
   969→  const entry = entryOverride ?? snap?.price ?? 0;
   970→  if (!entry) throw new Error("no price");
   971→  const atrPct = snap?.atrPct ?? 1.5;
   972→  const slDistPct = Math.min(5, Math.max(0.7, gate.slAtrMult * atrPct)) / 100;
   973→  const levRaw = Math.round(Math.min(20, Math.max(5, 60 / (slDistPct * 100))));
   974→  const leverage = gate.leveredRiskCap > 0 && slDistPct * 100 * levRaw > gate.leveredRiskCap ? Math.max(5, Math.floor(gate.leveredRiskCap / (slDistPct * 100))) : levRaw;
   975→  const rr = RR_BASE[strategy] ?? 2.6;
   976→  const tp2Mult = Math.min(2.2, Math.max(1.5, 1.2 + rr * 0.22));
   977→  const [ttlMin, ttlMax] = TTL_BY_STRATEGY[strategy] ?? [90, 180];
   978→  const isLong = direction === "LONG";
   979→  const risk = entry * slDistPct;
   980→  const fx = (v: number) => Number(v.toFixed(entry > 100 ? 2 : 6));
   981→  return {
   982→    entry: fx(entry),
   983→    initialStop: fx(isLong ? entry - risk : entry + risk),
   984→    stop: fx(isLong ? entry - risk : entry + risk),
   985→    tp1: fx(isLong ? entry + risk : entry - risk),
   986→    tp2: fx(isLong ? entry + risk * tp2Mult : entry - risk * tp2Mult),
   987→    tp3: fx(isLong ? entry + risk * rr : entry - risk * rr),
   988→    leverage,
   989→    slDistPct,
   990→    rr,
   991→    ttlMinutes: Math.round(ttlMin + Math.random() * (ttlMax - ttlMin)),
   992→  };
   993→}

  1054→/** Per-tick sweep — engine passes the fresh price map. Mirrors the real
  1055→ * resolver: rung ladders with trailing stops, tranche-weighted closes, fees
  1056→ * via feeBreakdown, TTL expiry. In-memory, cheap, never throws. */
  1057→export async function shadowPaperSweep(priceMap: Map<string, number>): Promise<number> {
  1058→  try {
  1059→    if (papers.length === 0) return 0;
  1060→    const now = Date.now();
  1061→    let closed = 0;
  1062→    for (const p of papers) {
  1063→      if (p.outcome) continue;
  1064→      const price = priceMap.get(p.symbol);
  1065→      if (!price) continue;
  1066→      const isLong = p.direction === "LONG";
  1067→      const hitSl = isLong ? price <= p.stop : price >= p.stop;
  1068→      const hitTp1 = isLong ? price >= p.tp1 : price <= p.tp1;
  1069→      const hitTp2 = isLong ? price >= p.tp2 : price <= p.tp2;
  1070→      const hitTp3 = isLong ? price >= p.tp3 : price <= p.tp3;
  1071→
  1072→      if (hitSl) {
  1073→        closePaper(p, "SL", movePctAt(p, p.stop), now, `stopped @ ${p.stop} (trailed)`);
  1074→        closed++;
  1075→      } else if (hitTp3) {
  1076→        p.reachedTp = 3;
  1077→        closePaper(p, "TP3", movePctAt(p, p.tp3), now, `final target ${p.rr.toFixed(1)}R secured`);
  1078→        closed++;
  1079→      } else if (hitTp2 && p.reachedTp < 2) {
  1080→        p.reachedTp = 2;
  1081→        p.status = "TP2";
  1082→        p.stop = p.tp1; // trail to TP1 — +1R locked (engine semantics)
  1083→      } else if (hitTp1 && p.reachedTp < 1) {
  1084→        p.reachedTp = Math.max(p.reachedTp, 1);
  1085→        p.status = "TP1";
  1086→        p.stop = p.entry; // breakeven trail (engine semantics)
  1087→      } else if (now > p.at + p.ttlMinutes * 60_000) {
  1088→        closePaper(p, "EXPIRED", movePctAt(p, price), now, `TTL ${p.ttlMinutes}m elapsed at mark`);
  1089→        closed++;
  1090→      }
  1091→      // track MFE
  1092→      p.maxPnlPercent = Math.max(p.maxPnlPercent, Number((movePctAt(p, price) * p.leverage).toFixed(2)));
  1093→    }
  1094→    return closed;
  1095→  } catch {
  1096→    return 0;
  1097→  }
  1098→}

  1325→ *           lower bound of the win rate already clears coin-flip odds, so
  1326→ *           waiting for 30 adds latency without adding certainty).

  1329→export function promotionEvidence(arms: ArmRow[]): PromotionEvidence {
  1330→  const realKeep = arms[0]!;
  1331→  const realVeto = arms[1]!;
  1332→  const paperVeto = arms[3]!;
  1333→  const both = realKeep.n >= 30 && realVeto.n >= 30;
  1334→  // v10.11 SEQ-STOP — the sequential early-decision layer (always-valid, OF
  1335→  // alpha-spending). It computes on the SAME arms the fixed gate sees (frozen
  1336→  // when a COHORT-FREEZE exists): a PROMOTE-EARLY state satisfies promotion
  1337→  // WITHOUT waiting for n=30 (min-n guard 12/arm inside), ABORT-EARLY vetoes
  1338→  // the hypothesis outright, CONTINUE leaves the fixed gate as sole path.

  1340→    { wins: realKeep.wins, n: realKeep.n, netUsd: realKeep.netUsd },
  1341→    { wins: realVeto.wins, n: realVeto.n, netUsd: realVeto.netUsd },
  1342→  );
  1343→  const okN = (a: ArmRow): boolean => a.n >= 30 || (a.n >= 20 && (a.wilsonLB ?? 0) >= 50);
  1344→  const nDetail = (role: string, a: ArmRow): string => {
  1345→    const fast = a.n >= 20 && a.n < 30 && (a.wilsonLB ?? 0) >= 50;
  1346→    return `rolling-30 ${role} cohort ${a.n}/30${a.n >= 20 && a.n < 30 ? ` (fast path n≥20 + LB≥50%: ${fast ? "MET" : "unmet — LB " + (a.wilsonLB !== null ? a.wilsonLB.toFixed(1) : "n/a") + "%"})` : ""}`;
  1347→  };
  1348→  const criteria = [
  1349→    { id: "n-veto", ok: okN(realVeto), detail: nDetail("would-veto", realVeto) },
  1350→    { id: "n-keep", ok: okN(realKeep), detail: nDetail("would-keep", realKeep) },
  1351→    { id: "wr-sep", ok: both && (realVeto.wilsonLB ?? 0) < (realKeep.wilsonLB ?? 100), detail: `Wilson-LB veto ${realVeto.wilsonLB !== null ? `${realVeto.wilsonLB.toFixed(1)}%` : "n/a"} < keep ${realKeep.wilsonLB !== null ? `${realKeep.wilsonLB.toFixed(1)}%` : "n/a"} (95% separation)` },

  1353→  ];
  1354→  const legacyReady = criteria.every((c) => c.ok);
  1355→  const ready = legacyReady || sequential.state === "PROMOTE-EARLY";
  1356→  criteria.push({
  1357→    id: "seq-stop",
  1358→    ok: sequential.state === "PROMOTE-EARLY",
  1359→    detail: `${sequential.detail}${legacyReady ? " — legacy gate already satisfied, sequential is confirmatory" : ""}`,
  1360→  });
  1361→  const revPressure = paperVeto.n >= 15 && (paperVeto.wilsonLB ?? 0) >= 55 && paperVeto.netUsd > 0;
  1362→  return {
  1363→    ready,
  1364→    criteria,
  1365→    sequential,
  1366→    revocation: {
  1367→      pressure: revPressure,
  1368→      detail: revPressure
  1369→        ? `blocked-pass counterfactuals are winning without the veto (n=${paperVeto.n}, LB ${paperVeto.wilsonLB!.toFixed(1)}%, net $${paperVeto.netUsd.toFixed(2)}) — consider demoting to shadow`
  1370→        : `no revocation pressure (paper-veto n=${paperVeto.n}/15, LB ${paperVeto.wilsonLB !== null ? `${paperVeto.wilsonLB.toFixed(1)}%` : "n/a"}, net $${paperVeto.netUsd.toFixed(2)})`,
  1371→    },
  1372→  };
  1373→}

  1611→  if (s < 3600) return `${Math.round(s / 60)}m`;

  1633→  const d24 = now - 24 * 3600_000;

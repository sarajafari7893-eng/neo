   706→// v10.14 SQN telemetry (supervisor proposal msg 3920, operator APPROVE
   707→// 2026-09-19T08:56Z msg 861802419): every realizedEdgeFilter evaluation
   708→// records the SQN-style R-normalized statistic of the rolling window next to
   709→// the legacy mean-% statistic. SQN = √N × mean(R)/std(R) on the same
   710→// pnlPercent series — scale-free, outlier-discounted via std, √N-discounted
   711→// by construction. DECISION RULES ARE UNCHANGED (the −3%/−2% floors still
   712→// decide); the SQN column exists so the advisor's future KNOB_ADJUST
   713→// (triggered by the GATE-NEAR-MISS cohort per msg 3920) compares both
   714→// statistics on live data, not opinion. Restart-safe: in-memory mirror only,
   715→// re-populated by the next real gate evaluation.
   716→const mgSqnTelemetry = new Map<string, { sqn: number; sqnTrim: number; n: number; expPct: number; at: number }>();
   717→
   718→function sqnOf(pnls: number[]): { sqn: number; sqnTrim: number } {
   719→  const n = pnls.length;
   720→  const mean = pnls.reduce((a, b) => a + b, 0) / n;
   721→  const sd = Math.sqrt(pnls.reduce((a, b) => a + (b - mean) * (b - mean), 0) / n); // population (ddof 0)
   722→  const sqn = sd > 0 ? Math.sqrt(n) * (mean / sd) : 0;
   723→  const sorted = [...pnls].sort((a, b) => a - b);
   724→  const k = Math.floor(n * 0.1);
   725→  const trim = k > 0 && n - 2 * k >= 5 ? sorted.slice(k, n - k) : sorted;
   726→  const tm = trim.reduce((a, b) => a + b, 0) / trim.length;
   727→  const tsd = Math.sqrt(trim.reduce((a, b) => a + (b - tm) * (b - tm), 0) / trim.length);
   728→  const sqnTrim = tsd > 0 ? Math.sqrt(trim.length) * (tm / tsd) : 0;
   729→  return { sqn: Number(sqn.toFixed(2)), sqnTrim: Number(sqnTrim.toFixed(2)) };
   730→}

   732→/** v10.14 — live SQN-vs-legacy snapshot of the memory-gate rolling windows
   733→ * (what the advisor KNOB_ADJUST will compare). Keys: strategy name; combo
   734→ * windows under `symbol:strategy`. Read by /api/ai/near-miss (gateSqn block). */
   735→export function memoryGateSqnSnapshot(): { strategies: { key: string; n: number; expPct: number; sqn: number; sqnTrim: number; ageMin: number }[]; note: string } {
